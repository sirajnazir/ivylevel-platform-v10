-- Award temporal functions
CREATE OR REPLACE FUNCTION award_nth(
  p_student_id text,
  p_n integer,
  p_as_of date DEFAULT CURRENT_DATE
) RETURNS TABLE(...) AS $$
  -- Return nth award as of given date
$$;

CREATE OR REPLACE FUNCTION award_first(p_student_id text, p_as_of date)
  RETURNS TABLE(...) AS $$
  SELECT * FROM award_nth(p_student_id, 1, p_as_of);
$$;

CREATE OR REPLACE FUNCTION award_latest(p_student_id text, p_as_of date)
  RETURNS TABLE(...) AS $$
  SELECT * FROM kb_items
  WHERE student_id = p_student_id
    AND item_type = 'Award_Competition'
    AND tier1_state = 'Outcome'
    AND outcome_date <= p_as_of
  ORDER BY outcome_date DESC
  LIMIT 1;
$$;

-- GPA temporal functions
CREATE OR REPLACE FUNCTION gpa_as_of(p_student_id text, p_as_of date)
  RETURNS numeric AS $$
  SELECT value::numeric FROM vital_facts
  WHERE student_id = p_student_id
    AND category = 'Academic'
    AND fact_type = 'GPA'
    AND recorded_at <= p_as_of
  ORDER BY recorded_at DESC
  LIMIT 1;
$$;

-- Deadline temporal functions
CREATE OR REPLACE FUNCTION deadline_latest(
  p_student_id text,
  p_category text,
  p_as_of date
) RETURNS date AS $$
  SELECT deadline_date FROM kb_items
  WHERE student_id = p_student_id
    AND item_type = p_category
    AND tier1_state IN ('Planned', 'In Transit')
    AND deadline_date >= p_as_of
  ORDER BY deadline_date ASC
  LIMIT 1;
$$;

### Enhancement 3: Tool Bus + Policy Gate (P0, Wave 2)

**Goal:** Centralize tool execution with governance (scopes, budgets, ACLs).

**Current State:** Agents call `resolverTools.ts` directly, no governance.

**Required Components:**

1. **Tool Manifest Schema:**
```typescript
// src/cognition/tool-manifest.ts
export interface ToolManifest {
  name: string;
  version: string;
  description: string;
  inputs: Record<string, any>;
  outputs: Record<string, any>;
  side_effects: 'none' | 'read' | 'write';
  scopes: string[]; // e.g., ['student:read', 'facts:query']
  cost: {
    estimated_tokens: number;
    estimated_latency_ms: number;
  };
  requires_chips: boolean; // True if must produce chips
}
```

2. **Tool Bus:**
```typescript
// src/cognition/tool-bus.ts
export class ToolBus {
  private registry: Map<string, ToolManifest> = new Map();
  private policyGate: PolicyGate;

  register(manifest: ToolManifest): void {
    this.registry.set(manifest.name, manifest);
  }

  async execute(
    toolName: string,
    params: any,
    context: ExecutionContext
  ): Promise<ToolResult> {
    const manifest = this.registry.get(toolName);
    if (!manifest) throw new Error(`Tool ${toolName} not registered`);

    // Policy check
    await this.policyGate.check(manifest, context);

    // Execute tool
    const startTime = Date.now();
    const result = await this.executeTool(toolName, params, context);
    const latency = Date.now() - startTime;

    // Validate chip requirement
    if (manifest.requires_chips && !result.chip) {
      throw new Error(`Tool ${toolName} must produce a chip`);
    }

    // Log execution
    await this.logExecution(toolName, latency, result.chip, context.trace_id);

    return result;
  }
}
```

3. **Policy Gate:**
```typescript
// src/cognition/policy-gate.ts
export class PolicyGate {
  async check(manifest: ToolManifest, context: ExecutionContext): Promise<void> {
    // 1. Check scopes
    if (!this.hasRequiredScopes(manifest.scopes, context.user_scopes)) {
      throw new Error403(`Missing required scopes: ${manifest.scopes}`);
    }

    // 2. Check budget
    const currentUsage = await this.getCurrentUsage(context.agent_run_id);
    if (currentUsage.tokens + manifest.cost.estimated_tokens > context.token_budget) {
      throw new Error429(`Token budget exceeded`);
    }

    // 3. Check RLS (coach_id isolation)
    if (!this.isCoachAuthorized(context.coach_id, context.student_id)) {
      throw new Error403(`Coach not authorized for this student`);
    }
  }
}
```

2. **EQ Adapter:**
```typescript
// src/cognition/eq-adapter.ts
export class EQAdapter {
  async apply(
    rawText: string,
    coachId: string,
    audience: 'student' | 'parent' | 'coach',
    context: {
      studentId: string;
      traceId: string;
      overrides?: any;
    }
  ): Promise<{ styledText: string; chip: Chip }> {
    // 1. Load EQ profile
    const profile = await this.loadProfile(coachId);

    // 2. Load student-specific overrides
    const tuning = await this.loadTuning(context.studentId, coachId);

    // 3. Apply tone transformation
    const styled = await this.transform(rawText, profile, tuning, audience);

    // 4. Filter banned phrases
    const filtered = this.filterBanned(styled, tuning.banned_phrases);

    // 5. Create EQ chip
    const chip = ChipCreator.createEQChip(
      coachId,
      { tone_vector: profile.tone_vector, audience },
      context.traceId,
      context.studentId
    );

    return { styledText: filtered, chip };
  }

  private async transform(
    text: string,
    profile: EQProfile,
    tuning: Tuning,
    audience: string
  ): Promise<string> {
    // If fine-tuned model exists, use it
    if (profile.model_id) {
      return await this.callFineTunedModel(text, profile.model_id, audience);
    }

    // Otherwise, use tone adapter (lexical substitution + structure)
    return await this.applyToneVector(text, profile.tone_vector, profile.lex_map, audience);
  }
}
```

// src/observability/span-enforcer.ts
export function withSpan<T>(
  operationName: string,
  layer: 'perception' | 'knowledge' | 'cognition' | 'action' | 'interface',
  fn: (span: Span) => Promise<T>
): Promise<T> {
  return tracer.startActiveSpan(operationName, async (span) => {
    span.setAttribute('layer', layer);

    try {
      const result = await fn(span);
      span.setStatus({ code: SpanStatusCode.OK });
      return result;
    } catch (error) {
      span.setStatus({ code: SpanStatusCode.ERROR, message: error.message });
      span.recordException(error);
      throw error;
    } finally {
      span.end();
    }
  });
}

// src/evaluation/quality-scorer.ts
export class QualityScorer {
  async scoreResponse(
    responseId: string,
    agentRunId: string,
    response: string,
    studentId: string
  ): Promise<void> {
    // 1. LLM-based scoring (accuracy, empathy, actionability, clarity)
    const scores = await this.llmScore(response);

    // 2. Rule-based checks (has evidence chips? appropriate length?)
    const ruleScores = await this.ruleBasedScore(responseId);

    // 3. Store all scores
    for (const [dimension, score] of Object.entries({ ...scores, ...ruleScores })) {
      await pool.query(`
        INSERT INTO quality_scores (agent_run_id, response_id, dimension, score, scorer)
        VALUES ($1, $2, $3, $4, $5)
      `, [agentRunId, responseId, dimension, score, score < 0.5 ? 'rule_based' : 'llm']);
    }

    // 4. If any score < 0.7, flag for coach review
    if (Object.values(scores).some(s => s < 0.7)) {
      await this.flagForReview(responseId, scores);
    }
  }

// src/training/playbook-refresher.ts
export class PlaybookRefresher {
  async refresh(): Promise<void> {
    // 1. Calculate average effectiveness per tactic
    const tactics = await pool.query(`
      SELECT tactic_id, AVG(effectiveness_score) as avg_effectiveness
      FROM tactic_effectiveness
      WHERE recorded_at > NOW() - INTERVAL '30 days'
      GROUP BY tactic_id
    `);

    // 2. Update moat_tactic_chips with new effectiveness scores
    for (const tactic of tactics.rows) {
      await pool.query(`
        UPDATE moat_tactic_chips
        SET effectiveness_score = $1, updated_at = NOW()
        WHERE tactic_id = $2
      `, [tactic.avg_effectiveness, tactic.tactic_id]);
    }

    // 3. Promote top tactics, demote low-performing ones
    await this.reorderTacticsByEffectiveness();
  }
}

  agent_run_id uuid,
  response_id uuid,
  dimension text NOT NULL,
  score numeric CHECK (score BETWEEN 0 AND 1),
  scorer text NOT NULL,
  feedback text,
  created_at timestamptz DEFAULT now()
);

-- Tactic Effectiveness
CREATE TABLE tactic_effectiveness (
  tactic_id text,
  student_id text,
  outcome_type text,
  effectiveness_score numeric CHECK (effectiveness_score BETWEEN 0 AND 1),
  recorded_at timestamptz DEFAULT now()
);

CREATE INDEX idx_quality_scores_run ON quality_scores(agent_run_id);
CREATE INDEX idx_tactic_effectiveness_tactic ON tactic_effectiveness(tactic_id, recorded_at);
