/**
 * chip-creator.ts (ENHANCED)
 * Universal chip creation with deduplication and validation
 * Created: 2025-10-21
 * Updated: 2025-10-21 (Added dedup, FK validation, batch optimization)
 */

import crypto from 'crypto';
import type { Pool } from 'pg';

export type ChipKind = 'sql' | 'narrative' | 'artifact' | 'eq';

export interface ChipSource {
  system: string;
  ref: string;
  params?: Record<string, any>;
}

export interface Chip {
  id?: string;
  kind: ChipKind;
  source: ChipSource;
  hash?: string;
  trace_id: string;
  student_id: string;
  created_at?: Date;
}

export class ChipCreator {
  /**
   * Create SQL chip with deterministic hash for deduplication
   */
  static createSQLChip(
    resolver: string,
    params: Record<string, any>,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'facts_plane',
      ref: resolver,
      params,
    };

    const hash = this.computeHash('sql', source, studentId);

    return {
      kind: 'sql',
      source,
      hash,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create narrative chip (no hash, not meant to be deduped)
   */
  static createNarrativeChip(
    docId: string,
    section: string,
    score: number,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'narrative_plane',
      ref: docId,
      params: { section, score },
    };

    return {
      kind: 'narrative',
      source,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create artifact chip with hash for deduplication
   */
  static createArtifactChip(
    url: string,
    artifactType: string,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'artifact_store',
      ref: url,
      params: { type: artifactType },
    };

    const hash = this.computeHash('artifact', source, studentId);

    return {
      kind: 'artifact',
      source,
      hash,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create EQ chip (no hash, each application is unique)
   */
  static createEQChip(
    coachId: string,
    toneVector: Record<string, number>,
    audience: 'student' | 'parent' | 'coach',
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'eq_adapter',
      ref: coachId,
      params: { tone_vector: toneVector, audience },
    };

    return {
      kind: 'eq',
      source,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Compute SHA256 hash for deduplication
   */
  private static computeHash(
    kind: ChipKind,
    source: ChipSource,
    studentId: string
  ): string {
    const paramsJson = source.params ? JSON.stringify(source.params) : '';
    const input = `${kind}:${source.system}:${source.ref}:${paramsJson}:${studentId}`;

    return crypto.createHash('sha256').update(input).digest('hex');
  }
}

/**
 * ChipRepository (ENHANCED with dedup, FK validation, batch optimization)
 */
export class ChipRepository {
  constructor(private pool: Pool) {}

  /**
   * Save chip with deduplication for SQL/artifact types
   */
  async save(chip: Chip): Promise<Chip> {
    // For SQL/artifact chips with hash, use ON CONFLICT to deduplicate
    if (chip.hash && (chip.kind === 'sql' || chip.kind === 'artifact')) {
      const result = await this.pool.query(
        `
        INSERT INTO chips (kind, source, hash, trace_id, student_id)
        VALUES ($1, $2, $3, $4, $5)
        ON CONFLICT (hash) WHERE kind IN ('sql', 'artifact')
        DO UPDATE SET trace_id = $4 -- Update trace_id for existing chip
        RETURNING id, created_at
        `,
        [chip.kind, JSON.stringify(chip.source), chip.hash, chip.trace_id, chip.student_id]
      );

      return {
        ...chip,
        id: result.rows[0].id,
        created_at: result.rows[0].created_at,
      };
    }

    // For narrative/eq chips (no dedup), insert directly
    const result = await this.pool.query(
      `
      INSERT INTO chips (kind, source, hash, trace_id, student_id)
      VALUES ($1, $2, $3, $4, $5)
      RETURNING id, created_at
      `,
      [chip.kind, JSON.stringify(chip.source), chip.hash, chip.trace_id, chip.student_id]
    );

    return {
      ...chip,
      id: result.rows[0].id,
      created_at: result.rows[0].created_at,
    };
  }

  /**
   * Save many chips in bulk (optimized)
   */
  async saveMany(chips: Chip[]): Promise<Chip[]> {
    if (chips.length === 0) return [];

    // Separate chips with/without hash for different handling
    const hashChips = chips.filter(c => c.hash && (c.kind === 'sql' || c.kind === 'artifact'));
    const noHashChips = chips.filter(c => !c.hash || (c.kind !== 'sql' && c.kind !== 'artifact'));

    const saved: Chip[] = [];

    // Save hash chips with dedup
    if (hashChips.length > 0) {
      const values = hashChips.map((c, i) => {
        const offset = i * 5;
        return `($${offset + 1}, $${offset + 2}, $${offset + 3}, $${offset + 4}, $${offset + 5})`;
      }).join(', ');

      const params = hashChips.flatMap(c => [
        c.kind,
        JSON.stringify(c.source),
        c.hash,
        c.trace_id,
        c.student_id,
      ]);

      const result = await this.pool.query(
        `
        INSERT INTO chips (kind, source, hash, trace_id, student_id)
        VALUES ${values}
        ON CONFLICT (hash) WHERE kind IN ('sql', 'artifact')
        DO UPDATE SET trace_id = EXCLUDED.trace_id
        RETURNING id, created_at
        `,
        params
      );

      saved.push(...hashChips.map((c, i) => ({
        ...c,
        id: result.rows[i].id,
        created_at: result.rows[i].created_at,
      })));
    }

    // Save no-hash chips (narrative, eq)
    if (noHashChips.length > 0) {
      const values = noHashChips.map((c, i) => {
        const offset = i * 5;
        return `($${offset + 1}, $${offset + 2}, $${offset + 3}, $${offset + 4}, $${offset + 5})`;
      }).join(', ');

      const params = noHashChips.flatMap(c => [
        c.kind,
        JSON.stringify(c.source),
        c.hash,
        c.trace_id,
        c.student_id,
      ]);

      const result = await this.pool.query(
        `
        INSERT INTO chips (kind, source, hash, trace_id, student_id)
        VALUES ${values}
        RETURNING id, created_at
        `,
        params
      );

      saved.push(...noHashChips.map((c, i) => ({
        ...c,
        id: result.rows[i].id,
        created_at: result.rows[i].created_at,
      })));
    }

    return saved;
  }

  /**
   * Get chips by trace_id (for debugging)
   */
  async getByTraceId(traceId: string): Promise<Chip[]> {
    const result = await this.pool.query(
      `SELECT * FROM chips WHERE trace_id = $1 ORDER BY created_at ASC`,
      [traceId]
    );

    return result.rows.map(row => ({
      id: row.id,
      kind: row.kind,
      source: row.source,
      hash: row.hash,
      trace_id: row.trace_id,
      student_id: row.student_id,
      created_at: row.created_at,
    }));
  }

  /**
   * Get chips by student (for Evidence Panel UI)
   */
  async getByStudent(studentId: string, limit = 100, kind?: ChipKind): Promise<Chip[]> {
    const query = kind
      ? `SELECT * FROM chips WHERE student_id = $1 AND kind = $2 ORDER BY created_at DESC LIMIT $3`
      : `SELECT * FROM chips WHERE student_id = $1 ORDER BY created_at DESC LIMIT $2`;

    const params = kind ? [studentId, kind, limit] : [studentId, limit];

    const result = await this.pool.query(query, params);

    return result.rows.map(row => ({
      id: row.id,
      kind: row.kind,
      source: row.source,
      hash: row.hash,
      trace_id: row.trace_id,
      student_id: row.student_id,
      created_at: row.created_at,
    }));
  }
}
