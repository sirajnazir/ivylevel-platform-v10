#### 1.1 Chain-of-Thought (CoT) Integration

**What:** Explicit step-by-step reasoning before final answer.

**Implementation:**

```typescript
// services/agent-framework/src/reasoning/CoTProcessor.ts

export interface CoTStep {
  step_number: number;
  thought: string;           // What I'm thinking
  reasoning: string;          // Why I'm thinking this
  evidence: string[];         // Which facts support this
  confidence: number;         // 0-1 score
}

export interface CoTResponse {
  steps: CoTStep[];
  final_answer: string;
  total_confidence: number;
  reasoning_trace_id: string;
}

export class CoTProcessor {
  /**
   * Generate Chain-of-Thought reasoning for complex queries
   * Based on: Agentic Design Patterns Chapter 17 (Reasoning Techniques)
   */
  async processWithCoT(
    query: string,
    context: StudentContext,
    agentRole: string
  ): Promise<CoTResponse> {

    // Step 1: Decompose query into reasoning steps
    const decompositionPrompt = `
You are ${agentRole} helping ${context.student_name}.

Query: "${query}"

Break this down into logical reasoning steps. For each step:
1. State what you're thinking
2. Explain why this thought is relevant
3. Identify which facts/data support it
4. Rate your confidence (0-1)

Return JSON with format: {steps: [...]}
`;

    const decomposition = await this.openai.chat.completions.create({
      model: 'o1-preview',  // Use reasoning model
      messages: [
        { role: 'user', content: decompositionPrompt }
      ],
      response_format: { type: 'json_object' }
    });

    const cotSteps: CoTStep[] = JSON.parse(decomposition.choices[0].message.content).steps;

    // Step 2: For each reasoning step, ground in SQL facts
    for (const step of cotSteps) {
      const facts = await this.retrieveEvidenceForStep(step, context);
      step.evidence = facts.map(f => f.fact_text);
    }

    // Step 3: Generate final answer based on reasoning chain
    const synthesisPrompt = `
Based on this reasoning chain:
${JSON.stringify(cotSteps, null, 2)}

Provide a final, comprehensive answer to: "${query}"

Ensure answer is:
- Grounded in facts from reasoning steps
- Actionable for student
- Maintains ${agentRole} voice and expertise
`;

    const synthesis = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [
        { role: 'user', content: synthesisPrompt }
      ]
    });

    const finalAnswer = synthesis.choices[0].message.content;

    // Step 4: Calculate total confidence (average of step confidences)
    const totalConfidence = cotSteps.reduce((sum, s) => sum + s.confidence, 0) / cotSteps.length;

    // Step 5: Store reasoning trace for observability
    const traceId = await this.storeReasoningTrace({
      query,
      steps: cotSteps,
      final_answer: finalAnswer,
      confidence: totalConfidence,
      agent_role: agentRole,
      student_id: context.student_id
    });

    return {
      steps: cotSteps,
      final_answer: finalAnswer,
      total_confidence: totalConfidence,
      reasoning_trace_id: traceId
    };
  }

  private async retrieveEvidenceForStep(
    step: CoTStep,
    context: StudentContext
  ): Promise<VitalFact[]> {
    // Query vital_facts table for supporting evidence
    const result = await pool.query(`
      SELECT fact_text, fact_type, source_reference
      FROM vital_facts
      WHERE student_id = $1
        AND embedding <-> (
          SELECT embedding FROM generate_embedding($2)
        ) < 0.3
      LIMIT 3
    `, [context.student_id, step.thought]);

    return result.rows;
  }

  private async storeReasoningTrace(trace: any): Promise<string> {
    const traceId = uuidv4();

    await pool.query(`
      INSERT INTO reasoning_traces (
        trace_id, student_id, query, steps, final_answer,
        confidence, agent_role, created_at
      )
      VALUES ($1, $2, $3, $4::jsonb, $5, $6, $7, NOW())
    `, [
      traceId,
      trace.student_id,
      trace.query,
      JSON.stringify(trace.steps),
      trace.final_answer,
      trace.confidence,
      trace.agent_role
    ]);

    return traceId;
  }
}
```

#### 1.2 ReAct (Reasoning + Acting) Loop

**What:** Interleave reasoning with tool execution, reflecting after each action.

**Implementation:**

```typescript
// services/agent-framework/src/reasoning/ReActLoop.ts

export interface ReActStep {
  type: 'thought' | 'action' | 'observation' | 'reflection';
  content: string;
  tool_name?: string;
  tool_args?: Record<string, any>;
  tool_result?: any;
  timestamp: Date;
}

export class ReActLoop {
  /**
   * Execute ReAct loop: Think → Act → Observe → Reflect → Repeat
   * Based on: Agentic Design Patterns Chapter 1 (ReAct Pattern)
   */
  async executeReAct(
    query: string,
    context: StudentContext,
    maxIterations: number = 5
  ): Promise<{
    steps: ReActStep[];
    final_answer: string;
    iterations: number;
  }> {

    const steps: ReActStep[] = [];
    let currentQuery = query;
    let iterations = 0;

    while (iterations < maxIterations) {
      iterations++;

      // THOUGHT: What should I do next?
      const thought = await this.generateThought(currentQuery, context, steps);
      steps.push({
        type: 'thought',
        content: thought.reasoning,
        timestamp: new Date()
      });

      // Decision: Do I need to use a tool, or can I answer now?
      if (thought.needs_tool) {
        // ACTION: Execute tool
        const action = await this.executeAction(thought.tool_name, thought.tool_args, context);
        steps.push({
          type: 'action',
          content: `Calling ${thought.tool_name} with args: ${JSON.stringify(thought.tool_args)}`,
          tool_name: thought.tool_name,
          tool_args: thought.tool_args,
          timestamp: new Date()
        });

        // OBSERVATION: What did the tool return?
        steps.push({
          type: 'observation',
          content: `Tool returned: ${JSON.stringify(action.result)}`,
          tool_result: action.result,
          timestamp: new Date()
        });

        // REFLECTION: What does this mean for my answer?
        const reflection = await this.generateReflection(query, steps, action.result);
        steps.push({
          type: 'reflection',
          content: reflection.insight,
          timestamp: new Date()
        });

        // Update query for next iteration if needed
        if (!reflection.is_sufficient) {
          currentQuery = reflection.next_question;
        } else {
          // We have enough information, synthesize final answer
          break;
        }
      } else {
        // No tool needed, we can answer
        break;
      }
    }

    // Final synthesis
    const finalAnswer = await this.synthesizeFinalAnswer(query, steps);

    return {
      steps,
      final_answer: finalAnswer,
      iterations
    };
  }

  private async generateThought(
    query: string,
    context: StudentContext,
    priorSteps: ReActStep[]
  ): Promise<{
    reasoning: string;
    needs_tool: boolean;
    tool_name?: string;
    tool_args?: Record<string, any>;
  }> {
    const prompt = `
You are reasoning about how to answer: "${query}"

Student: ${context.student_name}
Prior steps: ${JSON.stringify(priorSteps.slice(-3))}

THINK: What should I do next?
- Do I need more information? (If yes, which tool should I call?)
- Can I answer with what I know?

Return JSON: {
  reasoning: "Your thought process",
  needs_tool: true/false,
  tool_name: "tool_name" (if needs_tool),
  tool_args: {...} (if needs_tool)
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: prompt }],
      response_format: { type: 'json_object' }
    });

    return JSON.parse(response.choices[0].message.content);
  }

  private async executeAction(
    toolName: string,
    toolArgs: Record<string, any>,
    context: StudentContext
  ): Promise<{ result: any }> {
    // Use existing tool execution from BaseAgent
    const result = await executeResolverTool(toolName, {
      ...toolArgs,
      student_id: context.student_id
    });

    return { result };
  }

  private async generateReflection(
    originalQuery: string,
    steps: ReActStep[],
    latestResult: any
  ): Promise<{
    insight: string;
    is_sufficient: boolean;
    next_question?: string;
  }> {
    const prompt = `
Original query: "${originalQuery}"
Latest observation: ${JSON.stringify(latestResult)}

REFLECT:
1. What did I learn from this observation?
2. Is this sufficient to answer the original query?
3. If not, what else do I need to know?

Return JSON: {
  insight: "What I learned",
  is_sufficient: true/false,
  next_question: "What I still need to know" (if not sufficient)
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: prompt }],
      response_format: { type: 'json_object' }
    });

    return JSON.parse(response.choices[0].message.content);
  }

  private async synthesizeFinalAnswer(
    query: string,
    steps: ReActStep[]
  ): Promise<string> {
    const prompt = `
Query: "${query}"

Reasoning process:
${steps.map((s, i) => `${i+1}. [${s.type.toUpperCase()}] ${s.content}`).join('\n')}

Based on this reasoning chain, provide a final, comprehensive answer.
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: prompt }]
    });

    return response.choices[0].message.content;
  }
}
```

#### 1.3 Self-Critique Loop

**What:** Agent evaluates its own response and iterates if quality is insufficient.

**Implementation:**

```typescript
// services/agent-framework/src/reasoning/SelfCritique.ts

export interface CritiqueResult {
  quality_score: number;        // 0-10
  issues: Array<{
    type: 'factual_error' | 'incomplete' | 'unclear' | 'off_topic' | 'unsafe';
    description: string;
    severity: 'low' | 'medium' | 'high';
  }>;
  is_acceptable: boolean;
  suggestions: string[];
}

export class SelfCritique {
  private readonly QUALITY_THRESHOLD = 7.5;

  /**
   * Self-critique loop: Generate → Critique → Improve → Repeat
   * Based on: Agentic Design Patterns Chapter 4 (Reflection Pattern)
   */
  async generateWithCritique(
    query: string,
    context: StudentContext,
    maxIterations: number = 3
  ): Promise<{
    final_answer: string;
    iterations: number;
    critique_history: CritiqueResult[];
  }> {

    let currentAnswer = '';
    let iterations = 0;
    const critiqueHistory: CritiqueResult[] = [];

    while (iterations < maxIterations) {
      iterations++;

      // GENERATE: Create or improve answer
      if (iterations === 1) {
        currentAnswer = await this.generateInitialAnswer(query, context);
      } else {
        const lastCritique = critiqueHistory[critiqueHistory.length - 1];
        currentAnswer = await this.improveAnswer(
          query,
          currentAnswer,
          lastCritique.issues,
          lastCritique.suggestions
        );
      }

      // CRITIQUE: Evaluate quality
      const critique = await this.critiqueAnswer(query, currentAnswer, context);
      critiqueHistory.push(critique);

      // DECIDE: Is quality sufficient?
      if (critique.is_acceptable || iterations >= maxIterations) {
        break;
      }
    }

    return {
      final_answer: currentAnswer,
      iterations,
      critique_history: critiqueHistory
    };
  }

  private async generateInitialAnswer(
    query: string,
    context: StudentContext
  ): Promise<string> {
    // Use existing agent logic to generate first draft
    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [
        { role: 'system', content: `You are an expert college admissions coach helping ${context.student_name}.` },
        { role: 'user', content: query }
      ]
    });

    return response.choices[0].message.content;
  }

  private async critiqueAnswer(
    query: string,
    answer: string,
    context: StudentContext
  ): Promise<CritiqueResult> {
    const critiquePrompt = `
You are a quality assurance specialist reviewing coaching advice.

Student Query: "${query}"
Agent Response: "${answer}"

Evaluate this response on:
1. Factual accuracy (is it grounded in student data?)
2. Completeness (does it fully answer the query?)
3. Clarity (is it understandable for a high school student?)
4. Relevance (does it stay on topic?)
5. Safety (is it appropriate for a minor?)

Return JSON:
{
  quality_score: 0-10,
  issues: [
    {type: "factual_error" | "incomplete" | "unclear" | "off_topic" | "unsafe",
     description: "specific issue",
     severity: "low" | "medium" | "high"}
  ],
  is_acceptable: true/false (quality_score >= 7.5),
  suggestions: ["how to improve"]
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: critiquePrompt }],
      response_format: { type: 'json_object' }
    });

    const critique: CritiqueResult = JSON.parse(response.choices[0].message.content);
    critique.is_acceptable = critique.quality_score >= this.QUALITY_THRESHOLD;

    return critique;
  }

  private async improveAnswer(
    query: string,
    currentAnswer: string,
    issues: CritiqueResult['issues'],
    suggestions: string[]
  ): Promise<string> {
    const improvementPrompt = `
Original query: "${query}"
Current answer: "${currentAnswer}"

Issues identified:
${issues.map(i => `- [${i.severity.toUpperCase()}] ${i.type}: ${i.description}`).join('\n')}

Suggestions for improvement:
${suggestions.map((s, i) => `${i+1}. ${s}`).join('\n')}

Rewrite the answer to address all issues and incorporate suggestions.
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: improvementPrompt }]
    });

    return response.choices[0].message.content;
  }
}
```

```typescript
// services/agent-framework/src/core/BaseAgent.ts

export abstract class BaseAgent {
  protected cotProcessor: CoTProcessor;
  protected reactLoop: ReActLoop;
  protected selfCritique: SelfCritique;

  constructor(manifest: AgentManifest) {
    // ... existing code ...

    // NEW: Initialize reasoning components
    this.cotProcessor = new CoTProcessor();
    this.reactLoop = new ReActLoop();
    this.selfCritique = new SelfCritique();
  }

  async execute(context: AgentExecutionContext): Promise<AgentExecutionResult> {
    // Determine complexity of query
    const complexity = await this.assessQueryComplexity(context.user_message);

    let response: string;
    let reasoningTrace: any;

    if (complexity === 'high') {
      // Use CoT for complex strategic questions
      const cotResult = await this.cotProcessor.processWithCoT(
        context.user_message,
        context.session.context,
        this.manifest.agent_id
      );

      response = cotResult.final_answer;
      reasoningTrace = { type: 'cot', steps: cotResult.steps, trace_id: cotResult.reasoning_trace_id };

      // If confidence is low, apply self-critique
      if (cotResult.total_confidence < 0.75) {
        const critiqueResult = await this.selfCritique.generateWithCritique(
          context.user_message,
          context.session.context
        );
        response = critiqueResult.final_answer;
        reasoningTrace.critique = critiqueResult.critique_history;
      }

    } else if (complexity === 'medium') {
      // Use ReAct for queries needing data lookup + reasoning
      const reactResult = await this.reactLoop.executeReAct(
        context.user_message,
        context.session.context
      );

      response = reactResult.final_answer;
      reasoningTrace = { type: 'react', steps: reactResult.steps };

    } else {
      // Use traditional flow for simple queries
      response = await this.generateSimpleResponse(context);
      reasoningTrace = { type: 'simple' };
    }

    return {
      answer: response,
      reasoning_trace: reasoningTrace,
      // ... existing fields ...
    };
  }

  private async assessQueryComplexity(query: string): Promise<'low' | 'medium' | 'high'> {
    // Use existing complexity scoring from v14.0
    // services/agent-framework/src/orchestrator/agentChat-utfa.ts:156-222
    const score = await this.scoreQueryComplexity(query);

    if (score > 7) return 'high';
    if (score > 4) return 'medium';
    return 'low';
  }
}
```

```typescript
// services/agent-framework/src/collaboration/DebateMode.ts

export interface DebateRound {
  round_number: number;
  proposals: Array<{
    agent_id: string;
    proposal: string;
    supporting_evidence: string[];
    confidence: number;
  }>;
  critiques: Array<{
    critic_agent_id: string;
    target_proposal_agent_id: string;
    critique: string;
    severity: 'minor' | 'major' | 'critical';
  }>;
}

export interface DebateResult {
  rounds: DebateRound[];
  consensus_proposal: string;
  winning_agent_id: string;
  consensus_confidence: number;
  dissenting_views: string[];
}

export class DebateMode {
  /**
   * Multi-agent debate with iterative critique and convergence
   * Based on: Multi-Agent Debate pattern (Chapter 7)
   */
  async executeDebate(context: CollaborationContext): Promise<DebateResult> {
    const maxRounds = context.max_rounds || 3;
    const rounds: DebateRound[] = [];

    // Round 1: Initial proposals
    const initialProposals = await this.collectInitialProposals(context);
    rounds.push({
      round_number: 1,
      proposals: initialProposals,
      critiques: []
    });

    // Rounds 2-N: Critique and refine
    for (let i = 2; i <= maxRounds; i++) {
      const critiques = await this.collectCritiques(
        context,
        rounds[i-2].proposals
      );

      const refinedProposals = await this.refineProposals(
        context,
        rounds[i-2].proposals,
        critiques
      );

      rounds.push({
        round_number: i,
        proposals: refinedProposals,
        critiques
      });

      // Check for convergence
      const hasConverged = await this.checkConvergence(refinedProposals);
      if (hasConverged) break;
    }

    // Final synthesis: Select best proposal
    const finalRound = rounds[rounds.length - 1];
    const consensus = await this.synthesizeConsensus(context, finalRound);

    return {
      rounds,
      consensus_proposal: consensus.proposal,
      winning_agent_id: consensus.winning_agent_id,
      consensus_confidence: consensus.confidence,
      dissenting_views: consensus.dissenting_views
    };
  }

  private async collectInitialProposals(
    context: CollaborationContext
  ): Promise<DebateRound['proposals']> {
    const proposals = [];

    for (const agentId of context.participating_agents) {
      const agent = this.registry.getAgent(agentId);

      const proposalPrompt = `
You are ${agent.manifest.display_name} in a team debate.

Student Query: "${context.query}"

Generate your INITIAL PROPOSAL for how to answer this query.

Include:
1. Your proposed answer/strategy
2. Supporting evidence (cite student facts)
3. Your confidence level (0-1)

Be specific and actionable. This is round 1 of debate - other agents will critique your proposal.

Return JSON: {proposal: "...", supporting_evidence: [...], confidence: 0.0-1.0}
`;

      const response = await this.openai.chat.completions.create({
        model: 'gpt-4o',
        messages: [
          { role: 'system', content: agent.manifest.system_prompt },
          { role: 'user', content: proposalPrompt }
        ],
        response_format: { type: 'json_object' }
      });

      const { proposal, supporting_evidence, confidence } = JSON.parse(response.choices[0].message.content);

      proposals.push({
        agent_id: agentId,
        proposal,
        supporting_evidence,
        confidence
      });
    }

    return proposals;
  }

  private async collectCritiques(
    context: CollaborationContext,
    proposals: DebateRound['proposals']
  ): Promise<DebateRound['critiques']> {
    const critiques = [];

    // Each agent critiques other agents' proposals
    for (const criticAgentId of context.participating_agents) {
      for (const targetProposal of proposals) {
        if (targetProposal.agent_id === criticAgentId) continue; // Don't critique self

        const agent = this.registry.getAgent(criticAgentId);

        const critiquePrompt = `
You are ${agent.manifest.display_name} reviewing a colleague's proposal.

Student Query: "${context.query}"

Colleague's Proposal (by ${targetProposal.agent_id}):
"${targetProposal.proposal}"

Evidence cited: ${targetProposal.supporting_evidence.join(', ')}

CRITIQUE this proposal:
1. What are potential flaws or gaps?
2. What assumptions might be wrong?
3. What alternatives would work better?

Be constructive but rigorous.

Return JSON: {
  critique: "Your critique",
  severity: "minor" | "major" | "critical"
}
`;

        const response = await this.openai.chat.completions.create({
          model: 'gpt-4o',
          messages: [
            { role: 'system', content: agent.manifest.system_prompt },
            { role: 'user', content: critiquePrompt }
          ],
          response_format: { type: 'json_object' }
        });

        const { critique, severity } = JSON.parse(response.choices[0].message.content);

        critiques.push({
          critic_agent_id: criticAgentId,
          target_proposal_agent_id: targetProposal.agent_id,
          critique,
          severity
        });
      }
    }

    return critiques;
  }

  private async refineProposals(
    context: CollaborationContext,
    originalProposals: DebateRound['proposals'],
    critiques: DebateRound['critiques']
  ): Promise<DebateRound['proposals']> {
    const refinedProposals = [];

    for (const original of originalProposals) {
      // Get all critiques targeting this proposal
      const relevantCritiques = critiques.filter(
        c => c.target_proposal_agent_id === original.agent_id
      );

      if (relevantCritiques.length === 0) {
        // No critiques, keep original
        refinedProposals.push(original);
        continue;
      }

      const agent = this.registry.getAgent(original.agent_id);

      const refinePrompt = `
You are ${agent.manifest.display_name}.

Your original proposal:
"${original.proposal}"

Critiques received:
${relevantCritiques.map(c =>
  `- [${c.severity.toUpperCase()}] from ${c.critic_agent_id}: ${c.critique}`
).join('\n')}

REFINE your proposal to address the critiques.

Return JSON: {proposal: "refined proposal", supporting_evidence: [...], confidence: 0.0-1.0}
`;

      const response = await this.openai.chat.completions.create({
        model: 'gpt-4o',
        messages: [
          { role: 'system', content: agent.manifest.system_prompt },
          { role: 'user', content: refinePrompt }
        ],
        response_format: { type: 'json_object' }
      });

      const { proposal, supporting_evidence, confidence } = JSON.parse(response.choices[0].message.content);

      refinedProposals.push({
        agent_id: original.agent_id,
        proposal,
        supporting_evidence,
        confidence
      });
    }

    return refinedProposals;
  }

  private async checkConvergence(proposals: DebateRound['proposals']): Promise<boolean> {
    // Check if proposals are becoming similar (convergence)
    const proposalTexts = proposals.map(p => p.proposal);

    const similarityPrompt = `
Compare these proposals and determine if they've converged (saying similar things):

${proposalTexts.map((p, i) => `Proposal ${i+1}: "${p}"`).join('\n\n')}

Return JSON: {
  has_converged: true/false,
  similarity_score: 0.0-1.0,
  reasoning: "why"
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: similarityPrompt }],
      response_format: { type: 'json_object' }
    });

    const { has_converged } = JSON.parse(response.choices[0].message.content);
    return has_converged;
  }

  private async synthesizeConsensus(
    context: CollaborationContext,
    finalRound: DebateRound
  ): Promise<{
    proposal: string;
    winning_agent_id: string;
    confidence: number;
    dissenting_views: string[];
  }> {
    const synthesisPrompt = `
You are a meta-agent synthesizing a multi-agent debate.

Student Query: "${context.query}"

Final proposals:
${finalRound.proposals.map(p =>
  `- ${p.agent_id} (confidence: ${p.confidence}): "${p.proposal}"`
).join('\n')}

Synthesize the BEST final answer by:
1. Identifying consensus areas
2. Resolving disagreements (choose best approach)
3. Combining strengths of all proposals

Return JSON: {
  consensus_proposal: "final synthesized answer",
  winning_agent_id: "agent whose approach was most adopted",
  confidence: 0.0-1.0,
  dissenting_views: ["any unresolved disagreements"]
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: synthesisPrompt }],
      response_format: { type: 'json_object' }
    });

    return JSON.parse(response.choices[0].message.content);
  }
}
```

```typescript
export class DelegateMode {

   * Dynamic delegation based on agent confidence bids
   * Based on: Delegation pattern (Chapter 7)
   */
  async executeDelegate(context: CollaborationContext): Promise<{
    selected_agent_id: string;
    bids: AgentBid[];
    response: AgentResponse;

    const bids = await this.collectBids(context);

      current.confidence > best.confidence ? current : best

      user_message: context.query,
      session: await this.getSession(context.session_id),
      agent_manifest: agent.getManifest()

    await this.storeDelegationRecord(context, bids, selectedAgent);

      selected_agent_id: selectedAgent.agent_id,
      bids,
      response
    };
```

### Feature 3: Safety Guardrails + Crisis Detection

**Problem:** No input validation for unsafe/crisis queries; risk with minors.

**Solution:** Multi-layered safety system with crisis detection, prompt injection defense, PII redaction.

#### 3.1 Input Safety Classifier

```typescript
// services/agent-framework/src/safety/InputSafetyClassifier.ts

export interface SafetyClassification {
  is_safe: boolean;
  concerns: Array<{
    type: 'crisis' | 'prompt_injection' | 'off_topic' | 'inappropriate' | 'competitor';
    severity: 'low' | 'medium' | 'high' | 'critical';
    description: string;
  }>;
  recommended_action: 'proceed' | 'escalate_human' | 'reject';
  escalation_urgency?: 'immediate' | 'within_1_hour' | 'within_24_hours';
}

export class InputSafetyClassifier {
  /**
   * Multi-layered input safety classification
   * Based on: Guardrails pattern (Chapter 18)
   */
  async classifyInput(
    userMessage: string,
    context: StudentContext
  ): Promise<SafetyClassification> {

    const concerns = [];

    // Layer 1: Crisis detection (highest priority)
    const crisisSignals = await this.detectCrisisSignals(userMessage);
    if (crisisSignals.detected) {
      concerns.push({
        type: 'crisis',
        severity: 'critical',
        description: crisisSignals.description
      });

      return {
        is_safe: false,
        concerns,
        recommended_action: 'escalate_human',
        escalation_urgency: 'immediate'
      };
    }

    // Layer 2: Prompt injection detection
    const promptInjection = this.detectPromptInjection(userMessage);
    if (promptInjection.detected) {
      concerns.push({
        type: 'prompt_injection',
        severity: 'high',
        description: promptInjection.pattern
      });
    }

    // Layer 3: Topic appropriateness
    const topicCheck = await this.checkTopicAppropriateness(userMessage);
    if (!topicCheck.is_appropriate) {
      concerns.push({
        type: topicCheck.reason,
        severity: 'medium',
        description: topicCheck.explanation
      });
    }

    // Determine final action
    const hasCritical = concerns.some(c => c.severity === 'critical');
    const hasHigh = concerns.some(c => c.severity === 'high');

    return {
      is_safe: concerns.length === 0,
      concerns,
      recommended_action: hasCritical ? 'escalate_human' :
                          hasHigh ? 'escalate_human' :
                          concerns.length > 0 ? 'proceed' : 'proceed',
      escalation_urgency: hasCritical ? 'immediate' : hasHigh ? 'within_1_hour' : undefined
    };
  }

  private async detectCrisisSignals(message: string): Promise<{
    detected: boolean;
    description: string;
  }> {
    // Pattern-based detection (fast first pass)
    const crisisPatterns = [
      /\b(suicide|kill myself|end my life|self harm|cut myself)\b/i,
      /\b(abuse|being hurt|unsafe at home)\b/i,
      /\b(depressed|hopeless|can't take it anymore)\b/i
    ];

    for (const pattern of crisisPatterns) {
      if (pattern.test(message)) {
        return {
          detected: true,
          description: `Crisis keyword detected: ${pattern.source}`
        };
      }
    }

    // LLM-based nuance detection
    const crisisPrompt = `
Analyze this message from a high school student for crisis signals:

"${message}"

Crisis signals include:
- Suicidal ideation (explicit or implicit)
- Self-harm mentions
- Abuse at home
- Severe depression/hopelessness
- Immediate safety concerns

Return JSON: {
  detected: true/false,
  severity: "none" | "low" | "medium" | "high" | "critical",
  explanation: "what you detected"
}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: crisisPrompt }],
      response_format: { type: 'json_object' }
    });

    const { detected, severity, explanation } = JSON.parse(response.choices[0].message.content);

    return {
      detected: severity === 'critical' || severity === 'high',
      description: explanation
    };
  }

  private detectPromptInjection(message: string): {
    detected: boolean;
    pattern: string;
  } {
    const injectionPatterns = [
      { pattern: /ignore (previous|all) (instructions|prompts)/i, desc: 'Instruction override attempt' },
      { pattern: /you are now/i, desc: 'Role redefinition attempt' },
      { pattern: /new role:/i, desc: 'Role injection attempt' },
      { pattern: /system message:/i, desc: 'System message injection' },
      { pattern: /\[SYSTEM\]/i, desc: 'System tag injection' },
      { pattern: /\<\|im_start\|\>/i, desc: 'Delimiter injection' }
    ];

    for (const { pattern, desc } of injectionPatterns) {
```

```typescript
export interface FilteredOutput {
  original_text: string;
  filtered_text: string;
  redactions: Array<{
    type: 'pii' | 'inappropriate' | 'competitor_mention';
    original: string;
    replacement: string;
    position: number;
  }>;
  is_modified: boolean;
}

export class OutputSafetyFilter {
  /**
   * Filter agent outputs for PII, inappropriate content
   * Based on: Safety patterns (Chapter 18)
   */
  async filterOutput(agentResponse: string): Promise<FilteredOutput> {
    let filtered = agentResponse;
    const redactions = [];

    // PII detection and redaction
    const piiRedactions = this.detectAndRedactPII(filtered);
    for (const redaction of piiRedactions) {
      filtered = filtered.replace(redaction.original, redaction.replacement);
      redactions.push(redaction);
    }

    // Inappropriate content check
    const inappropriateCheck = await this.checkInappropriateContent(filtered);
    if (inappropriateCheck.detected) {
      // Log but don't auto-replace (requires human review)
      log.warn('Inappropriate content detected in agent response', {
        response: filtered.substring(0, 100),
        reason: inappropriateCheck.reason
      });
    }

    return {
      original_text: agentResponse,
      filtered_text: filtered,
      redactions,
      is_modified: redactions.length > 0
    };
  }

  private detectAndRedactPII(text: string): Array<{
    type: 'pii';
    original: string;
    replacement: string;
    position: number;
  }> {
    const redactions = [];

    // Email addresses
    const emailRegex = /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b/g;
    let match;
    while ((match = emailRegex.exec(text)) !== null) {
      redactions.push({
        type: 'pii',
        original: match[0],
        replacement: '[EMAIL_REDACTED]',
        position: match.index
      });
    }

    // Phone numbers (various formats)
    const phoneRegex = /\b\d{3}[-.\s]?\d{3}[-.\s]?\d{4}\b/g;
    while ((match = phoneRegex.exec(text)) !== null) {
      redactions.push({
        type: 'pii',
        original: match[0],
        replacement: '[PHONE_REDACTED]',
        position: match.index
      });
    }

    // SSN patterns
    const ssnRegex = /\b\d{3}-\d{2}-\d{4}\b/g;
    while ((match = ssnRegex.exec(text)) !== null) {
      redactions.push({
        type: 'pii',
        original: match[0],
        replacement: '[SSN_REDACTED]',
        position: match.index
      });
    }

    return redactions;
  }

  private async checkInappropriateContent(text: string): Promise<{
    detected: boolean;
    reason: string;
  }> {
    // Use OpenAI moderation API
    const moderation = await this.openai.moderations.create({
      input: text
    });

    const flagged = moderation.results[0].flagged;
    const categories = moderation.results[0].categories;

    if (flagged) {
      const flaggedCategories = Object.entries(categories)
        .filter(([_, isFlagged]) => isFlagged)
        .map(([cat, _]) => cat);

      return {
        detected: true,
        reason: `Flagged categories: ${flaggedCategories.join(', ')}`
      };
    }

    return { detected: false, reason: '' };
  }
}
```

```typescript
// services/agent-framework/src/memory/SessionService.ts

export interface PersistedSession {
  session_id: string;
  student_id: string;
  coach_id?: string;

  // Conversation history (short-term memory)
  messages: ChatCompletionMessageParam[];

  // Persistent state with scopes
  state: {
    // User-specific state (persists across sessions)
    [key: `user:${string}`]: any;

    // Application-level state
    [key: `app:${string}`]: any;

    // Temporary state (cleared after session ends)
    [key: `temp:${string}`]: any;
  };

  // Metadata
  created_at: Date;
  last_active: Date;
  turn_count: number;
  ended_at?: Date;
}

export class SessionService {
  constructor(private pool: Pool) {}

  /**
   * Create new session with empty state
   */
  async createSession(
    studentId: string,
    coachId?: string
  ): Promise<PersistedSession> {
    const sessionId = uuidv4();

    await this.pool.query(`
      INSERT INTO agent_sessions (
        session_id, student_id, coach_id, messages, state,
        created_at, last_active, turn_count
      )
      VALUES ($1, $2, $3, $4::jsonb, $5::jsonb, NOW(), NOW(), 0)
    `, [
      sessionId,
      studentId,
      coachId || null,
      JSON.stringify([]),
      JSON.stringify({})
    ]);

    return {
      session_id: sessionId,
      student_id: studentId,
      coach_id: coachId,
      messages: [],
      state: {},
      created_at: new Date(),
      last_active: new Date(),
      turn_count: 0
    };
  }

  /**
   * Load session from database
   */
  async getSession(sessionId: string): Promise<PersistedSession | null> {
    const result = await this.pool.query(`
      SELECT * FROM agent_sessions WHERE session_id = $1
    `, [sessionId]);

    if (result.rows.length === 0) return null;

    const row = result.rows[0];
    return {
      session_id: row.session_id,
      student_id: row.student_id,
      coach_id: row.coach_id,
      messages: row.messages || [],
      state: row.state || {},
      created_at: row.created_at,
      last_active: row.last_active,
      turn_count: row.turn_count,
      ended_at: row.ended_at
    };
  }

  /**
   * Update session state (with scoped keys)
   */
  async updateState(
    sessionId: string,
    updates: Record<string, any>
  ): Promise<void> {
    // Validate scope prefixes
    for (const key of Object.keys(updates)) {
      if (!key.match(/^(user|app|temp):/)) {
        throw new Error(`Invalid state key "${key}" - must have scope prefix (user:, app:, temp:)`);
      }
    }

    // Merge updates into existing state
    await this.pool.query(`
      UPDATE agent_sessions
      SET
        state = state || $2::jsonb,
        last_active = NOW()
      WHERE session_id = $1
    `, [sessionId, JSON.stringify(updates)]);
  }

  /**
   * Add message to conversation history
   */
  async addMessage(
    sessionId: string,
    message: ChatCompletionMessageParam
  ): Promise<void> {
    await this.pool.query(`
      UPDATE agent_sessions
      SET
        messages = messages || $2::jsonb,
        turn_count = turn_count + 1,
        last_active = NOW()
      WHERE session_id = $1
    `, [sessionId, JSON.stringify([message])]);
  }

  /**
   * End session and clear temp: state
   */
  async endSession(sessionId: string): Promise<void> {
    await this.pool.query(`
      UPDATE agent_sessions
      SET
        ended_at = NOW(),
        state = (
          SELECT jsonb_object_agg(key, value)
          FROM jsonb_each(state)
          WHERE key NOT LIKE 'temp:%'
        )
      WHERE session_id = $1
    `, [sessionId]);
  }

  /**
   * Get or create session (idempotent)
   */
  async getOrCreateSession(
    studentId: string,
    sessionId?: string,
    coachId?: string
  ): Promise<PersistedSession> {
    if (sessionId) {
      const existing = await this.getSession(sessionId);
      if (existing) return existing;
    }

    // Create new session
    return await this.createSession(studentId, coachId);
  }
}
```

```typescript
// services/agent-framework/src/memory/MemoryService.ts

export interface MemoryEntry {
  memory_id: string;
  student_id: string;
  content: string;              // The memory content
  memory_type: 'semantic' | 'episodic' | 'procedural';
  metadata: Record<string, any>;
  embedding?: number[];         // Vector embedding for semantic search
  created_at: Date;
  expires_at?: Date;            // Optional expiration
}

export class MemoryService {
  constructor(
    private pool: Pool,
    private pinecone: PineconeClient  // Optional: for vector search
  ) {}

  /**
   * Add memory to long-term storage
   */
  async addMemory(memory: {
    student_id: string;
    content: string;
    memory_type: 'semantic' | 'episodic' | 'procedural';
    metadata: Record<string, any>;
    expires_at?: Date;
  }): Promise<string> {

    const memoryId = uuidv4();

    // Generate embedding for semantic search
    const embedding = await this.generateEmbedding(memory.content);

    // Store in PostgreSQL
    await this.pool.query(`
      INSERT INTO long_term_memory (
        memory_id, student_id, content, memory_type,
        metadata, embedding, created_at, expires_at
      )
      VALUES ($1, $2, $3, $4, $5::jsonb, $6, NOW(), $7)
    `, [
      memoryId,
      memory.student_id,
      memory.content,
      memory.memory_type,
      JSON.stringify(memory.metadata),
      JSON.stringify(embedding),  // Store as JSONB for now
      memory.expires_at || null
    ]);

    // Optionally store embedding in Pinecone for scalable search
    if (this.pinecone && memory.memory_type === 'semantic') {
      await this.pinecone.upsert([{
        id: memoryId,
        values: embedding,
        metadata: {
          student_id: memory.student_id,
          content: memory.content,
          memory_type: memory.memory_type,
          ...memory.metadata
        }
      }]);
    }

    return memoryId;
  }

  /**
   * Search memories semantically
   */
  async searchMemories(query: {
    student_id: string;
    query_text: string;
    memory_type?: 'semantic' | 'episodic' | 'procedural';
    top_k?: number;
    min_similarity?: number;
  }): Promise<MemoryEntry[]> {

    // Generate query embedding
    const queryEmbedding = await this.generateEmbedding(query.query_text);

    // Search using cosine similarity in PostgreSQL
    // (For production, use Pinecone or pgvector extension)
    const result = await this.pool.query(`
      SELECT
        memory_id,
        student_id,
        content,
        memory_type,
        metadata,
        created_at,
        expires_at,
        1 - (embedding::jsonb::text::vector <=> $1::vector) as similarity
      FROM long_term_memory
      WHERE student_id = $2
        ${query.memory_type ? 'AND memory_type = $3' : ''}
        AND (expires_at IS NULL OR expires_at > NOW())
      ORDER BY embedding::jsonb::text::vector <=> $1::vector
      LIMIT ${query.top_k || 5}
    `, query.memory_type
        ? [JSON.stringify(queryEmbedding), query.student_id, query.memory_type]
        : [JSON.stringify(queryEmbedding), query.student_id]
    );

    return result.rows
      .filter(row => !query.min_similarity || row.similarity >= query.min_similarity)
      .map(row => ({
        memory_id: row.memory_id,
        student_id: row.student_id,
        content: row.content,
        memory_type: row.memory_type,
        metadata: row.metadata,
        created_at: row.created_at,
        expires_at: row.expires_at
      }));
  }

  /**
   * Get episodic memories (time-based)
   */
  async getEpisodicMemories(
    studentId: string,
    startDate: Date,
    endDate: Date
  ): Promise<MemoryEntry[]> {
    const result = await this.pool.query(`
      SELECT * FROM long_term_memory
      WHERE student_id = $1
        AND memory_type = 'episodic'
        AND created_at BETWEEN $2 AND $3
        AND (expires_at IS NULL OR expires_at > NOW())
      ORDER BY created_at DESC
    `, [studentId, startDate, endDate]);

    return result.rows;
  }

  private async generateEmbedding(text: string): Promise<number[]> {
    const response = await this.openai.embeddings.create({
      model: 'text-embedding-3-large',
      input: text
    });
    return response.data[0].embedding;
  }
}
```

### Feature 5: Human-in-the-Loop (HITL) Escalation Engine

**Problem:** No automated escalation when AI is uncertain or encounters edge cases.

**Solution:** Confidence-based escalation triggers with coach notification system.

#### 5.1 HITL Escalation Logic

```typescript
// services/agent-framework/src/hitl/EscalationEngine.ts

export interface EscalationTrigger {
  type: 'low_confidence' | 'complex_decision' | 'safety_concern' | 'user_request';
  confidence_score?: number;
  reasoning: string;
  urgency: 'immediate' | 'within_1_hour' | 'within_24_hours' | 'next_session';
}

export interface EscalationRecord {
  escalation_id: string;
  student_id: string;
  session_id: string;
  query: string;
  trigger: EscalationTrigger;
  agent_draft_response?: string;
  coach_id?: string;
  status: 'pending' | 'in_progress' | 'resolved';
  resolution?: {
    coach_response: string;
    coach_id: string;
    resolved_at: Date;
    feedback_to_ai?: string;
  };
  created_at: Date;
}

export class EscalationEngine {
  private readonly CONFIDENCE_THRESHOLD = 0.80;

  /**
   * Determine if query should escalate to human coach
   */
  async shouldEscalate(context: {
    query: string;
    agent_response: string;
    confidence_score: number;
    reasoning_trace: any;
    student_id: string;
  }): Promise<EscalationTrigger | null> {

    // Trigger 1: Low confidence
    if (context.confidence_score < this.CONFIDENCE_THRESHOLD) {
      return {
        type: 'low_confidence',
        confidence_score: context.confidence_score,
        reasoning: `Confidence ${(context.confidence_score * 100).toFixed(1)}% below ${(this.CONFIDENCE_THRESHOLD * 100)}% threshold`,
        urgency: 'within_24_hours'
      };
    }

    // Trigger 2: Complex strategic decision (detected via LLM)
    const isComplexDecision = await this.detectComplexDecision(context.query);
    if (isComplexDecision.is_complex) {
      return {
        type: 'complex_decision',
        reasoning: isComplexDecision.explanation,
        urgency: 'within_1_hour'
      };
    }

    // Trigger 3: User explicitly requests human
    if (this.detectUserRequestForHuman(context.query)) {
      return {
        type: 'user_request',
        reasoning: 'Student explicitly requested to speak with coach',
        urgency: 'next_session'
      };
    }

    return null;  // No escalation needed
  }

  /**
   * Create escalation and notify coach
   */
  async escalate(context: {
    student_id: string;
    session_id: string;
    query: string;
    agent_draft_response?: string;
    trigger: EscalationTrigger;
  }): Promise<EscalationRecord> {

    const escalationId = uuidv4();

    // Get assigned coach for student
    const coach = await this.getAssignedCoach(context.student_id);

    // Insert escalation record
    await pool.query(`
      INSERT INTO hitl_escalations (
        escalation_id, student_id, session_id, query,
        trigger, agent_draft_response, coach_id,
        status, created_at
      )
      VALUES ($1, $2, $3, $4, $5::jsonb, $6, $7, 'pending', NOW())
    `, [
      escalationId,
      context.student_id,
      context.session_id,
      context.query,
      JSON.stringify(context.trigger),
      context.agent_draft_response || null,
      coach?.coach_id || null
    ]);

    // Send notification to coach
    if (coach) {
      await this.notifyCoach(coach, {
        escalation_id: escalationId,
        student_id: context.student_id,
        query: context.query,
        urgency: context.trigger.urgency
      });
    }

    return {
      escalation_id: escalationId,
      student_id: context.student_id,
      session_id: context.session_id,
      query: context.query,
      trigger: context.trigger,
      agent_draft_response: context.agent_draft_response,
      coach_id: coach?.coach_id,
      status: 'pending',
      created_at: new Date()
    };
  }

  /**
   * Coach resolves escalation
   */
  async resolveEscalation(
    escalationId: string,
    coachId: string,
    resolution: {
      coach_response: string;
      feedback_to_ai?: string;
    }
  ): Promise<void> {
    await pool.query(`
      UPDATE hitl_escalations
      SET
        status = 'resolved',
        resolution = $2::jsonb,
        resolved_at = NOW()
      WHERE escalation_id = $1
    `, [
      escalationId,
      JSON.stringify({
        coach_response: resolution.coach_response,
        coach_id: coachId,
        resolved_at: new Date(),
        feedback_to_ai: resolution.feedback_to_ai
      })
    ]);

    // If coach provided feedback, store for learning
    if (resolution.feedback_to_ai) {
      await this.storeCoachFeedback(escalationId, resolution.feedback_to_ai);
    }
  }

  private async detectComplexDecision(query: string): Promise<{
    is_complex: boolean;
    explanation: string;
  }> {
    const prompt = `
Is this a complex strategic decision requiring human judgment?

Query: "${query}"

Complex decisions include:
- Early decision vs regular decision trade-offs
- Major life choices (switching schools, gap year)
- Financial aid strategy
- Ethical dilemmas
- Family conflict resolution

Return JSON: {is_complex: true/false, explanation: "why"}
`;

    const response = await this.openai.chat.completions.create({
      model: 'gpt-4o',
      messages: [{ role: 'user', content: prompt }],
      response_format: { type: 'json_object' }
    });

    return JSON.parse(response.choices[0].message.content);
  }

  private detectUserRequestForHuman(query: string): boolean {
    const patterns = [
      /\b(talk to|speak with|contact) (coach|human|counselor)\b/i,
      /\breal person\b/i,
      /\bhuman help\b/i
    ];

    return patterns.some(p => p.test(query));
  }

  private async getAssignedCoach(studentId: string): Promise<{ coach_id: string } | null> {
    const result = await pool.query(`
      SELECT coach_id FROM student_coach_assignments
      WHERE student_id = $1 AND is_active = true
      LIMIT 1
    `, [studentId]);

    return result.rows[0] || null;
  }

  private async notifyCoach(
    coach: { coach_id: string },
    escalation: {
      escalation_id: string;
      student_id: string;
      query: string;
      urgency: string;
    }
  ): Promise<void> {
    // Send email, SMS, Slack notification
    // Implementation depends on notification infrastructure
    log.info('Coach notified of escalation', {
      coach_id: coach.coach_id,
      escalation_id: escalation.escalation_id,
      urgency: escalation.urgency
    });
  }

  private async storeCoachFeedback(
    escalationId: string,
    feedback: string
  ): Promise<void> {
    // Store in coaching_feedback table for learning loop
    await pool.query(`
      INSERT INTO coaching_feedback (
        feedback_id, escalation_id, feedback_text, created_at
      )
      VALUES ($1, $2, $3, NOW())
    `, [uuidv4(), escalationId, feedback]);
  }
}
```
