## v15.2 Feature Set

### Feature 1: LangChain LCEL Integration for Strategy Orchestration

**Problem:** v15.1 Universal Strategy Engine has DAG-based strategy nodes but no explicit chain orchestration framework.

**Solution:** Wrap strategy execution in LangChain LCEL (LangChain Expression Language) for:
- Explicit chain definitions with pipe operators
- Built-in retry and error handling
- Streaming support for real-time progress
- Integration with 1000+ LangChain tools

#### Architecture Integration

```typescript
// v15.1 EXISTING: Strategy node execution (PRESERVED)
interface StrategyNode {
  node_id: string;
  node_type: "ec_discovery" | "essay_planning" | "schedule_optimization";
  prerequisites: string[];
  outcomes: { ivyscore_lift_avg: number; success_rate: number };
}

// v15.2 ADDITION: LangChain wrapper for strategy execution
import { RunnableSequence, RunnablePassthrough } from "langchain/runnables";

class LangChainStrategyOrchestrator {
  constructor(
    private strategyEngine: UniversalStrategyEngine,
    private voiceAdapter: VoiceAdapter
  ) {}

  async executeStrategyChain(
    strategyNode: StrategyNode,
    studentContext: StudentContext,
    coachId: string
  ): Promise<StrategyResult> {
    // Build LCEL chain: Universal Strategy → Coach Voice Adapter → Student Filter
    const chain = RunnableSequence.from([
      // Step 1: Execute universal strategy (v15.1 existing)
      new RunnablePassthrough().pipe(async (input) => {
        return await this.strategyEngine.execute(strategyNode, input.studentContext);
      }),

      // Step 2: Adapt to coach voice (v15.1 existing)
      new RunnablePassthrough().pipe(async (strategyOutput) => {
        return await this.voiceAdapter.adaptToCoachVoice(
          strategyOutput.response,
          input.coachId,
          input.studentContext.state
        );
      }),

      // Step 3: Filter for student archetype (v15.1 existing)
      new RunnablePassthrough().pipe(async (coachAdaptedResponse) => {
        return await this.filterForArchetype(
          coachAdaptedResponse,
          input.studentContext.archetype
        );
      }),
    ]);

    // Execute with built-in error handling and retries
    const result = await chain.invoke({
      studentContext,
      coachId,
    });

    return result;
  }
}
```

### Feature 2: LLM-Based Routing for Coach-Agnostic Intent Classification

**Problem:** v15.1 has coach-agnostic strategy nodes but routing is still rule-based. Gap analysis shows "20% routing maturity" needs LLM-based classification.

**Solution:** Add LLM-powered intent classifier that works across all coach personas.

#### Architecture

```typescript
// v15.2 ADDITION: LLM-based intent router
import { ChatPromptTemplate } from "langchain/prompts";
import { ChatOpenAI } from "langchain/chat_models/openai";

class UniversalIntentRouter {
  private llm: ChatOpenAI;

  constructor() {
    this.llm = new ChatOpenAI({ modelName: "gpt-4", temperature: 0 });
  }

  async classifyIntent(studentQuery: string, context: StudentContext): Promise<Intent> {
    const routerPrompt = ChatPromptTemplate.fromMessages([
      ["system", `
        You are an intent classifier for college admissions coaching.
        Classify the student query into one of these intents:

        - "gameplan_strategy": Questions about overall college strategy, timeline, planning
        - "ec_discovery": Questions about extracurriculars, activities, awards
        - "essay_help": Essay brainstorming, editing, topics
        - "scholarship_search": Financial aid, scholarships, merit opportunities
        - "schedule_optimization": Course selection, GPA strategy, academic planning
        - "mental_health": Stress, burnout, anxiety, wellbeing
        - "clarification_needed": Query is ambiguous, need more info

        Context: Student is {archetype}, grade {grade}, burnout {burnout_level}.

        Respond with ONLY the intent slug (e.g., "gameplan_strategy").
      `],
      ["user", "{query}"]
    ]);

    const chain = routerPrompt.pipe(this.llm);

    const intent = await chain.invoke({
      query: studentQuery,
      archetype: context.archetype,
      grade: context.grade,
      burnout_level: context.state.burnout_level
    });

    return this.parseIntent(intent.content);
  }

  async routeToStrategy(intent: Intent): Promise<StrategyNode> {
    // Map intent to v15.1 strategy graph
    const intentToStrategyMap: Record<string, string> = {
      "gameplan_strategy": "comprehensive_planning",
      "ec_discovery": "opportunity_discovery",
      "essay_help": "narrative_development",
      "scholarship_search": "financial_optimization",
      "schedule_optimization": "academic_rigor_balancing",
      "mental_health": "wellbeing_intervention",
    };

    const nodeType = intentToStrategyMap[intent.slug];
    return await this.strategyEngine.getStrategyNode(nodeType);
  }
}
```

### Feature 3: Reflection v2 - Producer-Critic Loops

**Problem:** v15.1 has self-critique (single pass) but gap analysis shows "0% reflection maturity" needs multi-iteration producer-critic loops.

**Solution:** Add iterative refinement with quality gates before responses reach students.

#### Architecture

```typescript
// v15.2 ADDITION: Producer-Critic pattern
class ProducerCriticReflectionLoop {
  constructor(
    private producerAgent: UniversalStrategyEngine,
    private criticAgent: CriticAgent,
    private voiceAdapter: VoiceAdapter
  ) {}

  async generateWithReflection(
    strategyNode: StrategyNode,
    studentContext: StudentContext,
    coachId: string,
    maxIterations: number = 3,
    qualityThreshold: number = 0.8
  ): Promise<ReflectedResponse> {
    let currentResponse: string;
    let iteration = 0;
    let critiques: Critique[] = [];

    // Initial production
    currentResponse = await this.producerAgent.execute(strategyNode, studentContext);

    while (iteration < maxIterations) {
      // Critic evaluates against quality goals
      const critique = await this.criticAgent.evaluate(currentResponse, {
        goals: [
          { criteria: "actionable", weight: 0.3 },
          { criteria: "empathetic", weight: 0.2 },
          { criteria: "data_grounded", weight: 0.3 },  // v15.1 SQL grounding
          { criteria: "ivyscore_optimal", weight: 0.2 }
        ],
        studentContext,
      });

      critiques.push(critique);

      // Check if quality threshold met
      if (critique.overall_score >= qualityThreshold) {
        // PASSED: Adapt to coach voice and return
        const adaptedResponse = await this.voiceAdapter.adaptToCoachVoice(
          currentResponse,
          coachId,
          studentContext.state
        );

        return {
          response: adaptedResponse,
          quality_score: critique.overall_score,
          iterations_needed: iteration + 1,
          critiques,
        };
      }

      // FAILED: Refine based on critique
      currentResponse = await this.producerAgent.refine(
        currentResponse,
        critique.suggestions,
        strategyNode,
        studentContext
      );

      iteration++;
    }

    // Max iterations reached - return best effort with warning
    const adaptedResponse = await this.voiceAdapter.adaptToCoachVoice(
      currentResponse,
      coachId,
      studentContext.state
    );

    return {
      response: adaptedResponse,
      quality_score: critiques[critiques.length - 1].overall_score,
      iterations_needed: iteration,
      critiques,
      warning: "Quality threshold not met - escalate to human coach",
    };
  }
}
```

**Critic Agent Implementation:**

```typescript
class CriticAgent {
  async evaluate(
    response: string,
    evaluationCriteria: { goals: Goal[]; studentContext: StudentContext }
  ): Promise<Critique> {
    const critiquePrompt = `
      Evaluate this coaching response against these criteria:

      Response: "${response}"

      Criteria:
      ${evaluationCriteria.goals.map(g => `- ${g.criteria} (weight: ${g.weight})`).join('\n')}

      Student Context:
      - Archetype: ${evaluationCriteria.studentContext.archetype}
      - Burnout Level: ${evaluationCriteria.studentContext.state.burnout_level}
      - Current IvyScore: ${evaluationCriteria.studentContext.ivyscore}

      For each criterion, provide:
      1. Score (0-1)
      2. Specific suggestions for improvement

      Format as JSON:
      {
        "criteria_scores": {"actionable": 0.8, ...},
        "suggestions": ["Add specific deadlines", ...],
        "overall_score": 0.75
      }
    `;

    const critiqueResponse = await this.llm.complete(critiquePrompt);
    return JSON.parse(critiqueResponse);
  }
}
```

```typescript
// v15.2 ADDITION: Context engineering for optimal prompts
class ContextEngineeringPipeline {
  async constructOptimalContext(
    strategyNode: StrategyNode,
    studentContext: StudentContext,
    coachPersona: CoachPersona
  ): Promise<EngineeeredContext> {
    // Step 1: Retrieve relevant student history (v15.1 existing)
    const relevantHistory = await this.semanticMemory.retrieve(
      studentContext.student_id,
      strategyNode.node_type,
      top_k: 5
    );

    // Step 2: Extract key facts (SQL grounding from v15.1)
    const groundingFacts = await this.extractGroundingFacts(studentContext);

    // Step 3: Construct coach-specific examples (NEW)
    const fewShotExamples = await this.selectFewShotExamples(
      coachPersona,
      strategyNode.node_type,
      count: 3
    );

    // Step 4: Optimize prompt structure for coach's style
    const optimizedPrompt = this.assemblePrompt({
      systemPrompt: this.buildSystemPrompt(coachPersona),
      fewShotExamples,
      groundingFacts,
      relevantHistory,
      currentQuery: studentContext.current_query,
      styleGuide: coachPersona.linguistic_style,
    });

    return {
      prompt: optimizedPrompt,
      context_length: this.countTokens(optimizedPrompt),
      compression_applied: optimizedPrompt.length > MAX_CONTEXT,
      metadata: {
        coach_id: coachPersona.coach_id,
        strategy_node: strategyNode.node_id,
        facts_count: groundingFacts.length,
      },
    };
  }

  private buildSystemPrompt(persona: CoachPersona): string {
    // Combine universal coaching principles with coach-specific style
    return `
      You are a college admissions coach helping high school students.

      UNIVERSAL PRINCIPLES (all coaches):
      - Always ground advice in student's actual data (GPA, ECs, awards)
      - Prioritize mental health and sustainable pacing
      - Optimize for IvyScore growth (college admission probability)

      YOUR COACHING STYLE (${persona.coach_name}):
      - Warmth: ${persona.eq_profile.warmth * 100}% (${this.warmthDescription(persona.eq_profile.warmth)})
      - Directness: ${persona.eq_profile.directness * 100}% (${this.directnessDescription(persona.eq_profile.directness)})
      - Humor: ${persona.eq_profile.humor * 100}% (${this.humorDescription(persona.eq_profile.humor)})

      LINGUISTIC PREFERENCES:
      - Formality: ${persona.linguistic_style.formality}
      - Emoji use: ${persona.linguistic_style.emoji_use ? 'frequent' : 'minimal'}
      - Enthusiasm: ${persona.linguistic_style.enthusiasm}

      Adapt your response to match this style while maintaining universal principles.
    `;
  }

  private async selectFewShotExamples(
    persona: CoachPersona,
    strategyType: string,
    count: number
  ): Promise<FewShotExample[]> {
    // Retrieve best examples from this coach's action corpus (v15.1 table)
    const examples = await db.query(`
      SELECT
        context,
        action,
        outcomes
      FROM coaching_action_corpus
      WHERE coach_id = $1
        AND intervention_type = $2
        AND outcomes->>'ivyscore_delta' > '5'  -- High-impact examples
      ORDER BY outcomes->>'ivyscore_delta' DESC
      LIMIT $3
    `, [persona.coach_id, strategyType, count]);

    return examples.rows.map(row => ({
      query: row.context.student_query,
      response: row.action.response,
      outcome: `IvyScore +${row.outcomes.ivyscore_delta}`,
    }));
  }
}
```

### Feature 5: Framework Abstraction Layer (Prevent Lock-In)

**Problem:** Adding LangChain creates vendor lock-in risk. Gap analysis warns: "Framework Lock-in: LOW risk but mitigate with abstraction layers."

**Solution:** Abstract agentic primitives so we can swap frameworks later.

#### Architecture

```typescript
// v15.2 ADDITION: Framework abstraction
interface IAgenticFramework {
  // Chain orchestration
  createChain(steps: ChainStep[]): RunnableChain;
  executeChain(chain: RunnableChain, input: any): Promise<any>;

  // Routing
  routeIntent(query: string, routes: Route[]): Promise<string>;

  // Memory management
  storeMemory(key: string, value: any, scope: MemoryScope): Promise<void>;
  retrieveMemory(key: string, scope: MemoryScope): Promise<any>;

  // Tool calling
  registerTool(tool: Tool): void;
  executeTool(toolName: string, params: any): Promise<any>;
}

// LangChain implementation
class LangChainFramework implements IAgenticFramework {
  async createChain(steps: ChainStep[]): RunnableChain {
    return RunnableSequence.from(
      steps.map(step => new RunnablePassthrough().pipe(step.fn))
    );
  }

  async executeChain(chain: RunnableChain, input: any): Promise<any> {
    return await chain.invoke(input);
  }

  // ... other methods
}

// LlamaIndex implementation (future)
class LlamaIndexFramework implements IAgenticFramework {
  // ... implement with LlamaIndex primitives
}

// Google ADK implementation (future)
class GoogleADKFramework implements IAgenticFramework {
  // ... implement with ADK primitives
}

// Dependency injection
class FrameworkProvider {
  private static instance: IAgenticFramework;

  static initialize(frameworkType: "langchain" | "llamaindex" | "adk") {
    switch (frameworkType) {
      case "langchain":
        this.instance = new LangChainFramework();
        break;
      case "llamaindex":
        this.instance = new LlamaIndexFramework();
        break;
      case "adk":
        this.instance = new GoogleADKFramework();
        break;
    }
  }

  static getFramework(): IAgenticFramework {
    return this.instance;
  }
}
```

**Configuration:**

```typescript
// Environment-based framework selection
const FRAMEWORK = process.env.AGENTIC_FRAMEWORK || "langchain";

FrameworkProvider.initialize(FRAMEWORK);

// Usage in application code
const framework = FrameworkProvider.getFramework();
const chain = await framework.createChain([...]);
const result = await framework.executeChain(chain, input);
```

**Why This Matters:**
- Can migrate from LangChain → LlamaIndex → Google ADK without rewriting business logic
- v15.1 Universal Strategy Engine never depends on specific framework
- De-risks technology choices

## Appendix A: Database Migrations

### Migration Script: v15.2 Tables

```sql
-- Migration: v15.2_framework_integration.sql
-- Run after v15.1 migrations are complete

BEGIN;

-- 1. Strategy chain traces (LangChain LCEL)
CREATE TABLE strategy_chain_traces (
  trace_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_node_id UUID NOT NULL REFERENCES strategy_nodes(node_id),
  student_id TEXT NOT NULL REFERENCES students(student_id),
  coach_id TEXT NOT NULL,
  chain_steps JSONB NOT NULL,
  total_duration_ms INTEGER NOT NULL,
  success BOOLEAN NOT NULL,
  error_details TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_chain_traces_student ON strategy_chain_traces(student_id, created_at DESC);
CREATE INDEX idx_chain_traces_performance ON strategy_chain_traces(total_duration_ms, success);

-- 2. Intent classifications (LLM routing)
CREATE TABLE intent_classifications (
  classification_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id TEXT NOT NULL REFERENCES students(student_id),
  query TEXT NOT NULL,
  classified_intent TEXT NOT NULL,
  confidence NUMERIC(3,2),
  context JSONB,
  strategy_node_id UUID REFERENCES strategy_nodes(node_id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_intent_by_student ON intent_classifications(student_id, created_at DESC);
CREATE INDEX idx_intent_analytics ON intent_classifications(classified_intent, created_at DESC);

-- 3. Reflection iterations (Producer-Critic)
CREATE TABLE reflection_iterations (
  reflection_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_execution_id UUID NOT NULL,
  student_id TEXT NOT NULL REFERENCES students(student_id),
  coach_id TEXT NOT NULL,
  iteration_number INTEGER NOT NULL CHECK (iteration_number >= 1),
  producer_output TEXT NOT NULL,
  critic_feedback JSONB NOT NULL,
  quality_threshold NUMERIC(3,2) NOT NULL CHECK (quality_threshold BETWEEN 0 AND 1),
  met_threshold BOOLEAN NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_reflection_quality ON reflection_iterations(met_threshold, quality_threshold);
CREATE INDEX idx_reflection_student ON reflection_iterations(student_id, created_at DESC);

-- 4. Context engineering logs
CREATE TABLE context_engineering_logs (
  log_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  strategy_execution_id UUID NOT NULL,
  coach_id TEXT NOT NULL,
  student_id TEXT NOT NULL REFERENCES students(student_id),
  context_length_tokens INTEGER NOT NULL CHECK (context_length_tokens >= 0),
  compression_applied BOOLEAN NOT NULL,
  few_shot_examples_count INTEGER NOT NULL CHECK (few_shot_examples_count >= 0),
  grounding_facts_count INTEGER NOT NULL CHECK (grounding_facts_count >= 0),
  context_optimization_time_ms INTEGER NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_context_performance ON context_engineering_logs(
  coach_id,
  compression_applied,
  context_length_tokens
);

COMMIT;
```
