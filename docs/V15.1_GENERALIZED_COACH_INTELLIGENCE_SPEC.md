## Executive Summary

### The Paradigm Shift: From Cloning to Generalization

**v15.0 Challenge:** Multi-agent collaboration built on Jenny's coaching patterns - works brilliantly for students like Huda, but doesn't scale to diverse coaches and student archetypes.

**v15.1 Solution:** Extract the **transferable scaffolding of elite coaching** from Jenny's data, then support infinite coaches/students through:
1. **Universal Strategy Graph** - What "moves" produce IvyScore lift (coach-agnostic)
2. **Coach Persona Adapters** - How each coach's voice/style manifests (per-coach)
3. **Student State Models** - How each student's archetype responds (per-student)

### The Formula

```
Coach Twin ≠ Jenny Clone

Coach Twin = Strategy Engine (universal)
           + Coach Voice Adapter (per-coach)
           + Student State Model (per-student)
```

### Key Principle: NO BESPOKE

**Golden Standard → Generalization:**
- Use Jenny + Huda data to establish **first golden standard**
- Abstract Jenny's coaching cognition into **generalizable intelligence layers**
- Support **multiple coaches** with distinct EQ/personas/tactics
- Support **multiple students** with different archetypes/pacing/goals

**Result:** By Coach #4, universal models **outperform any individual coach's heuristics**.

```
  → IvyScore Lift: +12 points (average across all students who matched criteria)
```

**Training Process:**

```typescript
// services/agent-framework/src/intelligence/StrategyGraphBuilder.ts

export class StrategyGraphBuilder {
  /**
   * Extract strategy patterns from coaching session data
   * Sources: Jenny transcripts, Coach #2 sessions, Coach #3 sessions, ...
   */
  async buildFromCoachingData(coaches: string[]): Promise<void> {
    for (const coachId of coaches) {
      // Get all coaching actions by this coach
      const actions = await this.extractCoachingActions(coachId);

      for (const action of actions) {
        // Find or create strategy node
        const node = await this.findOrCreateNode(action.type, action.description);

        // Update outcomes with this coach's data
        await this.updateNodeOutcomes(node.node_id, {
          coach_id: coachId,
          student_id: action.student_id,
          ivyscore_before: action.ivyscore_before,
          ivyscore_after: action.ivyscore_after,
          hgti_delta: action.hgti_delta,
          success: action.success
        });

        // Extract prerequisites from context
        if (action.prerequisites) {
          await this.addPrerequisites(node.node_id, action.prerequisites);
        }
      }
    }

    // Compute aggregate statistics across all coaches
    await this.computeAggregateStatistics();
  }

  private async extractCoachingActions(coachId: string): Promise<CoachingAction[]> {
    // Parse coaching session transcripts
    // OR query structured action logs from conversation_turns
    const result = await pool.query(`
      SELECT
        student_id,
        message,
        metadata,
        created_at
      FROM conversation_turns
      WHERE coach_id = $1
        AND role = 'assistant'
        AND metadata->>'action_type' IS NOT NULL
      ORDER BY created_at
    `, [coachId]);

    return result.rows.map(row => this.parseAction(row));
  }

  private async computeAggregateStatistics(): Promise<void> {
    // For each strategy node, compute:
    // - Average IvyScore lift across all coaches
    // - Standard deviation (to measure consistency)
    // - Coach agreement rate (% who recommend this strategy)
    // - Success rate (% students who succeeded after this action)

    await pool.query(`
      UPDATE strategy_nodes
      SET
        outcomes = (
          SELECT jsonb_build_object(
            'ivyscore_lift_avg', AVG((outcomes->>'ivyscore_lift')::numeric),
            'ivyscore_lift_std', STDDEV((outcomes->>'ivyscore_lift')::numeric),
            'hgti_delta_avg', AVG((outcomes->>'hgti_delta')::numeric),
            'success_rate', AVG(CASE WHEN (outcomes->>'success')::boolean THEN 1.0 ELSE 0.0 END)
          )
          FROM strategy_node_outcomes
          WHERE node_id = strategy_nodes.node_id
        ),
        coach_agreement_rate = (
          SELECT COUNT(DISTINCT coach_id)::numeric / (SELECT COUNT(DISTINCT coach_id) FROM coaches)
          FROM strategy_node_outcomes
          WHERE node_id = strategy_nodes.node_id
        ),
        updated_at = NOW()
    `);
  }
}
```

#### 1.2 Outcome Elasticity Model

**What:** Which actions create **biggest future leverage** (not just immediate lift).

**Example:**
```
Action: "Start passion project in CS"
  Immediate IvyScore Lift: +3 points
  6-Month Elasticity: +18 points (projects compound via publications, scaling, awards)
  12-Month Elasticity: +35 points
  → High leverage action (prioritize even if slow payoff)

Action: "Retake SAT from 1480 → 1520"
  Immediate IvyScore Lift: +5 points
  6-Month Elasticity: +5 points (no compounding)
  12-Month Elasticity: +5 points
  → Low leverage action (diminishing returns)
```

```typescript
// services/agent-framework/src/intelligence/StudentStateTracker.ts

export class StudentStateTracker {
  /**
   * Update student state based on recent activity
   * Sources: HGTI deltas, adherence rates, sentiment analysis
   */
  async updateState(studentId: string): Promise<StudentState> {
    // Get recent metrics
    const recentMetrics = await this.getRecentMetrics(studentId, 14); // Last 2 weeks

    // Detect burnout signals
    const burnoutLevel = this.detectBurnout({
      task_completion_rate: recentMetrics.adherence,
      response_time_avg: recentMetrics.avg_response_time,
      sentiment_trend: recentMetrics.sentiment_slope
    });

    // Detect momentum
    const momentumLevel = this.detectMomentum({
      hgti_delta: recentMetrics.hgti_change,
      wins_count: recentMetrics.verified_wins,
      engagement_score: recentMetrics.engagement
    });

    // Detect confidence
    const confidenceLevel = this.detectConfidence({
      self_efficacy_mentions: recentMetrics.efficacy_count,
      positive_sentiment: recentMetrics.positive_ratio
    });

    // Update database
    await pool.query(`
      UPDATE student_state_tracking
      SET
        current_state = jsonb_build_object(
          'burnout_level', $2,
          'momentum', $3,
          'confidence', $4,
          'timestamp', NOW()
        ),
        state_history = state_history || jsonb_build_object(
          'timestamp', NOW(),
          'state', jsonb_build_object(
            'burnout_level', $2,
            'momentum', $3,
            'confidence', $4
          )
        )::jsonb,
        last_updated = NOW()
      WHERE student_id = $1
    `, [studentId, burnoutLevel, momentumLevel, confidenceLevel]);

    return {
      student_id: studentId,
      burnout_level: burnoutLevel,
      momentum: momentumLevel,
      confidence: confidenceLevel
    };
  }

  private detectBurnout(metrics: {
    task_completion_rate: number;
    response_time_avg: number;
    sentiment_trend: number;
  }): number {
    // Burnout signals:
    // - Dropping adherence
    // - Slower response times
    // - Negative sentiment trend

    let burnout = 0;

    if (metrics.task_completion_rate < 0.6) burnout += 0.3;
    if (metrics.response_time_avg > 48) burnout += 0.2;  // >48 hours avg
    if (metrics.sentiment_trend < -0.1) burnout += 0.3;

    return Math.min(burnout, 1.0);
  }

  private detectMomentum(metrics: {
    hgti_delta: number;
    wins_count: number;
    engagement_score: number;
  }): number {
    // Momentum signals:
    // - Positive HGTI change
    // - Recent wins
    // - High engagement

    let momentum = 0;

    if (metrics.hgti_delta > 0) momentum += 0.4;
    momentum += Math.min(metrics.wins_count * 0.2, 0.4);
    momentum += metrics.engagement_score * 0.2;

    return Math.min(momentum, 1.0);
  }

  private detectConfidence(metrics: {
    self_efficacy_mentions: number;
    positive_sentiment: number;
  }): number {
    // Confidence signals:
    // - Self-efficacy language ("I can", "I will")
    // - Positive sentiment

    return Math.min(
      metrics.self_efficacy_mentions * 0.1 + metrics.positive_sentiment * 0.5,
      1.0
    );
  }
}
```

```typescript
- Momentum: ${studentState.momentum * 100}%
```

```typescript
export class StudentAdapter {
  /**
   * Adapt coach-voiced response to student's archetype
   */
  async adaptToStudentArchetype(
    coachVoicedResponse: string,
    studentId: string
  ): Promise<string> {

    // Load student archetype
    const archetype = await this.loadStudentArchetype(studentId);
    const state = await this.loadStudentState(studentId);

    // Check for intervention triggers
    if (state.burnout_level > 0.7) {
      // HIGH BURNOUT: Add recovery guidance
      return await this.addBurnoutIntervention(coachVoicedResponse, archetype);
    }

    if (state.momentum > 0.8 && state.confidence > 0.7) {
      // HIGH MOMENTUM: Accelerate with stretch goals
      return await this.addAcceleration(coachVoicedResponse, archetype);
    }

    if (state.confidence < 0.4) {
      // LOW CONFIDENCE: Add more encouragement
      return await this.addEncouragement(coachVoicedResponse, archetype);
    }

    // Normal state: minimal adaptation
    return coachVoicedResponse;
  }
```

### Enhancement 2: Adaptive Prioritization with Student Archetype

**v15.0 Problem:** Task prioritization is one-size-fits-all.

**v15.1 Solution:** Adjust prioritization by student archetype.

```typescript
// Enhanced prioritization with archetype awareness

async prioritizeTasks(
  tasks: Task[],
  studentId: string
): Promise<Task[]> {

  // Load student archetype
  const archetype = await this.loadStudentArchetype(studentId);
  const state = await this.loadStudentState(studentId);

  // Score tasks by elasticity AND archetype fit
  const scored = await Promise.all(tasks.map(async (task) => {
    const baseScore = await this.getElasticityScore(task);

    // Archetype adjustments
    let archetypeMultiplier = 1.0;

    if (archetype.archetype_name === 'Overachiever') {
      // Overachievers: Prioritize high-difficulty tasks (they thrive on challenge)
      if (task.difficulty === 'high') archetypeMultiplier = 1.3;
    } else if (archetype.archetype_name === 'Late Bloomer') {
      // Late bloomers: Prioritize quick wins (build momentum)
      if (task.time_to_completion_weeks <= 2) archetypeMultiplier = 1.4;
    } else if (archetype.archetype_name === 'Risk-Averse') {
      // Risk-averse: Prioritize low-risk tasks
      if (task.risk_level === 'low') archetypeMultiplier = 1.2;
    }

    // State adjustments
    if (state.burnout_level > 0.7) {
      // High burnout: Deprioritize time-intensive tasks
      if (task.estimated_hours > 10) archetypeMultiplier *= 0.6;
    }

    if (state.momentum > 0.8) {
      // High momentum: Boost stretch goals
      if (task.is_stretch_goal) archetypeMultiplier *= 1.3;
    }

    return {
      task,
      score: baseScore * archetypeMultiplier
    };
  }));

  // Sort by adjusted score
  scored.sort((a, b) => b.score - a.score);

  return scored.map(s => s.task);
}
```

```typescript
// services/agent-framework/src/training/RLHFTrainer.ts

export class RLHFTrainer {
  /**
   * Train universal strategy model from coaching corpus
   * Runs weekly to incorporate new data
   */
  async trainWeeklyUpdate(): Promise<void> {
    console.log('Starting weekly RLHF training...\n');

    // STEP 1: Extract high-quality examples (positive outcomes)
    const highQualityExamples = await this.extractHighQualityExamples();
    console.log(`Extracted ${highQualityExamples.length} high-quality examples`);

    // STEP 2: Extract low-quality examples (negative outcomes)
    const lowQualityExamples = await this.extractLowQualityExamples();
    console.log(`Extracted ${lowQualityExamples.length} low-quality examples`);

    // STEP 3: Create preference pairs for DPO
    const preferencePairs = this.createPreferencePairs(
      highQualityExamples,
      lowQualityExamples
    );
    console.log(`Created ${preferencePairs.length} preference pairs`);

    // STEP 4: Fine-tune model via OpenAI Fine-Tuning API
    const fineTuneJob = await this.triggerFineTuning(preferencePairs);
    console.log(`Fine-tuning job started: ${fineTuneJob.id}`);

    // STEP 5: Update strategy graph with new outcomes
    await this.updateStrategyGraph(highQualityExamples);
    console.log('Strategy graph updated with new evidence\n');

    console.log('✅ Weekly training complete!');
  }

  private async extractHighQualityExamples(): Promise<CoachingActionRecord[]> {
    const result = await pool.query(`
      SELECT *
      FROM coaching_action_corpus
      WHERE
        (outcomes->>'ivyscore_delta')::numeric > 5          -- Significant IvyScore lift
        AND (outcomes->>'adherence_rate')::numeric > 0.8    -- High adherence
        AND (outcomes->>'verified_win')::boolean = true     -- Achieved measurable win
        AND created_at > NOW() - INTERVAL '7 days'          -- Last week only
      ORDER BY (outcomes->>'ivyscore_delta')::numeric DESC
      LIMIT 100
    `);

    return result.rows;
  }

  private async extractLowQualityExamples(): Promise<CoachingActionRecord[]> {
    const result = await pool.query(`
      SELECT *
      FROM coaching_action_corpus
      WHERE
        (outcomes->>'ivyscore_delta')::numeric < 0           -- Negative impact
        OR (outcomes->>'adherence_rate')::numeric < 0.5      -- Low adherence
        AND created_at > NOW() - INTERVAL '7 days'
      ORDER BY (outcomes->>'ivyscore_delta')::numeric ASC
      LIMIT 50
    `);

    return result.rows;
  }

  private createPreferencePairs(
    chosen: CoachingActionRecord[],
    rejected: CoachingActionRecord[]
  ): Array<{prompt: string; chosen: string; rejected: string}> {
    const pairs = [];

    // Match by similar context
    for (const chosenExample of chosen) {
      const matchingRejected = rejected.find(r =>
        r.student_archetype === chosenExample.student_archetype &&
        r.intervention_type === chosenExample.intervention_type
      );

      if (matchingRejected) {
        pairs.push({
          prompt: `Query: ${chosenExample.context.query}\nStudent State: ${JSON.stringify(chosenExample.context.student_state_before)}`,
          chosen: chosenExample.action.response_text,
          rejected: matchingRejected.action.response_text
        });
      }
    }

    return pairs;
  }

  private async triggerFineTuning(
    preferencePairs: Array<{prompt: string; chosen: string; rejected: string}>
  ): Promise<{id: string}> {
    // Convert to OpenAI fine-tuning format
    const trainingData = preferencePairs.map(pair => ({
      messages: [
        { role: 'system', content: 'You are an expert college admissions coach.' },
        { role: 'user', content: pair.prompt },
        { role: 'assistant', content: pair.chosen }  // Chosen response
      ]
    }));

    // Save to JSONL
    const jsonl = trainingData.map(d => JSON.stringify(d)).join('\n');
    const filePath = `/tmp/training_${Date.now()}.jsonl`;
    await fs.writeFile(filePath, jsonl);

    // Upload and start fine-tuning
    const file = await this.openai.files.create({
      file: fs.createReadStream(filePath),
      purpose: 'fine-tune'
    });

    const fineTune = await this.openai.fineTuning.jobs.create({
      training_file: file.id,
      model: 'gpt-4o-mini-2024-07-18',
      suffix: 'ivylevel-weekly'
    });

    return { id: fineTune.id };
  }

  private async updateStrategyGraph(examples: CoachingActionRecord[]): Promise<void> {
    for (const example of examples) {
      if (!example.action.strategy_node_id) continue;

      // Update node outcomes with new evidence
      await pool.query(`
        UPDATE strategy_nodes
        SET
          outcomes = jsonb_set(
            outcomes,
            '{ivyscore_lift_avg}',
            to_jsonb((
              (outcomes->>'ivyscore_lift_avg')::numeric * evidence_count + $2
            ) / (evidence_count + 1))
          ),
          evidence_count = evidence_count + 1,
          success_rate = (
            (success_rate * evidence_count + CASE WHEN $3 THEN 1.0 ELSE 0.0 END)
            / (evidence_count + 1)
          ),
          updated_at = NOW()
        WHERE node_id = $1
      `, [
        example.action.strategy_node_id,
        example.outcomes.ivyscore_delta,
        example.outcomes.verified_win
      ]);
    }
  }
}
```
