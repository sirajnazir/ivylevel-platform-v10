## Executive Summary

This guide provides step-by-step implementation instructions for building the v1.0 Universal Agent Platform with OpenAI components integration.

**Key Strategy:** Use OpenAI Agents SDK as the **execution engine** while preserving v14's **zero-hallucination architecture** (105 temporal views + SQL resolvers).

### What We're Building

```
v14 (Current)                           v1.0 (Target with OpenAI)
─────────────                           ─────────────────────────

jenny-api                               agent-framework
  Single monolithic agent         →       8 modular agents (OpenAI.Agent)
  Custom event loop              →       OpenAI Runner (battle-tested)
  Direct resolver calls          →       v14 resolvers as OpenAI Tools
  Rule-based routing             →       OpenAI Handoffs + LangGraph
  Custom sessions                →       IvyLevelSession extends OpenAI.Session

test-chat-ui                           ivylevel-chat-ui
  Custom React UI (weeks)        →       ChatKit base (hours) + custom widgets

(no visual builder)                    agent-builder
                                  →       Custom React Flow builder
```

### Time Savings from OpenAI

| Component | Custom Build | With OpenAI SDK | Saved |
|-----------|-------------|-----------------|-------|
| Agent Execution | 3 weeks | Built-in | **3 weeks** |
| Tool Calling | 2 weeks | Automatic | **2 weeks** |
| Streaming | 2 weeks | Native | **2 weeks** |
| Handoffs | 2 weeks | Built-in | **2 weeks** |
| UI Base | 4 weeks | ChatKit | **4 weeks** |
| **TOTAL** | **33 weeks** | **20 weeks** | **13 weeks** |

```sql
-- v15_001_knowledge_moat.sql
-- Knowledge Moat: Category 2 Data (DS1-DS8)

BEGIN;

-- DS1: Common Data Set (CDS) - College benchmarks
CREATE TABLE moat_cds_colleges (
  college_id VARCHAR(50) PRIMARY KEY,
  college_name VARCHAR(255) NOT NULL,
  class_year INTEGER NOT NULL,

  -- Admissions stats
  acceptance_rate DECIMAL(5,4),
  total_applicants INTEGER,
  total_admits INTEGER,
  total_enrolled INTEGER,

  -- Academic benchmarks
  gpa_weighted_25 DECIMAL(3,2),
  gpa_weighted_75 DECIMAL(3,2),
  gpa_unweighted_25 DECIMAL(3,2),
  gpa_unweighted_75 DECIMAL(3,2),

  sat_total_25 INTEGER,
  sat_total_75 INTEGER,
  sat_math_25 INTEGER,
  sat_math_75 INTEGER,
  sat_ebrw_25 INTEGER,
  sat_ebrw_75 INTEGER,

  act_composite_25 INTEGER,
  act_composite_75 INTEGER,

  -- Metadata
  ingested_at TIMESTAMP DEFAULT NOW(),
  source_url TEXT
);

CREATE INDEX idx_moat_cds_class_year ON moat_cds_colleges(class_year);

-- DS2: Rubric Factors - Admissions evaluation factors
CREATE TABLE moat_rubric_factors (
  factor_id SERIAL PRIMARY KEY,
  college_id VARCHAR(50) REFERENCES moat_cds_colleges(college_id),
  factor_name VARCHAR(255) NOT NULL,
  factor_category VARCHAR(50), -- academic, extracurricular, personal, etc.
  importance VARCHAR(20), -- critical, very_important, important, considered
  description TEXT,
  scoring_guide TEXT,

  ingested_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_moat_rubric_college ON moat_rubric_factors(college_id);

-- DS3: School Profiles - Hyperlocal high school data
CREATE TABLE moat_school_profiles (
  school_id VARCHAR(50) PRIMARY KEY,
  school_name VARCHAR(255) NOT NULL,
  city VARCHAR(100),
  state VARCHAR(2),

  -- Demographics
  total_students INTEGER,
  student_teacher_ratio DECIMAL(4,1),
  free_reduced_lunch_pct DECIMAL(5,2),

  -- College outcomes
  college_enrollment_rate DECIMAL(5,2),
  four_year_college_pct DECIMAL(5,2),

  -- AP/IB offerings
  ap_courses_offered INTEGER,
  ib_program BOOLEAN,

  ingested_at TIMESTAMP DEFAULT NOW()
);

-- DS4: Student Enrichments - Identity, legacy, hooks
CREATE TABLE moat_student_enrichments (
  student_id VARCHAR(50) REFERENCES students(student_id),

  -- Identity
  first_generation BOOLEAN,
  underrepresented_minority BOOLEAN,
  legacy_status JSONB, -- {college_id: relationship}

  -- Geographic
  rural BOOLEAN,
  urban BOOLEAN,
  suburban BOOLEAN,

  -- Hooks
  recruited_athlete BOOLEAN,
  international BOOLEAN,

  ingested_at TIMESTAMP DEFAULT NOW(),

  PRIMARY KEY (student_id)
);

-- DS5: Past Admits - Twin profile matching
CREATE TABLE moat_past_admits (
  admit_id SERIAL PRIMARY KEY,

  -- Profile (anonymized)
  profile_embedding vector(1536), -- For semantic search
  gpa_weighted DECIMAL(3,2),
  sat_total INTEGER,

  -- Activities summary
  activities_summary TEXT,
  awards_summary TEXT,

  -- Outcomes
  admitted_colleges JSONB, -- [{college_id, scholarship_amount}]
  matriculated_college VARCHAR(50),

  -- Metadata
  class_year INTEGER,
  ingested_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_moat_past_admits_embedding ON moat_past_admits
  USING ivfflat (profile_embedding vector_cosine_ops);

-- DS6: Programs Library - Summer programs + awards catalog
CREATE TABLE moat_programs_library (
  program_id SERIAL PRIMARY KEY,
  program_name VARCHAR(255) NOT NULL,
  program_type VARCHAR(50), -- summer_program, competition, award

  -- Categorization
  subject_areas TEXT[], -- {STEM, CS, Math, Biology, etc.}
  prestige_tier VARCHAR(20), -- international, national, regional, local

  -- Requirements
  eligibility_criteria TEXT,
  time_commitment_hours INTEGER,
  cost_usd INTEGER,

  -- Deadlines
  application_deadline DATE,
  notification_date DATE,

  -- Outcomes
  acceptance_rate DECIMAL(5,2),
  college_admissions_boost TEXT,

  ingested_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_moat_programs_subject ON moat_programs_library
  USING GIN (subject_areas);

-- DS7-DS8: Research Papers - Academic research (Chetty, etc.)
CREATE TABLE moat_research_papers (
  paper_id SERIAL PRIMARY KEY,
  title VARCHAR(500) NOT NULL,
  authors TEXT[],
  published_year INTEGER,

  -- Content
  abstract TEXT,
  key_findings TEXT,
  paper_url TEXT,

  -- Categorization
  topics TEXT[], -- college_admissions, social_mobility, etc.

  ingested_at TIMESTAMP DEFAULT NOW()
);

-- VIEWS for easy querying

CREATE VIEW v_college_benchmarks AS
SELECT
  college_id,
  college_name,
  class_year,
  acceptance_rate,
  gpa_weighted_25,
  gpa_weighted_75,
  sat_total_25,
  sat_total_75
FROM moat_cds_colleges
ORDER BY acceptance_rate ASC;

CREATE VIEW v_hyperlocal_context AS
SELECT
  s.student_id,
  s.high_school,
  sp.school_name,
  sp.college_enrollment_rate,
  sp.ap_courses_offered,
  se.first_generation,
  se.underrepresented_minority
FROM students s
LEFT JOIN moat_school_profiles sp ON s.high_school = sp.school_id
LEFT JOIN moat_student_enrichments se ON s.student_id = se.student_id;

COMMIT;
```

### Issue: Guardrails Failing

**Symptom:** Zero-hallucination guardrail rejects valid response.

**Solution:** Check that tool results are being passed correctly:

```typescript
const guardrailResults = await applyGuardrails(
  message,
  { content: result.content },
  { tool_calls: result.tool_calls } // MUST include tool_calls
```
