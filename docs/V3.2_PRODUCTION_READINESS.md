### N2: HGTI Materialized View Refresh Strategy

**Problem:** Calling `REFRESH MATERIALIZED VIEW` on every growth_events insert causes performance faceplant.

**Risk:** Database locks under concurrent inserts, slow response times.

**Fix:** Scheduled CONCURRENT refresh + on-the-fly delta computation.

**Code:**

```sql
-- ============================================================================
-- Migration 004b: HGTI Materialized View - CONCURRENT Refresh
-- Created: 2025-10-21
-- Purpose: Performance optimization for HGTI scoring
-- ============================================================================

-- Drop existing non-concurrent materialized view
DROP MATERIALIZED VIEW IF EXISTS mv_hgti_scores;

-- Recreate with CONCURRENT refresh support (requires unique index)
CREATE MATERIALIZED VIEW mv_hgti_scores AS
SELECT
  student_id,
  AVG(transformation_delta) AS avg_transformation,
  SUM(CASE WHEN breakthrough THEN 1 ELSE 0 END) AS breakthrough_count,
  COUNT(*) AS total_events,
  MAX(occurred_at) AS last_event_at
FROM growth_events
GROUP BY student_id;

-- CRITICAL: Unique index required for CONCURRENT refresh
CREATE UNIQUE INDEX idx_mv_hgti_student ON mv_hgti_scores(student_id);

-- Refresh function (called by cron job, NOT on insert)
CREATE OR REPLACE FUNCTION refresh_hgti_scores() RETURNS void AS $$
BEGIN
  REFRESH MATERIALIZED VIEW CONCURRENTLY mv_hgti_scores;
END;
$$ LANGUAGE plpgsql;

COMMENT ON FUNCTION refresh_hgti_scores IS 'Refresh HGTI scores (call from cron every 5 minutes)';
```

**GrowthTracker changes:**

```typescript
    // ✅ ADDED: Refresh happens on schedule (every 5 minutes)

  async getHGTIScore(studentId: string): Promise<number> {
    // Read from materialized view (fast)
    const mvResult = await pool.query(
      `SELECT avg_transformation, last_event_at FROM mv_hgti_scores WHERE student_id = $1`,
      [studentId]
    );

    if (mvResult.rows.length === 0) {
      return 0; // No events yet
    }

    const mvScore = mvResult.rows[0].avg_transformation;
    const lastRefresh = mvResult.rows[0].last_event_at;

    // Compute delta since last refresh (for fresh reads)
    const deltaResult = await pool.query(
      `
      SELECT AVG(transformation_delta) AS delta_avg
      FROM growth_events
      WHERE student_id = $1 AND occurred_at > $2
      `,
      [studentId, lastRefresh]
    );

    if (deltaResult.rows.length > 0 && deltaResult.rows[0].delta_avg !== null) {
      // Weighted average: 90% materialized + 10% delta
      return mvScore * 0.9 + deltaResult.rows[0].delta_avg * 0.1;
    }

    return mvScore;
  }
}
```

### N3: RLS Session Variables - Per-Request Isolation

**Problem:** RLS policies use `current_setting('app.student_id')` but session variables can bleed between requests if not reset.

**Risk:** Cross-tenant data leakage.

**Fix:** Set RLS variables per request AND enable RLS on all student-scoped tables.

**Code:**

```typescript
// db/pool.ts (ENHANCED - per-request RLS isolation)
import { Pool, PoolClient } from 'pg';

export const pool = new Pool({ connectionString: process.env.DATABASE_URL });

/**
 * Execute query with RLS context (ENHANCED - guaranteed isolation)
 */
export async function withRLSContext<T>(
  studentId: string,
  coachId: string,
  callback: (client: PoolClient) => Promise<T>
): Promise<T> {
  const client = await pool.connect();
  try {
    // Set RLS variables (LOCAL = transaction-scoped, auto-reset)
    await client.query('BEGIN');
    await client.query(`SELECT set_config('app.student_id', $1, true)`, [studentId]);
    await client.query(`SELECT set_config('app.coach_id', $1, true)`, [coachId]);

    // Execute callback
    const result = await callback(client);

    await client.query('COMMIT');
    return result;
  } catch (error) {
    await client.query('ROLLBACK');
    throw error;
  } finally {
    // Variables are auto-reset on transaction end (true flag above)
    client.release();
  }
}
```

```typescript
// evidence/chip-creator.ts (ENHANCED - PII scrubbing)

const PII_KEYS = new Set(['email', 'phone', 'address', 'ssn', 'dob', 'password', 'api_key']);

/**
 * Scrub PII from params before storing in chip
 */
function scrubParams(params: Record<string, any>): Record<string, any> {
  return Object.fromEntries(
    Object.entries(params).map(([key, value]) => [
      key,
      PII_KEYS.has(key.toLowerCase()) ? '[REDACTED]' : value,
    ])
  );
}

  static createSQLChip(

      system: 'facts_plane',
      ref: resolver,
      params: scrubbedParams, // ← Use scrubbed params
    };

    const hash = this.computeHash('sql', source, studentId);

    return {
      kind: 'sql',
      source,
      hash,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  // Apply to all chip creation methods (narrative, artifact, eq)
}
```

### N5: Exactly-Once Outbox Semantics

**Problem:** Outbox processor is a simple poller. Consumers may receive duplicate events.

**Risk:** Duplicate processing (e.g., chip saved twice, email sent twice).

**Fix:** Idempotent consumers + delivered_key in Redis.

**Code:**

```typescript
// action/outbox-processor.ts (ENHANCED - exactly-once delivery)
import Redis from 'ioredis';
const redis = new Redis(process.env.REDIS_URL);

/**
 * Outbox processor with exactly-once delivery guarantee
 */
export class OutboxProcessor {
  async processUndelivered(): Promise<void> {
    const undelivered = await pool.query(
      `SELECT * FROM outbox WHERE delivered_at IS NULL ORDER BY created_at ASC LIMIT 100`
    );

    for (const event of undelivered.rows) {
      try {
        // Check if already delivered (Redis dedup)
        const deliveryKey = `outbox:delivered:${event.id}`;
        const alreadyDelivered = await redis.get(deliveryKey);

        if (alreadyDelivered) {
          // Mark as delivered in DB (Redis says we already did this)
          await pool.query(
            `UPDATE outbox SET delivered_at = NOW() WHERE id = $1`,
            [event.id]
          );
          continue;
        }

        // Publish event
        await eventBus.publish(event.topic, event.payload);

        // Mark as delivered in Redis (24h TTL)
        await redis.setex(deliveryKey, 86400, '1');

        // Mark as delivered in DB
        await pool.query(
          `UPDATE outbox SET delivered_at = NOW() WHERE id = $1`,
          [event.id]
        );
      } catch (error) {
        console.error('Failed to deliver event:', event.id, error);
        // Retry on next cycle
      }
    }
  }
}

// Run every 5 seconds
setInterval(() => {
  const processor = new OutboxProcessor();
  processor.processUndelivered().catch(err => {
    console.error('Outbox processor error:', err);
  });
}, 5000);
```

```typescript
// Propagate baggage (student_id, agent_name, coach_id)
import { propagation, context, trace } from '@opentelemetry/api';

export function withBaggage<T>(
  studentId: string,
  agentName: string,
  coachId: string,
  fn: () => Promise<T>
): Promise<T> {
  // Set baggage on context
  const currentContext = context.active();
  const baggageContext = propagation.setBaggage(currentContext, {
    'student_id': studentId,
    'agent_name': agentName,
    'coach_id': coachId,
  });

  // Run function with baggage context
  return context.with(baggageContext, fn);
}
```

### N7: EQ Quality Floor and Safety Rails

**Problem:** EQ adapter can produce low-quality or unsafe outputs.

**Risk:** Offensive/biased responses, tone drift from coach's voice.

**Fix:** Similarity scorer + profanity filter + QA samples.

```typescript
import { cosineSimilarity, embed } from '../lib/embeddings.js'; // OpenAI embeddings API

const GLOBAL_BANNED_PHRASES = [
  // Profanity
  'fuck', 'shit', 'damn',
  // Biased language
  'you people', 'those people',
  // Inappropriate
  'stupid', 'idiot', 'dumb',
];

export class EQAdapter {
  async apply(
    rawText: string,
    coachId: string,
    audience: 'student' | 'parent' | 'coach',
    context: { studentId: string; traceId: string }
  ): Promise<{ styledText: string; chip: Chip }> {
    // ... existing logic ...

    // 1. Apply tone transformation
    let styled = this.applyToneVector(rawText, profile.tone_vector, tuning?.tone_overrides, audience);

    // 2. Safety: Filter global banned phrases
    styled = this.filterBannedPhrases(styled, GLOBAL_BANNED_PHRASES);

    // 3. Safety: Filter coach-specific banned phrases
    if (tuning?.banned_phrases) {
      styled = this.filterBannedPhrases(styled, tuning.banned_phrases);
    }

    // 4. Quality: Check similarity to coach's tone distribution
    const similarity = await this.computeStyleSimilarity(rawText, styled, profile);

    if (similarity < 0.75) {
      // Similarity too low, log warning and use raw text
      console.warn(`EQ similarity too low (${similarity}) for coach ${coachId}, using raw text`);
      styled = rawText; // Fallback to raw
    }

    // 5. QA: Sample 1% of responses for audit
    if (Math.random() < 0.01) {
      await this.saveQASample(rawText, styled, profile.coach_id, similarity);
    }

    // 6. Create chip
    const chip = ChipCreator.createEQChip(coachId, profile.tone_vector, audience, context.traceId, context.studentId);

    return { styledText: styled, chip };
  }

  /**
   * Compute cosine similarity between raw and styled embeddings
   */
  private async computeStyleSimilarity(
    rawText: string,
    styledText: string,
    profile: EQProfile
  ): Promise<number> {
    // Embed both texts
    const [rawEmbed, styledEmbed] = await Promise.all([
      embed(rawText),
      embed(styledText),
    ]);

    // Cosine similarity
    return cosineSimilarity(rawEmbed, styledEmbed);
  }

  /**
   * Save QA sample for monthly audit
   */
  private async saveQASample(
    rawText: string,
    styledText: string,
    coachId: string,
    similarity: number
  ): Promise<void> {
    await pool.query(
      `
      INSERT INTO eq_qa_samples (coach_id, raw_text, styled_text, similarity, created_at)
      VALUES ($1, $2, $3, $4, NOW())
      `,
      [coachId, rawText, styledText, similarity]
    );
  }
}
```

**Fix:** Secondary sort by priority, then item_id.

**Code:**

```sql
-- ============================================================================
-- Migration 002c: Deadline UDF - Deterministic Tie-Breaking
-- Created: 2025-10-21
-- Purpose: Ensure consistent ordering when deadlines tie
-- ============================================================================

CREATE OR REPLACE FUNCTION deadline_latest(
  p_student_id text,
  p_category text,
  p_as_of date DEFAULT CURRENT_DATE
) RETURNS TABLE(
  item_id text,
  deadline_date date,
  priority integer
) AS $$
BEGIN
  RETURN QUERY
  SELECT
    d.item_id,
    d.deadline_date,
    COALESCE(d.priority, 1000) AS priority
  FROM v_deadlines_facts d
  WHERE d.student_id = p_student_id
    AND d.category = p_category
    AND d.deadline_date >= p_as_of
  ORDER BY
    d.deadline_date ASC,            -- Primary: earliest deadline
    COALESCE(d.priority, 1000) ASC, -- Secondary: highest priority (lower number)
    d.item_id ASC                   -- Tertiary: alphabetical (deterministic)
  LIMIT 1;
END;
$$ LANGUAGE plpgsql STABLE;

COMMENT ON FUNCTION deadline_latest IS 'Deterministic tie-breaking: date → priority → item_id';
```

**Problem:** Deploying 28% HGTI weight on day one is a shock to users.

```typescript
// resolvers/readiness.ts (ENHANCED - gradual HGTI weight rollout)
export async function computeIvyScore(
  studentId: string,
  options?: { hgtiWeightOverride?: number }
): Promise<{ score: number; weights: Record<string, number>; audit: any }> {
  // Feature flag for HGTI weight (gradual rollout)
  const featureFlags = await getFeatureFlags(studentId);
  const hgtiWeight = options?.hgtiWeightOverride ?? featureFlags.hgti_weight;

  // Weights (must sum to 1.0)
  const weights = {
    academic: 0.40,
    achievement: (1.0 - 0.40 - hgtiWeight), // Adjust achievement to maintain sum
    hgti: hgtiWeight,
  };

  // Compute scores
  const academicScore = await getAcademicScore(studentId);
  const achievementScore = await getAchievementScore(studentId);
  const growthTracker = new GrowthTracker();
  const hgtiScore = await growthTracker.getHGTIScore(studentId);

  // Weighted IvyScore
  const ivyScore = (
    academicScore * weights.academic +
    achievementScore * weights.achievement +
    hgtiScore * weights.hgti
  );

  // Audit log (for reconstructing decisions)
  const audit = {
    student_id: studentId,
    weights,
    scores: { academic: academicScore, achievement: achievementScore, hgti: hgtiScore },
    ivy_score: ivyScore,
    computed_at: new Date().toISOString(),
  };

  // Store audit log
  await pool.query(
    `
    INSERT INTO ivyscore_audit (student_id, weights, scores, ivy_score, computed_at)
    VALUES ($1, $2, $3, $4, $5)
    `,
    [studentId, weights, audit.scores, ivyScore, new Date()]
  );

  return { score: ivyScore, weights, audit };
}

/**
 * Get feature flags for student (A/B cohorts)
 */
async function getFeatureFlags(studentId: string): Promise<{ hgti_weight: number }> {
  // Hash student ID to assign cohort (deterministic)
  const hash = crypto.createHash('md5').update(studentId).digest('hex');
  const cohort = parseInt(hash.substring(0, 2), 16) % 100; // 0-99

  // Gradual rollout:
  // Week 1: 10% of students get 10% weight
  // Week 3: 20% of students get 20% weight
  // Week 5: 100% of students get 28% weight

  const now = new Date();
  const startDate = new Date('2025-11-01'); // Deployment date
  const weeksSinceStart = Math.floor((now.getTime() - startDate.getTime()) / (7 * 24 * 60 * 60 * 1000));

  if (weeksSinceStart < 1) {
    // Week 0: No HGTI weight yet
    return { hgti_weight: 0.0 };
  } else if (weeksSinceStart < 3) {
    // Week 1-2: 10% of students get 10% weight
    return { hgti_weight: cohort < 10 ? 0.10 : 0.0 };
  } else if (weeksSinceStart < 5) {
    // Week 3-4: 20% of students get 20% weight
    return { hgti_weight: cohort < 20 ? 0.20 : 0.0 };
  } else {
    // Week 5+: All students get 28% weight
    return { hgti_weight: 0.28 };
  }
}
```

```typescript
// cognition/tool-bus.ts (ENHANCED - cached validators)
import Ajv from 'ajv';

const ajv = new Ajv();
const validatorCache = new Map<string, any>(); // Cache compiled validators

export class ToolBus {
  register(manifest: ToolManifest, executor: ToolExecutor): void {
    // Check schema version compatibility
    if (manifest.schema_version !== CURRENT_MANIFEST_SCHEMA_VERSION) {
      throw new Error(
        `Incompatible manifest schema version: ${manifest.schema_version} (expected ${CURRENT_MANIFEST_SCHEMA_VERSION})`
      );
    }

    // Compile and cache input validator
    const inputValidator = ajv.compile(manifest.inputs);
    validatorCache.set(`${manifest.name}:inputs`, inputValidator);

    // Register tool
    this.registry.set(manifest.name, { manifest, executor });
    log.info(`Tool registered: ${manifest.name} v${manifest.version} (schema ${manifest.schema_version})`);
  }

  async execute(toolName: string, params: any, context: ExecutionContext): Promise<ToolResult> {
    const entry = this.registry.get(toolName);
    if (!entry) {
      throw new Error(`Tool ${toolName} not registered`);
    }

    // Validate inputs with cached validator
    const inputValidator = validatorCache.get(`${toolName}:inputs`);
    if (!inputValidator) {
      throw new Error(`No input validator for ${toolName}`);
    }

    const valid = inputValidator(params);
    if (!valid) {
      throw new Error400(`Invalid params for ${toolName}: ${ajv.errorsText(inputValidator.errors)}`);
    }

    // ... rest of execution
  }
}
```
