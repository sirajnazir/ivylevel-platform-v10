### 2. **Detecting A2A Handover in API Response**

**Backend Response Structure (v28.0.5):**
```typescript
// POST /api/v26/agents/:agentId/message response
interface AgentMessageResponse {
  user_message_id: string;
  agent_message_id: string;
  agent_response: string;
  processing_time: number;
  confidence: number;
  intelligence_triggered: string[];
  metadata: {
    a2a_handover_complete?: boolean;
    handover_id?: string;
    agent_id?: string;  // New agent ID if handover occurred
  };
  // v28.0: A2A Handover object (present if handover occurred)
  a2a_handover: {
    handover_complete: true;
    handover_id: string;
    from_agent: string;      // e.g., "assessment-agent-v18"
    to_agent: string;         // e.g., "gameplan-agent"
    new_phase: string;        // e.g., "gameplan"
  } | null;
  v26_context: {
    real_student_id: string;
    clone_student_id: string;
    is_clone_student: boolean;
  };
}
```

**Frontend Detection Logic:**
```typescript
// In your message handler (likely agentClient.ts or similar)
async function sendMessageToAgent(agentId: string, message: string) {
  const response = await fetch(`/api/v26/agents/${agentId}/message`, {
    method: 'POST',
    body: JSON.stringify({ session_id, message, student_id }),
    headers: { 'Content-Type': 'application/json' }
  });

  const data: AgentMessageResponse = await response.json();

  // ✅ v28.0: Check for A2A handover
  if (data.a2a_handover && data.a2a_handover.handover_complete) {
    console.log('[FRONTEND_A2A] Handover detected!', {
      from: data.a2a_handover.from_agent,
      to: data.a2a_handover.to_agent,
      handover_id: data.a2a_handover.handover_id
    });

    // Update agent states
    handleAgentHandover({
      fromAgent: data.a2a_handover.from_agent,
      toAgent: data.a2a_handover.to_agent,
      handoverId: data.a2a_handover.handover_id,
      newPhase: data.a2a_handover.new_phase
    });
  }

  return data;
}
```

### 3. **State Management for Agent Cards**

**Recommended State Structure:**
```typescript
// In your MultiAgentChat component or context
interface AgentState {
  id: string;
  name: string;
  icon: string;
  status: 'ready' | 'active' | 'handed_off';
  handover_info?: {
    handed_to: string;
    handover_id: string;
    timestamp: Date;
  };
}

const [agents, setAgents] = useState<AgentState[]>([
  {
    id: 'assessment-agent-v18',
    name: 'Assessment Agent',
    icon: '📊',
    status: 'active' // Initially active
  },
  {
    id: 'gameplan-agent',
    name: 'GamePlan Agent',
    icon: '🎯',
    status: 'ready'
  },
  {
    id: 'execution-agent',
    name: 'Execution Agent',
    icon: '🚀',
    status: 'ready'
  },
  // ... other agents
]);
```

**Handover Handler Function:**
```typescript
function handleAgentHandover(handover: {
  fromAgent: string;
  toAgent: string;
  handoverId: string;
  newPhase: string;
}) {
  setAgents(prev => prev.map(agent => {
    // Mark previous agent as handed off
    if (agent.id === handover.fromAgent) {
      return {
        ...agent,
        status: 'handed_off',
        handover_info: {
          handed_to: handover.toAgent,
          handover_id: handover.handoverId,
          timestamp: new Date()
        }
      };
    }

    // Mark new agent as active
    if (agent.id === handover.toAgent) {
      return {
        ...agent,
        status: 'active'
      };
    }

    return agent;
  }));

  // Optionally scroll to new active agent
  scrollToAgent(handover.toAgent);

  // Log handover event
  addIntelligenceLog({
    type: 'a2a_handover',
    from_agent: handover.fromAgent,
    to_agent: handover.toAgent,
    handover_id: handover.handoverId,
    timestamp: new Date()
  });
}
```

### 5. **Automatic Agent Switching**

**When handover occurs:**
1. Collapse or minimize previous agent's chat
2. Expand and highlight new agent's chat
3. **Auto-focus** message input for new agent
4. Optionally show transition animation

**Example Implementation:**
```typescript
useEffect(() => {
  if (currentActiveAgent !== previousActiveAgent) {
    // Agent changed due to handover

    // Scroll to new agent card
    const agentCard = document.getElementById(`agent-${currentActiveAgent}`);
    agentCard?.scrollIntoView({ behavior: 'smooth', block: 'center' });

    // Highlight with brief animation
    agentCard?.classList.add('agent-card--transition');
    setTimeout(() => {
      agentCard?.classList.remove('agent-card--transition');
    }, 2000);

    // Focus message input
    const messageInput = document.getElementById(`message-input-${currentActiveAgent}`);
    messageInput?.focus();
  }
}, [currentActiveAgent]);
```
