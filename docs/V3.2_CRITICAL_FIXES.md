## G1: FACTS CONTAMINATION (kb_items misuse)

### Problem

Original plan has temporal UDFs reading from `kb_items` table:

```sql
-- ❌ WRONG: Reading "facts" from narrative store
SELECT * FROM kb_items
WHERE student_id = $1
  AND item_type = 'Award_Competition'
  AND tier1_state = 'Outcome'
```

**Risk:** Silent hallucinations. kb_items is narrative/KB territory (soft, human-authored notes). Enumerated facts must come from authoritative tables.

### Fix

Create authoritative facts views that map to real tables:

```sql
-- ✅ CORRECT: Authoritative facts view
CREATE OR REPLACE VIEW v_awards_facts AS
SELECT
  student_id,
  award_id,
  award_name AS title_name,
  award_tier AS tier,
  result_date AS outcome_date,
  result_status AS status
FROM student_awards
WHERE result_status IN ('WIN', 'FINALIST', 'HONORABLE_MENTION');

-- ✅ CORRECT: UDF reads from facts view
CREATE OR REPLACE FUNCTION award_nth(p_student_id text, p_n int, p_as_of date)
RETURNS TABLE(...) AS $$
  SELECT * FROM v_awards_facts
  WHERE student_id = p_student_id AND outcome_date <= p_as_of
  ORDER BY outcome_date ASC
  OFFSET GREATEST(p_n-1,0) LIMIT 1;
$$ LANGUAGE sql STABLE;
```

**Apply to:**
- `award_nth`, `award_first`, `award_latest` → read from `v_awards_facts`
- `sat_nth`, `sat_latest` → read from `v_tests_facts`
- `gpa_as_of` → read from `v_gpa_facts`
- `deadline_latest` → read from `v_deadlines_facts`

**Acceptance:**
- ✅ All temporal UDFs query facts tables/views only
- ✅ NO queries to kb_items from FactsPlane
- ✅ Golden tests pass with known data

## G2: NO IDEMPOTENCY/OUTBOX

### Problem

ToolBus logs to `system_events`, updates `agent_runs`, persists `chips` with no idempotency protection:

```typescript
// ❌ WRONG: No idempotency key
await pool.query(`INSERT INTO system_events (...) VALUES (...)`, [...]);
await pool.query(`UPDATE agent_runs SET tool_calls = tool_calls + 1 WHERE id = $1`, [runId]);
await chipRepo.save(chip);
```

**Risk:** Retries cause duplicate events, phantom chips, incorrect usage counts.

### Fix

Add outbox pattern with idempotency keys:

**1. Outbox Table:**

```sql
CREATE TABLE outbox (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  topic text NOT NULL,
  payload jsonb NOT NULL,
  idempotency_key text UNIQUE,
  created_at timestamptz DEFAULT now(),
  delivered_at timestamptz
);

CREATE INDEX idx_outbox_undelivered ON outbox(created_at) WHERE delivered_at IS NULL;
```

**2. Idempotent Writes:**

```typescript
// ✅ CORRECT: Write with idempotency key in transaction
async function logToolExecution(
  toolName: string,
  latency: number,
  chip: Chip,
  context: ExecutionContext
): Promise<void> {
  const idempotencyKey = `${context.agent_run_id}:${toolName}:${Date.now()}`;

  const client = await pool.connect();
  try {
    await client.query('BEGIN');

    // Insert into outbox with idempotency key
    await client.query(
      `
      INSERT INTO outbox (topic, payload, idempotency_key)
      VALUES ($1, $2, $3)
      ON CONFLICT (idempotency_key) DO NOTHING
      `,
      ['tool.executed', { toolName, latency, chip }, idempotencyKey]
    );

    // Update agent_runs
    await client.query(
      `UPDATE agent_runs SET tool_calls = tool_calls + 1, total_latency_ms = total_latency_ms + $1 WHERE id = $2`,
      [latency, context.agent_run_id]
    );

    await client.query('COMMIT');
  } catch (error) {
    await client.query('ROLLBACK');
    throw error;
  } finally {
    client.release();
  }
}
```

**3. Outbox Processor (background job):**

```typescript
// Process undelivered outbox events
setInterval(async () => {
  const undelivered = await pool.query(
    `SELECT * FROM outbox WHERE delivered_at IS NULL ORDER BY created_at ASC LIMIT 100`
  );

  for (const event of undelivered.rows) {
    try {
      await eventBus.publish(event.topic, event.payload);

      await pool.query(
        `UPDATE outbox SET delivered_at = NOW() WHERE id = $1`,
        [event.id]
      );
    } catch (error) {
      console.error('Failed to deliver event:', error);
      // Retry on next cycle
    }
  }
}, 5000); // Every 5 seconds
```

**Acceptance:**
- ✅ All writes use idempotency keys
- ✅ Retries don't create duplicates
- ✅ Outbox processor delivers events reliably

```typescript
// ✅ CORRECT: Real implementation
private async getCurrentUsage(agentRunId: string): Promise<{ tokens: number; elapsed_ms: number }> {
  const result = await pool.query(
    `SELECT tokens_used, total_latency_ms FROM agent_runs WHERE id = $1`,
    [agentRunId]
  );

  if (result.rows.length === 0) {
    throw new Error(`Agent run ${agentRunId} not found`);
  }

  return {
    tokens: result.rows[0].tokens_used,
    elapsed_ms: result.rows[0].total_latency_ms,
  };
}
```

**3. Redis Rate Limiter:**

```typescript
// ✅ CORRECT: Redis sliding window rate limiter
private async getRateLimitCount(key: string): Promise<number> {
  // Increment and set 60s TTL
  const count = await redis.incr(`ratelimit:${key}`);

  if (count === 1) {
    // First increment, set TTL
    await redis.expire(`ratelimit:${key}`, 60);
  }

  return count;
}
```

**4. Budget Check:**

```typescript
// ✅ CORRECT: Real budget enforcement
private async checkBudget(
  cost: { estimated_tokens: number; estimated_latency_ms: number },
  context: ExecutionContext
): Promise<void> {
  const usage = await this.getCurrentUsage(context.agent_run_id);

  if (usage.tokens + cost.estimated_tokens > context.token_budget) {
    // Update status to budget_exceeded
    await pool.query(
      `UPDATE agent_runs SET status = 'budget_exceeded' WHERE id = $1`,
      [context.agent_run_id]
    );

    throw new Error429(`Token budget exceeded (${usage.tokens + cost.estimated_tokens} > ${context.token_budget})`);
  }

  if (usage.elapsed_ms + cost.estimated_latency_ms > context.time_budget_ms) {
    throw new Error429(`Time budget exceeded`);
  }
}
```

### Problem

OTel spans are "best effort":

```typescript
// ❌ WRONG: Optional tracing

  await tracer.startSpan(...);
}
```

**Risk:** Traces missing when you need them most (production incidents).

### Fix

**1. Mandatory Span Wrapper:**

```typescript
// ✅ CORRECT: Mandatory span enforcement
export async function withSpan<T>(
  operationName: string,
  layer: 'perception' | 'knowledge' | 'cognition' | 'action' | 'interface',
  fn: (span: Span) => Promise<T>
): Promise<T> {
  return tracer.startActiveSpan(operationName, async (span) => {
    span.setAttribute('layer', layer);
    span.setAttribute('timestamp', Date.now());

    try {
      const result = await fn(span);
      span.setStatus({ code: SpanStatusCode.OK });
      return result;
    } catch (error) {
      span.setStatus({ code: SpanStatusCode.ERROR, message: error.message });
      span.recordException(error);
      throw error;
    } finally {
      span.end();
    }
  });
}
```

**2. Apply to Hot Paths:**

```typescript

  return withSpan(`tool.${toolName}`, 'cognition', async (span) => {
    span.setAttribute('tool_name', toolName);
    span.setAttribute('trace_id', context.trace_id);

    // ... execution logic
  });
}

// FactsPlane.query
async query(query: FactQuery, context: ExecutionContext): Promise<FactResult> {
  return withSpan(`facts.${query.resolver}`, 'knowledge', async (span) => {
    span.setAttribute('resolver', query.resolver);
    span.setAttribute('student_id', context.student_id);

    // ... query logic
  });
}
```

**2. EQ Adapter Middleware:**

```typescript
// cognition/eq-adapter.ts
export class EQAdapter {
  async apply(
    rawText: string,
    coachId: string,
    audience: 'student' | 'parent' | 'coach',
    context: { studentId: string; traceId: string }
  ): Promise<{ styledText: string; chip: Chip }> {
    // 1. Load profile
    const profile = await this.loadProfile(coachId);

    // 2. Load student-specific tuning
    const tuning = await this.loadTuning(context.studentId, coachId);

    // 3. Apply tone transformation
    let styled = rawText;

    // Lexical substitutions
    for (const [from, to] of Object.entries(profile.lex_map)) {
      styled = styled.replace(new RegExp(from, 'gi'), to);
    }

    // Tone scalars (sentence length, hedging, emojis)
    styled = this.applyToneVector(styled, profile.tone_vector, audience);

    // Filter banned phrases
    styled = this.filterBanned(styled, tuning?.banned_phrases || []);

    // 4. Create EQ chip
    const chip = ChipCreator.createEQChip(
      coachId,
      profile.tone_vector,
      audience,
      context.traceId,
      context.studentId
    );

    return { styledText: styled, chip };
  }

  private applyToneVector(
    text: string,
    toneVector: Record<string, number>,
    audience: string
  ): string {
    // Adjust based on tone scalars
    let adjusted = text;

    // Warmth (0-1): Add/remove softeners
    if (toneVector.warmth > 0.7) {
      adjusted = adjusted.replace(/\./g, '!'); // More enthusiasm
    }

    // Directness (0-1): Shorten sentences if high
    if (toneVector.direct > 0.7) {
      adjusted = adjusted.replace(/ however,/gi, ','); // Remove hedging
    }

    return adjusted;
  }

  private filterBanned(text: string, banned: string[]): string {
    let filtered = text;
    for (const phrase of banned) {
      filtered = filtered.replace(new RegExp(phrase, 'gi'), '***');
    }
    return filtered;
  }
}
```

### Fix

**1. Schema:**

```sql
CREATE TABLE growth_events (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id text NOT NULL,
  barrier_type text CHECK (barrier_type IN (
    'INTERNAL_CONFIDENCE', 'SOCIAL_EXCLUSION', 'NEURODIVERSITY',
    'PARENT_CONFLICT', 'SELF_IMAGE', 'MOTIVATION_DROP', 'BURNOUT', 'TIME_MANAGEMENT'
  )),
  trigger text,
  coach_reflection text,
  student_reflection text, -- Private, not shown to parents
  breakthrough boolean DEFAULT false,
  transformation_delta numeric CHECK (transformation_delta BETWEEN 0 AND 1),
  occurred_at date NOT NULL,
  linked_artifacts jsonb DEFAULT '[]'::jsonb, -- Links to chips, essays, etc.
  created_at timestamptz DEFAULT now()
);

CREATE INDEX idx_growth_events_student ON growth_events(student_id, occurred_at DESC);

-- Derived scores (materialized view for performance)
CREATE MATERIALIZED VIEW mv_hgti_scores AS
SELECT
  student_id,
  AVG(transformation_delta) AS avg_transformation,
  SUM(CASE WHEN breakthrough THEN 1 ELSE 0 END) AS breakthrough_count,
  COUNT(*) AS total_events
FROM growth_events
GROUP BY student_id;

CREATE UNIQUE INDEX idx_mv_hgti_student ON mv_hgti_scores(student_id);
```

**2. Growth Tracker:**

```typescript
// hgti/growth-tracker.ts
export class GrowthTracker {
  async recordEvent(event: {
    studentId: string;
    barrierType: string;
    trigger: string;
    coachReflection: string;
    studentReflection?: string;
    breakthrough: boolean;
    transformationDelta: number; // 0-1
    linkedArtifacts: string[];
  }): Promise<void> {
    await pool.query(
      `
      INSERT INTO growth_events (
        student_id, barrier_type, trigger, coach_reflection, student_reflection,
        breakthrough, transformation_delta, occurred_at, linked_artifacts
      )
      VALUES ($1, $2, $3, $4, $5, $6, $7, CURRENT_DATE, $8)
      `,
      [
        event.studentId,
        event.barrierType,
        event.trigger,
        event.coachReflection,
        event.studentReflection,
        event.breakthrough,
        event.transformationDelta,
        JSON.stringify(event.linkedArtifacts),
      ]
    );

    // Refresh materialized view
    await pool.query(`REFRESH MATERIALIZED VIEW mv_hgti_scores`);
  }

  async getTimeline(studentId: string): Promise<GrowthEvent[]> {
    const result = await pool.query(
      `
      SELECT * FROM growth_events
      WHERE student_id = $1
      ORDER BY occurred_at DESC
      `,
      [studentId]
    );

    return result.rows;
  }

  async getHGTIScore(studentId: string): Promise<number> {
    const result = await pool.query(
      `SELECT avg_transformation FROM mv_hgti_scores WHERE student_id = $1`,
      [studentId]
    );

    return result.rows[0]?.avg_transformation || 0;
  }
}
```

```typescript
  // Achievement score (32%): awards, ECs, programs

  const ivyScore = (

    achievementScore * 0.32 +
    hgtiScore * 0.28
  );

  return ivyScore;
}
```
