## EXECUTIVE SUMMARY

### Core Philosophy: INCREMENTAL ENHANCEMENT

**We will NOT rebuild.** We will enhance the existing v3.1 architecture incrementally, wave by wave, preserving all working components while adding PRD-required features.

### Clean Architecture Principles (Applied Incrementally)

1. **Separation of Concerns** - Each module has one responsibility
2. **Dependency Inversion** - High-level modules don't depend on low-level details
3. **Interface Segregation** - Small, focused interfaces
4. **Single Responsibility** - One reason to change per module
5. **Open/Closed** - Open for extension, closed for modification
6. **DRY (Don't Repeat Yourself)** - Canonical implementations only

### First-Principles Naming Conventions

**Before (v3.1):** Mixed abstractions, unclear boundaries
```
resolverTools.ts  // What kind of tools? SQL? API? Hybrid?
compose.ts        // Composes what? How?
humanizer.ts      // Humanizes what? When?
```

**After (v3.2):** Clear, intention-revealing names
```
knowledge/facts-plane.ts       // SQL-only, deterministic facts
knowledge/narrative-plane.ts   // RAG-only, soft matches
evidence/chip-creator.ts       // Creates provenance chips
cognition/tool-bus.ts          // Central tool execution bus
cognition/policy-gate.ts       // Access control + governance
cognition/eq-adapter.ts        // Emotional tone adaptation
```

-- Chips table: Universal provenance ledger
CREATE TABLE chips (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  kind text NOT NULL CHECK (kind IN ('sql', 'narrative', 'artifact', 'eq')),
  source jsonb NOT NULL, -- {system: 'facts_plane', ref: 'award_nth', params: {...}}
  hash text, -- SHA256 of (kind + source + student_id) for dedup
  trace_id text NOT NULL, -- OpenTelemetry trace ID for debugging
  student_id text NOT NULL,
  created_at timestamptz DEFAULT now()
);

-- Indexes for performance
CREATE INDEX idx_chips_trace ON chips(trace_id);
CREATE INDEX idx_chips_student ON chips(student_id, created_at DESC);
CREATE INDEX idx_chips_kind ON chips(kind, created_at DESC);
CREATE INDEX idx_chips_hash ON chips(hash) WHERE hash IS NOT NULL;

-- Comments for documentation
COMMENT ON TABLE chips IS 'Universal provenance ledger for all data sources';
COMMENT ON COLUMN chips.kind IS 'Type: sql (fact), narrative (RAG), artifact (file), eq (tone)';
COMMENT ON COLUMN chips.source IS 'JSONB: {system, ref, params} for reproducibility';
COMMENT ON COLUMN chips.hash IS 'SHA256 hash for deduplication';
COMMENT ON COLUMN chips.trace_id IS 'OpenTelemetry trace ID for debugging';

-- Get nth award as of a specific date
CREATE OR REPLACE FUNCTION award_nth(
  p_student_id text,
  p_n integer,
  p_as_of date DEFAULT CURRENT_DATE
) RETURNS TABLE(
  item_id text,
  title_name text,
  tier1_state text,
  tier2_substate text,
  outcome_date date,
  key_metric_value text
) AS $$
BEGIN

    kb.item_id,
    kb.title_name,

    kb.tier2_substate,
    kb.outcome_date,
    kb.key_metric_value

    AND kb.item_type = 'Award_Competition'
    AND kb.tier1_state = 'Outcome'
    AND kb.outcome_date <= p_as_of
  ORDER BY kb.outcome_date ASC
  LIMIT 1 OFFSET (p_n - 1);
END;
$$ LANGUAGE plpgsql STABLE;

### Task 1.3: Chip Creator Module (Day 2-3)

**File:** `services/agent-framework/src/evidence/chip-creator.ts`

```typescript
/**
 * chip-creator.ts
 * Universal chip creation for provenance tracking
 * Created: 2025-10-21 (Wave 1)
 *
 * Design Principles:
 * - Single Responsibility: Only creates chips, doesn't persist them
 * - Dependency Inversion: Returns data objects, doesn't know about DB
 * - Interface Segregation: Separate method per chip type
 */

import crypto from 'crypto';
import type { Pool } from 'pg';

// ============================================================================
// Type Definitions (Clean Interfaces)
// ============================================================================

export type ChipKind = 'sql' | 'narrative' | 'artifact' | 'eq';

export interface ChipSource {
  system: string; // 'facts_plane', 'narrative_plane', 'eq_adapter'
  ref: string; // Function/tool name
  params?: Record<string, any>; // Input parameters for reproducibility
}

export interface Chip {
  id?: string; // UUID (generated by DB)
  kind: ChipKind;
  source: ChipSource;
  hash?: string; // SHA256 for deduplication
  trace_id: string; // OpenTelemetry trace ID
  student_id: string;
  created_at?: Date;
}

// ============================================================================
// Chip Creator (Pure Functions)
// ============================================================================

export class ChipCreator {
  /**
   * Create SQL chip for fact-based queries
   * Example: award_nth, gpa_as_of, deadline_latest
   */
  static createSQLChip(
    resolver: string,
    params: Record<string, any>,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'facts_plane',
      ref: resolver,
      params,
    };

    const hash = this.computeHash('sql', source, studentId);

    return {
      kind: 'sql',
      source,
      hash,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create narrative chip for RAG retrievals
   * Example: DS6 essay examples, DS7 AO perspectives, DS-T tactics
   */
  static createNarrativeChip(
    docId: string,
    section: string,
    score: number,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'narrative_plane',
      ref: docId,
      params: { section, score },
    };

    // No hash for narrative chips (not meant to be deduped)
    return {
      kind: 'narrative',
      source,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create artifact chip for file uploads
   * Example: essays, transcripts, recommendation letters
   */
  static createArtifactChip(
    url: string,
    artifactType: string,
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'artifact_store',
      ref: url,
      params: { type: artifactType },
    };

    const hash = this.computeHash('artifact', source, studentId);

    return {
      kind: 'artifact',
      source,
      hash,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Create EQ chip for tone adaptation
   * Example: Jenny's tone applied to response
   */
  static createEQChip(
    coachId: string,
    toneVector: Record<string, number>,
    audience: 'student' | 'parent' | 'coach',
    traceId: string,
    studentId: string
  ): Chip {
    const source: ChipSource = {
      system: 'eq_adapter',
      ref: coachId,
      params: { tone_vector: toneVector, audience },
    };

    // No hash for EQ chips (each application is unique)
    return {
      kind: 'eq',
      source,
      trace_id: traceId,
      student_id: studentId,
    };
  }

  /**
   * Compute SHA256 hash for deduplication
   * Hash = SHA256(kind + system + ref + params_json + student_id)
   */
  private static computeHash(
    kind: ChipKind,
    source: ChipSource,
    studentId: string
  ): string {
    const paramsJson = source.params ? JSON.stringify(source.params) : '';
    const input = `${kind}:${source.system}:${source.ref}:${paramsJson}:${studentId}`;

    return crypto.createHash('sha256').update(input).digest('hex');
  }
}

// ============================================================================
// Chip Persistence (Repository Pattern)
// ============================================================================

export class ChipRepository {
  constructor(private pool: Pool) {}

  /**
   * Save chip to database
   * Returns chip with generated ID
   */
  async save(chip: Chip): Promise<Chip> {
    const result = await this.pool.query(
      `
      INSERT INTO chips (kind, source, hash, trace_id, student_id)
      VALUES ($1, $2, $3, $4, $5)
      RETURNING id, created_at
      `,
      [chip.kind, JSON.stringify(chip.source), chip.hash, chip.trace_id, chip.student_id]
    );

    return {
      ...chip,
      id: result.rows[0].id,
      created_at: result.rows[0].created_at,
    };
  }

  /**
   * Save multiple chips in a single transaction
   */
  async saveMany(chips: Chip[]): Promise<Chip[]> {
    if (chips.length === 0) return [];

    const values = chips.map((c, i) => {
      const offset = i * 5;
      return `($${offset + 1}, $${offset + 2}, $${offset + 3}, $${offset + 4}, $${offset + 5})`;
    }).join(', ');

    const params = chips.flatMap(c => [
      c.kind,
      JSON.stringify(c.source),
      c.hash,
      c.trace_id,
      c.student_id,
    ]);

    const result = await this.pool.query(
      `
      INSERT INTO chips (kind, source, hash, trace_id, student_id)
      VALUES ${values}
      RETURNING id, created_at
      `,
      params
    );

    return chips.map((c, i) => ({
      ...c,
      id: result.rows[i].id,
      created_at: result.rows[i].created_at,
    }));
  }

  /**
   * Get chips by trace_id (for debugging)
   */
  async getByTraceId(traceId: string): Promise<Chip[]> {
    const result = await this.pool.query(
      `SELECT * FROM chips WHERE trace_id = $1 ORDER BY created_at ASC`,
      [traceId]
    );

    return result.rows.map(row => ({
      id: row.id,
      kind: row.kind,
      source: row.source,
      hash: row.hash,
      trace_id: row.trace_id,
      student_id: row.student_id,
      created_at: row.created_at,
    }));
  }

  /**
   * Get chips by student (for Evidence Panel UI)
   */
  async getByStudent(studentId: string, limit = 100): Promise<Chip[]> {
    const result = await this.pool.query(
      `SELECT * FROM chips WHERE student_id = $1 ORDER BY created_at DESC LIMIT $2`,
      [studentId, limit]
    );

    return result.rows.map(row => ({
      id: row.id,
      kind: row.kind,
      source: row.source,
      hash: row.hash,
      trace_id: row.trace_id,
      student_id: row.student_id,
      created_at: row.created_at,
    }));
  }
}
```

/**
 * policy-gate.ts
 * Access control and resource governance for tool execution
 * Created: 2025-10-21 (Wave 2)
 *
 * Design Principles:
 * - Fail-Secure: Deny by default, allow explicitly
 * - Defense in Depth: Multiple layers of checks
 * - Least Privilege: Grant minimum required permissions
 */

import type { ToolManifest } from './tool-manifest.js';
import { pool } from '../db/pool.js';

export interface ExecutionContext {
  // Identity
  student_id: string;
  coach_id: string;
  agent_name: string;

  // Authorization
  user_scopes: string[]; // What permissions does this context have?

  // Resource Limits
  token_budget: number; // Max tokens for this agent run
  time_budget_ms: number; // Max execution time

  // Tracing
  trace_id: string;
  agent_run_id: string;
}

export class PolicyGate {
  /**
   * Check if tool execution is allowed
   * Throws Error403 or Error429 if not
   */
  async check(manifest: ToolManifest, context: ExecutionContext): Promise<void> {
    // 1. Scope check (authorization)
    this.checkScopes(manifest.scopes, context.user_scopes);

    // 2. Budget check (resource governance)
    await this.checkBudget(manifest.cost, context);

    // 3. RLS check (data access control)
    await this.checkRLS(context.coach_id, context.student_id);

    // 4. Rate limit check (abuse prevention)
    await this.checkRateLimit(context.agent_name, manifest.name);
  }

  /**
   * Check scopes (permissions)
   */
  private checkScopes(required: string[], available: string[]): void {
    const missing = required.filter(scope => !available.includes(scope));

    if (missing.length > 0) {
      throw new Error403(`Missing required scopes: ${missing.join(', ')}`);
    }
  }

  /**
   * Check budget (token + time limits)
   */
  private async checkBudget(
    cost: { estimated_tokens: number; estimated_latency_ms: number },
    context: ExecutionContext
  ): Promise<void> {
    // Get current usage for this agent run
    const usage = await this.getCurrentUsage(context.agent_run_id);

    // Check token budget
    if (usage.tokens + cost.estimated_tokens > context.token_budget) {
      throw new Error429(`Token budget exceeded (${usage.tokens + cost.estimated_tokens} > ${context.token_budget})`);
    }

    // Check time budget
    if (usage.elapsed_ms + cost.estimated_latency_ms > context.time_budget_ms) {
      throw new Error429(`Time budget exceeded`);
    }
  }

  /**
   * Check RLS (coach authorized for student)
   */
  private async checkRLS(coachId: string, studentId: string): Promise<void> {
    const result = await pool.query(
      `SELECT 1 FROM students WHERE student_id = $1 AND coach_id = $2`,
      [studentId, coachId]
    );

    if (result.rows.length === 0) {
      throw new Error403(`Coach ${coachId} not authorized for student ${studentId}`);
    }
  }

  /**
   * Check rate limit (per-agent-tool pair)
   */
  private async checkRateLimit(agentName: string, toolName: string): Promise<void> {
    // Simple rate limit: max 100 calls per tool per agent per minute

    const count = await this.getRateLimitCount(key);

      throw new Error429(`Rate limit exceeded for ${key} (${count}/100 per minute)`);

  }
}

**File:** `services/agent-framework/src/cognition/tool-bus.ts`

```typescript
/**
 * tool-bus.ts
 * Central tool execution hub with governance and observability
 * Created: 2025-10-21 (Wave 2)
 *
 * Design Principles:
 * - Single Entry Point: All tools go through here
 * - Policy Enforcement: Check before execute
 * - Observable: Log every execution with latency + chips
 */

import type { ToolManifest } from './tool-manifest.js';
import type { ExecutionContext } from './policy-gate.js';
import { PolicyGate } from './policy-gate.js';
import { ChipRepository, type Chip } from '../evidence/chip-creator.js';
import { pool } from '../db/pool.js';
import { createLogger } from '../../../../packages/observability/dist/unified-logger.js';

const log = createLogger('tool-bus');

export interface ToolResult {
  data: any;
  chip?: Chip;
  latency_ms: number;
}

export type ToolExecutor = (params: any, context: ExecutionContext) => Promise<{ data: any; chip?: Chip }>;

export class ToolBus {
  private registry: Map<string, { manifest: ToolManifest; executor: ToolExecutor }> = new Map();
  private policyGate: PolicyGate = new PolicyGate();
  private chipRepo: ChipRepository = new ChipRepository(pool);

  /**
   * Register a tool with its manifest and executor
   */
  register(manifest: ToolManifest, executor: ToolExecutor): void {
    this.registry.set(manifest.name, { manifest, executor });
    log.info(`Tool registered: ${manifest.name} v${manifest.version}`);
  }

  /**
   * Execute a tool with policy enforcement and observability
   */
  async execute(
    toolName: string,
    params: any,
    context: ExecutionContext
  ): Promise<ToolResult> {
    const entry = this.registry.get(toolName);
    if (!entry) {
      throw new Error(`Tool ${toolName} not registered`);
    }

    const { manifest, executor } = entry;

    // 1. Policy check (throws if not allowed)
    await this.policyGate.check(manifest, context);

    // 2. Execute tool with latency tracking
    const startTime = Date.now();
    let result: { data: any; chip?: Chip };

    try {
      result = await executor(params, context);
    } catch (error) {
      // Log error execution
      await this.logExecution(toolName, Date.now() - startTime, null, context, 'error');
      throw error;
    }

    const latency = Date.now() - startTime;

    // 3. Validate chip requirement
    if (manifest.requires_chips && !result.chip) {
      throw new Error(`Tool ${toolName} must produce a chip (requires_chips: true)`);
    }

    // 4. Log successful execution
    await this.logExecution(toolName, latency, result.chip, context, 'success');

    // 5. Return result
    return {
      data: result.data,
      chip: result.chip,
      latency_ms: latency,
    };
  }

  /**
   * Log tool execution to agent_runs and system_events
   */
  private async logExecution(
    toolName: string,
    latencyMs: number,
    chip: Chip | null,
    context: ExecutionContext,
    status: 'success' | 'error'
  ): Promise<void> {
    // Log to system_events
    await pool.query(
      `
      INSERT INTO system_events (event_type, trace_id, student_id, coach_id, agent_name, payload)
      VALUES ($1, $2, $3, $4, $5, $6)
      `,
      [
        `tool.${status}`,
        context.trace_id,
        context.student_id,
        context.coach_id,
        context.agent_name,
        JSON.stringify({
          tool_name: toolName,
          latency_ms: latencyMs,
          chip_id: chip?.id,
          status,
        }),
      ]
    );

    // Update agent_run with cumulative usage
    await pool.query(
      `
      UPDATE agent_runs
      SET
        tool_calls = tool_calls + 1,
        total_latency_ms = total_latency_ms + $1
      WHERE id = $2
      `,
      [latencyMs, context.agent_run_id]
    );
  }

  /**
   * Get all registered tools (for debugging)
   */
  getRegisteredTools(): string[] {
    return Array.from(this.registry.keys());
  }
}
```

/**
 * facts-plane.ts
 * SQL-only deterministic fact retrieval
 * Created: 2025-10-21 (Wave 3)
 *
 * Design Principles:
 * - Deterministic: Same input always returns same output
 * - Enumerable: All results are countable sets
 * - Provable: Every result has SQL chip provenance
 * - 412 on Missing: Return 412 error if no data exists
 */

import type { ToolBus } from '../cognition/tool-bus.js';
import type { ExecutionContext } from '../cognition/policy-gate.js';
import { Error412 } from '../errors/evidence-errors.js';
import type { Chip } from '../evidence/chip-creator.js';

export interface FactQuery {
  resolver: string; // Tool name (e.g., 'get_awards_list')
  params: Record<string, any>;
}

export interface FactResult {
  data: any;
  chip: Chip;
}

/**
 * Facts Plane: SQL-only deterministic fact retrieval
 *
 * Contract:
 * - All queries go through Tool Bus (enforced)
 * - All results include SQL chips (enforced)
 * - Missing data returns 412 error (enforced)
 */
export class FactsPlane {
  constructor(private toolBus: ToolBus) {}

  /**
   * Query facts with 412 enforcement
   */
  async query(query: FactQuery, context: ExecutionContext): Promise<FactResult> {
    // Execute through Tool Bus (already creates chips)
    const result = await this.toolBus.execute(query.resolver, query.params, context);

    // Check for missing evidence (412 enforcement)
    if (!result.data || (Array.isArray(result.data) && result.data.length === 0)) {
      throw new Error412(
        query.resolver,
        query.params,
        `No data found for ${query.resolver}. Please add this information to continue.`
      );
    }

    // Validate chip exists
    if (!result.chip) {
      throw new Error(`Facts Plane contract violation: ${query.resolver} did not produce a chip`);
    }

    return {
      data: result.data,
      chip: result.chip,
    };
  }

  /**
   * Query multiple facts in parallel
   */
  async queryMany(queries: FactQuery[], context: ExecutionContext): Promise<FactResult[]> {
    const promises = queries.map(q => this.query(q, context));
    return Promise.all(promises);
  }

  /**
   * Query facts with optional 412 (returns null instead of throwing)
   */
  async queryOptional(query: FactQuery, context: ExecutionContext): Promise<FactResult | null> {
    try {
      return await this.query(query, context);
    } catch (error) {
      if (error instanceof Error412) {
        return null; // Missing data is OK for optional queries
      }
      throw error; // Re-throw other errors
    }
  }
}
