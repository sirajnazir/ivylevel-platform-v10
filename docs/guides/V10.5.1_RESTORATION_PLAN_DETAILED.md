### Step 2.5: Create Production Views
**New file: `db/migrations/2025-10-11-v10.5.2-production-views.sql`**

```sql
-- v10.5.2 Production Views (replace compat layer)
-- Purpose: Clean SQL-friendly views for resolvers

BEGIN;

-- Awards (initial phase)
CREATE OR REPLACE VIEW v_awards_initial AS
SELECT
    student_id,
    item_key AS award_id,
    title_name AS award_name,
    (details_json->>'won_date')::TIMESTAMP AS won_date,
    details_json->>'level' AS level,
    source_id,
    created_at
FROM kb_items
WHERE category = 'award'
  AND (details_json->>'phase') = 'initial';

-- Awards (final phase)
CREATE OR REPLACE VIEW v_awards_final AS
SELECT
    student_id,
    item_key AS award_id,
    title_name AS award_name,
    (details_json->>'won_date')::TIMESTAMP AS won_date,
    details_json->>'level' AS level,
    source_id,
    created_at
FROM kb_items
WHERE category = 'award'
  AND (details_json->>'phase') = 'final';

-- ECs/Activities (final phase)
CREATE OR REPLACE VIEW v_ecs_final AS
SELECT
    student_id,
    item_key AS activity_id,
    title_name AS activity_name,
    subcategory AS category,
    (details_json->>'event_date')::TIMESTAMP AS event_date,
    source_id,
    created_at
FROM kb_items
WHERE category = 'ec'
  AND (details_json->>'phase') = 'final';

-- College List (all tiers)
CREATE OR REPLACE VIEW v_college_list AS
SELECT
    student_id,
    college_id,
    college_name,
    tier,
    attending,
    added_date,
    source_id
FROM college_list;

-- College Decisions (final attending)
CREATE OR REPLACE VIEW v_college_attending AS
SELECT
    student_id,
    college_id,
    college_name,
    tier,
    added_date
FROM college_list
WHERE attending = TRUE;

-- IvyScore Latest
CREATE OR REPLACE VIEW v_readiness_latest AS
SELECT
    student_id,
    as_of_date,
    ivy_score,
    raw_score,
    percentile,
    breakdown_json
FROM readiness_snapshots
WHERE (student_id, as_of_date) IN (
    SELECT student_id, MAX(as_of_date)
    FROM readiness_snapshots
    GROUP BY student_id
);

COMMIT;
```

  }
}

export async function resolveReadinessFeatures(studentId: string) {
  const start = Date.now();
  try {
    const result = await pool.query(
      `SELECT feature_key, weight, current_value, target_value, percentile
       FROM readiness_features
       WHERE student_id = $1
       ORDER BY weight DESC`,
      [studentId]
    );

    logger.info(`[RESOLVER] readiness.features → ${result.rows.length} rows (${Date.now() - start}ms)`);

    return {
      route: 'readiness.features',
      items: result.rows.map(r => ({
        feature_key: r.feature_key,
        weight: r.weight,
        current_value: r.current_value,
        target_value: r.target_value,
        percentile: r.percentile
      }))
    };
  } catch (error) {
    logger.error(`[RESOLVER] readiness.features ERROR: ${error}`);
    throw error;
  }
}
