**5.2 Run Assessment for Each Student**

```bash
# For each student in canary list
while read -r student_id email name; do
  echo "Testing student: $student_id ($name)"

  # Run assessment
  curl -X POST http://staging:3000/api/assessment/run \
    -H "Content-Type: application/json" \
    -d "{\"student_id\": \"$student_id\"}" \
    > /tmp/assessment_${student_id}.json

  if grep -q "error" /tmp/assessment_${student_id}.json; then
    echo "❌ FAILED for $student_id"
  else
    echo "✅ PASSED for $student_id"
  fi

  sleep 5  # Rate limit
done < /tmp/canary_students.txt
```

### Step 6: 48-Hour Soak Test

**6.1 Enable Monitoring**

```bash
# Set up monitoring alerts
cat > /tmp/staging_alerts.yml <<'EOF'
alerts:
  - name: high_error_rate
    condition: error_rate > 5%
    duration: 5m

  - name: mv_refresh_failure
    condition: mv_refresh_failed = true
    duration: 1m

  - name: outbox_backlog
    condition: undelivered_events > 1000
    duration: 10m

  - name: rls_violation
    condition: unauthorized_access_attempt = true
    duration: 1m
EOF
```

**6.2 Run Continuous Load**

```bash
# Run load test for 48 hours
cat > /tmp/load_test.sh <<'EOF'
#!/bin/bash
END_TIME=$(($(date +%s) + 172800))  # 48 hours from now

while [ $(date +%s) -lt $END_TIME ]; do
  # Random student from canary list
  STUDENT=$(shuf -n 1 /tmp/canary_students.txt | awk '{print $1}')

  # Run assessment
  curl -s -X POST http://staging:3000/api/assessment/run \
    -H "Content-Type: application/json" \
    -d "{\"student_id\": \"$STUDENT\"}" \
    >> /tmp/load_test_results.log

  # Random delay (1-10 seconds)
  sleep $((RANDOM % 10 + 1))
done
EOF

chmod +x /tmp/load_test.sh
nohup /tmp/load_test.sh &
```

**6.3 Monitor Key Metrics**

```bash
# Check metrics every hour
cat > /tmp/check_metrics.sh <<'EOF'
#!/bin/bash
while true; do
  echo "=== Metrics Check: $(date) ===" >> /tmp/metrics.log

  # Error rate
  psql $STAGING_DATABASE_URL -c "
    SELECT
      COUNT(*) FILTER (WHERE severity = 'ERROR') AS errors,
      COUNT(*) AS total,
      ROUND(100.0 * COUNT(*) FILTER (WHERE severity = 'ERROR') / COUNT(*), 2) AS error_pct
    FROM system_events
    WHERE created_at > NOW() - INTERVAL '1 hour';
  " >> /tmp/metrics.log

  # MV refresh status
  psql $STAGING_DATABASE_URL -c "
    SELECT event_type, COUNT(*)
    FROM system_events
    WHERE event_type LIKE 'mv.refresh%'
      AND created_at > NOW() - INTERVAL '1 hour'
    GROUP BY event_type;
  " >> /tmp/metrics.log

  # Outbox backlog
  psql $STAGING_DATABASE_URL -c "
    SELECT COUNT(*) AS undelivered
    FROM outbox
    WHERE delivered_at IS NULL;
  " >> /tmp/metrics.log

  sleep 3600  # Check every hour
done
EOF

chmod +x /tmp/check_metrics.sh
nohup /tmp/check_metrics.sh &
```
