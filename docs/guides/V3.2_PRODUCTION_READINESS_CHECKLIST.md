## FIX #3: RLS Session Hygiene

### Problem

### Fix

**1. Session checkout middleware:**

**File:** `services/agent-framework/src/db/pool.ts` (ENHANCED)

```typescript
import { Pool, PoolClient } from 'pg';

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  max: 20,
});

// Enhanced checkout with RLS enforcement
export async function getClientWithRLS(context: {
  student_id: string;
  coach_id: string;
}): Promise<PoolClient> {
  const client = await pool.connect();

  try {
    // Set RLS session variables
    await client.query(`SET LOCAL app.student_id = $1`, [context.student_id]);
    await client.query(`SET LOCAL app.coach_id = $1`, [context.coach_id]);

    return client;
  } catch (error) {
    client.release();
    throw error;
  }
}

// Clear session vars on release
export function releaseClientWithRLS(client: PoolClient): void {
  // PostgreSQL auto-resets LOCAL vars on transaction end
  client.release();
}

export { pool };
```

## FIX #4: Chip Parameter Scrubbing

### Problem

Sensitive data leaks into chip.source.params:


### Fix

**File:** `services/agent-framework/src/chips/chip-creator.ts` (ENHANCED)

```typescript
const PII_PATTERNS = {
  email: /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b/g,
  phone: /\b\d{3}[-.]?\d{3}[-.]?\d{4}\b/g,
  ssn: /\b\d{3}-\d{2}-\d{4}\b/g,
  dob: /\b\d{4}-\d{2}-\d{2}\b/g, // ISO date
  address: /\b\d+\s+[\w\s]+\b(street|st|avenue|ave|road|rd|lane|ln)\b/gi,
};

function scrubParams(params: Record<string, any>): Record<string, any> {
  const scrubbed = { ...params };

  for (const [key, value] of Object.entries(scrubbed)) {
    if (typeof value === 'string') {
      // Scrub PII patterns
      for (const [piiType, pattern] of Object.entries(PII_PATTERNS)) {
        scrubbed[key] = value.replace(pattern, `[REDACTED_${piiType.toUpperCase()}]`);
      }
    }

    // Scrub secret keys
    if (/password|secret|token|api_?key/i.test(key)) {
      scrubbed[key] = '[REDACTED_SECRET]';
    }
  }

  return scrubbed;
}

export class ChipCreator {
  static createSQLChip(
    resolver: string,
    params: Record<string, any>,
    traceId: string,
    studentId: string
  ): Chip {
    const scrubbedParams = scrubParams(params);

    const source = {
      resolver,
      params: scrubbedParams,
      invoked_at: new Date().toISOString(),
    };

    return {
      id: crypto.randomUUID(),
      kind: 'SQL',
      source,
      hash: hashChip(source),
      trace_id: traceId,
      student_id: studentId,
      created_at: new Date(),
    };
  }
}
```

```typescript
import Redis from 'ioredis';
import { pool } from '../db/pool';

const redis = new Redis(process.env.REDIS_URL);

// Run as separate worker process (not in API)
async function processOutbox() {
  const undelivered = await pool.query(
    `SELECT * FROM outbox WHERE delivered_at IS NULL ORDER BY created_at ASC LIMIT 100`
  );

  for (const event of undelivered.rows) {
    const deliveredKey = `outbox:delivered:${event.id}`;

    // Check if already delivered (Redis short-lived key)
    const alreadyDelivered = await redis.get(deliveredKey);
    if (alreadyDelivered) {
      console.log(`[Outbox] Event ${event.id} already delivered, marking...`);
      await markDelivered(event.id);
      continue;
    }

    try {
      await eventBus.publish(event.topic, event.payload);

      // Set short-lived delivered key (TTL 1 hour)
      await redis.setex(deliveredKey, 3600, '1');

      await markDelivered(event.id);
    } catch (error) {
      console.error(`[Outbox] Failed to deliver event ${event.id}:`, error);
      // Retry on next cycle
    }
  }
}

async function markDelivered(eventId: string): Promise<void> {
  await pool.query(
    `UPDATE outbox SET delivered_at = NOW() WHERE id = $1`,
    [eventId]
  );
}

// Run every 5 seconds
setInterval(processOutbox, 5000);
```

## FIX #6: OTel Propagation

### Problem

Traces don't correlate across services:

```typescript

```

### Fix

**File:** `services/agent-framework/src/telemetry/tracer.ts`

```typescript
import { NodeTracerProvider } from '@opentelemetry/sdk-trace-node';
import { BatchSpanProcessor } from '@opentelemetry/sdk-trace-base';
import { OTLPTraceExporter } from '@opentelemetry/exporter-trace-otlp-http';
import { ParentBasedSampler, AlwaysOnSampler } from '@opentelemetry/sdk-trace-base';
import { propagation, trace, context } from '@opentelemetry/api';
import { W3CBaggagePropagator, W3CTraceContextPropagator } from '@opentelemetry/core';

// Parent-based always_on sampler (never drop server calls)
const sampler = new ParentBasedSampler({
  root: new AlwaysOnSampler(),
});

const provider = new NodeTracerProvider({
  sampler,
});

provider.addSpanProcessor(
  new BatchSpanProcessor(
    new OTLPTraceExporter({
      url: process.env.OTEL_EXPORTER_OTLP_ENDPOINT,
    })
  )
);

provider.register({
  propagator: new W3CTraceContextPropagator(),
});

// Set baggage propagator
propagation.setGlobalPropagator(
  new W3CBaggagePropagator()
);

const tracer = trace.getTracer('agent-framework', '1.0.0');

// Enhanced withSpan with baggage
export async function withSpan<T>(
  operationName: string,
  layer: string,
  fn: (span: any) => Promise<T>,
  context_vars?: { student_id?: string; agent_name?: string; coach_id?: string }
): Promise<T> {
  return tracer.startActiveSpan(operationName, async (span) => {
    span.setAttribute('layer', layer);

    // Attach baggage for cross-service correlation
    if (context_vars) {
      const baggageEntries = {
        'app.student_id': context_vars.student_id,
        'app.agent_name': context_vars.agent_name,
        'app.coach_id': context_vars.coach_id,
      };

      // Set baggage in active context
      const currentContext = context.active();
      const baggageContext = propagation.setBaggage(currentContext, baggageEntries);
      context.with(baggageContext, () => {
        // Baggage now propagates to child spans
      });

      // Also add as span attributes for filtering
      for (const [key, value] of Object.entries(baggageEntries)) {
        if (value) span.setAttribute(key, value);
      }
    }

    try {
      const result = await fn(span);
      span.setStatus({ code: 1 }); // OK
      return result;
    } catch (error) {
      span.setStatus({ code: 2, message: error.message }); // ERROR
      span.recordException(error);
      throw error;
    } finally {
      span.end();
    }
  });
}

export { tracer };
```

## FIX #7: EQ Quality & Safety Rails

### Problem

EQ adapter could drift into clown-mode or offensive language:

```typescript
// ❌ WRONG: No quality/safety checks
return styled;
```

### Fix

**File:** `services/agent-framework/src/cognition/eq-adapter.ts` (ENHANCED)

```typescript
import { embed } from '../ml/embeddings';
import { cosineSimilarity } from '../ml/similarity';

export class EQAdapter {
  async apply(
    rawText: string,
    coachId: string,
    audience: string,
    context: { studentId: string; traceId: string }
  ): Promise<{ styledText: string; chip: Chip }> {
    const profile = await this.loadProfile(coachId);
    const tuning = await this.loadTuning(context.studentId, coachId);

    let styled = rawText;

    // 1. Lexical substitutions
    for (const [from, to] of Object.entries(profile.lex_map)) {
      styled = styled.replace(new RegExp(from, 'gi'), to);
    }

    // 2. Tone scalars
    styled = this.applyToneVector(styled, profile.tone_vector, audience);

    // 3. Filter global + coach-specific banned phrases
    const bannedPhrases = [
      ...GLOBAL_BANNED_PHRASES, // Profanity, bias, offensive terms
      ...(tuning?.banned_phrases || []),
    ];
    styled = this.filterBanned(styled, bannedPhrases);

    // 4. Embedding similarity guard (floor ≥ 0.85)
    const rawEmbedding = await embed(rawText);
    const styledEmbedding = await embed(styled);
    const similarity = cosineSimilarity(rawEmbedding, styledEmbedding);

    if (similarity < 0.85) {
      console.warn(`[EQ] Similarity too low (${similarity}), rejecting adaptation`);

      // Log to QA sample table for audit
      await this.logQASample(rawText, styled, similarity, context.traceId);

      // Fall back to raw text
      styled = rawText;
    }

    // 5. Log to QA sample table (monthly audit)
    await this.logQASample(rawText, styled, similarity, context.traceId);

    const chip = ChipCreator.createEQChip(
      coachId,
      profile.tone_vector,
      audience,
      context.traceId,
      context.studentId
    );

    return { styledText: styled, chip };
  }

  private filterBanned(text: string, banned: string[]): string {
    let filtered = text;
    for (const phrase of banned) {
      filtered = filtered.replace(new RegExp(phrase, 'gi'), '***');
    }
    return filtered;
  }

  private async logQASample(
    raw: string,
    styled: string,
    similarity: number,
    traceId: string
  ): Promise<void> {
    await pool.query(
      `
      INSERT INTO eq_qa_samples (trace_id, raw_text, styled_text, similarity_score, created_at)
      VALUES ($1, $2, $3, $4, NOW())
      `,
      [traceId, raw, styled, similarity]
    );
  }
}

// Global banned phrases (profanity, bias, offensive terms)
const GLOBAL_BANNED_PHRASES = [
  // Profanity
  'fuck', 'shit', 'damn', 'crap',
  // Bias/offensive
  'retarded', 'stupid', 'idiot', 'moron',
  // Add more from moderation list
];
```

## FIX #8: Deadline Determinism

### Problem

Tie-breaker undefined for deadline_latest:

```sql

WHERE student_id = $1

```

### Fix

**File:** `services/agent-framework/migrations/002_facts_views_production.sql` (ENHANCED)

```sql
-- ✅ CORRECT: Stable tie-breaker
CREATE OR REPLACE FUNCTION deadline_latest(p_student_id text)
RETURNS TABLE(
  college_name text,
  deadline_date date,
  deadline_type text,
  priority integer,
  created_at timestamptz
) AS $$
  SELECT
    college_name,
    deadline_date,
    deadline_type,
    priority,
    created_at
  FROM v_deadlines_facts
  WHERE student_id = p_student_id
  ORDER BY
    deadline_date ASC,
    priority DESC,        -- Higher priority first on tie
    created_at ASC,       -- Older entries first
```

### Problem

### Fix

**1. Feature flag + phased rollout:**

**File:** `services/agent-framework/src/resolvers/readiness.ts` (ENHANCED)

```typescript
export async function computeIvyScore(
  studentId: string,
  opts: { scoring_version?: number } = {}
): Promise<{ score: number; version: number }> {
  const academicScore = await getAcademicScore(studentId);
  const achievementScore = await getAchievementScore(studentId);
  const hgtiScore = await getHGTIScore(studentId);

  // Feature flag: phased HGTI rollout (10% → 20% → 28%)
  const hgtiWeight = getHGTIWeight(opts.scoring_version);

  // Rebalance academic/achievement to sum to (1 - hgtiWeight)
  const academicWeight = 0.40 * (1 - hgtiWeight) / 0.72;
  const achievementWeight = 0.32 * (1 - hgtiWeight) / 0.72;

  const ivyScore =
    academicScore * academicWeight +
    achievementScore * achievementWeight +
    hgtiScore * hgtiWeight;

  const version = opts.scoring_version || CURRENT_SCORING_VERSION;

  // Persist with version for historical reconstruction
  await pool.query(
    `
    INSERT INTO ivyscore_history (student_id, score, version, computed_at)
    VALUES ($1, $2, $3, NOW())
    `,
    [studentId, ivyScore, version]
  );

  return { score: ivyScore, version };
}

function getHGTIWeight(version?: number): number {
  const v = version || CURRENT_SCORING_VERSION;

  // Phased rollout schedule (5 weeks)
  if (v === 1) return 0.00; // Week 1: No HGTI (baseline)
  if (v === 2) return 0.10; // Week 2-3: 10%
  if (v === 3) return 0.20; // Week 4: 20%
  if (v === 4) return 0.28; // Week 5+: 28% (target)

  return 0.28; // Default to target
}

// Feature flag (update via env var or DB config)
let CURRENT_SCORING_VERSION = parseInt(process.env.IVYSCORE_VERSION || '1', 10);
```

## FIX #10: Ajv Manifest Versioning

### Problem

Schema changes break old tool manifests:

```typescript
// ❌ WRONG: No versioning
const validator = ajv.compile(manifest.schema);
```

### Fix

**File:** `services/agent-framework/src/governance/tool-bus.ts` (ENHANCED)

```typescript
import Ajv from 'ajv';

const ajv = new Ajv({ allErrors: true });

// Versioned validators cache
const validatorCache = new Map<string, any>();

export interface ToolManifest {
  name: string;
  schema_version: string; // e.g., "1.0.0"
  schema: {
    type: 'object';
    properties: Record<string, any>;
    required?: string[];
  };
}

function getValidator(manifest: ToolManifest): any {
  const cacheKey = `${manifest.name}@${manifest.schema_version}`;

  if (validatorCache.has(cacheKey)) {
    return validatorCache.get(cacheKey);
  }

  const validator = ajv.compile(manifest.schema);
  validatorCache.set(cacheKey, validator);

  return validator;
}

export class ToolBus {
  async execute(toolName: string, params: any, context: ExecutionContext): Promise<ToolResult> {
    const manifest = this.getManifest(toolName);

    // Validate against versioned schema
    const validator = getValidator(manifest);

    if (!validator(params)) {
      throw new ValidationError(
        `Invalid params for ${toolName}@${manifest.schema_version}: ${JSON.stringify(validator.errors)}`
      );
    }

    // ... execution
  }

  private getManifest(toolName: string): ToolManifest {
    const manifests: Record<string, ToolManifest> = {
      award_nth: {
        name: 'award_nth',
        schema_version: '1.0.0',
        schema: {
          type: 'object',
          properties: {
            student_id: { type: 'string' },
            n: { type: 'integer', minimum: 1 },
            as_of: { type: 'string', format: 'date' },
          },
          required: ['student_id', 'n'],
        },
      },
      // ... other manifests
    };

    return manifests[toolName];
  }
}
```

**CI check for breaking changes:**

```typescript
// tests/manifests/version-check.test.ts
describe('Tool Manifest Version Checks', () => {
  it('should fail CI on incompatible schema changes', () => {
    const oldManifest = loadManifestFromCommit('HEAD~1', 'award_nth');
    const newManifest = loadManifestFromCommit('HEAD', 'award_nth');

    // Check for breaking changes
    const breakingChanges = detectBreakingChanges(oldManifest.schema, newManifest.schema);

    if (breakingChanges.length > 0 && oldManifest.schema_version === newManifest.schema_version) {
      throw new Error(
        `Breaking schema changes detected but schema_version unchanged. Increment version!`
      );
    }
  });
});

function detectBreakingChanges(oldSchema: any, newSchema: any): string[] {
  const breaking: string[] = [];

  // Removed required field
  const removedRequired = oldSchema.required?.filter(
    (field: string) => !newSchema.required?.includes(field)
  );
  if (removedRequired?.length > 0) {
    breaking.push(`Removed required fields: ${removedRequired.join(', ')}`);
  }

  // Type changes
  for (const [field, oldProp] of Object.entries(oldSchema.properties)) {
    const newProp = newSchema.properties[field];
    if (newProp && oldProp.type !== newProp.type) {
      breaking.push(`Type change for ${field}: ${oldProp.type} → ${newProp.type}`);
    }
  }

  return breaking;
}
```

- ✅ CI fails on incompatible changes without version bump
