# Script: tools/ingest/migrate_kb_chips_to_eq.py

import json
import psycopg2
from pathlib import Path

# Connect to database
conn = psycopg2.connect("postgresql://postgres:postgres@localhost:5432/ivylevel")
cur = conn.cursor()

# Parse all KB intel chip files
kb_chips_dir = Path("/Users/snazir/ivylevel-platform-v10/data/kb_intel_chips/chips")

for chip_file in kb_chips_dir.glob("w*.json"):
    with open(chip_file) as f:
        data = json.load(f)

    # Extract week number from filename (e.g., w010 → week 10)
    week_num = int(chip_file.stem.split('_')[0][1:])  # "w010" → 10

    # Check if this is array or single object
    chips = data if isinstance(data, list) else [data]

    for chip in chips:
        # Create EQ chip from KB intel chip
        eq_chip = {
            "student_id": "huda-2025",  # Assume Huda since it's Jenny's main student
            "kind": "EQ",
            "source": {
                "session_id": f"w{week_num:03d}_migration",
                "week": week_num,
                "kb_chip_id": chip.get("chip_id", f"w{week_num:03d}_unknown"),
                "coaching_insight": chip.get("content", chip.get("insight", "")),
                "eq_signals": {
                    "self_awareness": extract_eq_signal(chip, "self_awareness"),
                    "self_regulation": extract_eq_signal(chip, "self_regulation"),
                    "motivation": extract_eq_signal(chip, "motivation"),
                    "empathy": extract_eq_signal(chip, "empathy"),
                    "social_skills": extract_eq_signal(chip, "social_skills")
                },
                "coaching_context": chip.get("coaching_context", {}),
                "tags": chip.get("tags", []),
                "migration_source": "kb_intel_chips"
            },
            "hash": md5(json.dumps(chip, sort_keys=True)),
            "trace_id": f"migration_kb_w{week_num:03d}"
        }

        # Insert into chips table
        cur.execute("""
            INSERT INTO chips (id, student_id, kind, source, hash, trace_id, created_at)
            VALUES (gen_random_uuid(), %s, %s, %s::jsonb, %s, %s, NOW())
            ON CONFLICT (hash) DO NOTHING
        """, (
            eq_chip["student_id"],
            eq_chip["kind"],
            json.dumps(eq_chip["source"]),
            eq_chip["hash"],
            eq_chip["trace_id"]
        ))

conn.commit()
print(f"Migrated {len(chips)} KB intel chips to EQ chips")
