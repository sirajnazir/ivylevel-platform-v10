**File:** `services/agent-framework/src/governance/tool-bus.ts`

```typescript
export class ToolBus {
  async execute(toolName: string, params: any, context: ExecutionContext): Promise<ToolResult> {
    const idempotencyKey = `${context.agent_run_id}:${toolName}:${Date.now()}`;

    const client = await pool.connect();
    try {
      await client.query('BEGIN');

      // Write to outbox with idempotency
      await client.query(
        `INSERT INTO outbox (topic, payload, idempotency_key)
         VALUES ($1, $2, $3) ON CONFLICT (idempotency_key) DO NOTHING`,
        ['tool.executed', { toolName, params }, idempotencyKey]
      );

      const result = await tools[toolName](params, context);
      await client.query('COMMIT');
      return result;
    } catch (error) {
      await client.query('ROLLBACK');
      throw error;
    } finally {
      client.release();
    }
  }
}
```

export class EQAdapter {
  async apply(rawText: string, coachId: string): Promise<string> {
    const profile = await this.loadProfile(coachId);

    let styled = rawText;
    for (const [from, to] of Object.entries(profile.lex_map)) {
      styled = styled.replace(new RegExp(from, 'gi'), to);
    }

    return styled;
  }

```typescript
export async function withSpan<T>(
  name: string,
  fn: (span: Span) => Promise<T>
): Promise<T> {
  return tracer.startActiveSpan(name, async (span) => {
    try {
      const result = await fn(span);
      span.setStatus({ code: SpanStatusCode.OK });
      return result;
    } finally {
      span.end();
    }
  });
}
```
