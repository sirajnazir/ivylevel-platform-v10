## 🚀 Quick Start

### 1. Access Dev Server
```bash
# Already running at:
http://localhost:5175/
```

### 2. Import Components
```tsx
// Evidence chips with provenance
import { EvidencePanel } from '@/components/v3.2/EvidencePanel';

// 412 missing evidence guidance
import { MissingEvidenceCard } from '@/components/v3.2/MissingEvidenceCard';

// Growth barrier scores
import { HGTIScoreCard } from '@/components/v3.2/HGTIScoreCard';

// Feature flags
import { useFeatureFlag } from '@/utils/featureFlags';
```

### 3. Use in Components
```tsx
function StudentDashboard() {
  const { student } = useStudent();
  const showEvidence = useFeatureFlag('evidencePanel');
  const showHGTI = useFeatureFlag('hgtiGraph');

  return (
    <div className="space-y-6">
      {showHGTI && <HGTIScoreCard studentId={student.id} mode="cached" />}
      {showEvidence && <EvidencePanel studentId={student.id} />}
    </div>
  );
}
```

**Props:**

## 🛠️ Common Patterns

### Conditional Rendering
```tsx
const showEvidence = useFeatureFlag('evidencePanel');

{showEvidence && <EvidencePanel studentId={student.id} />}
```

### Error Handling with 412 UI
```tsx
const [missingEvidence, setMissingEvidence] = useState(null);
const show412 = useFeatureFlag('missingEvidence');

async function loadGPA() {
  try {
    const fact = await v32ApiService.getFact(studentId, 'gpa');
    setGPA(fact.value);
    setMissingEvidence(null);
  } catch (err) {
    if (err.status === 412 && show412) {
      setMissingEvidence(err.data);
    } else {
      setError('Failed to load GPA');
    }
  }
}

// In render:
{missingEvidence && <MissingEvidenceCard missing={missingEvidence} />}
```

### Loading States
```tsx
const [loading, setLoading] = useState(true);

{loading ? (
  <div className="animate-pulse">
    <div className="h-4 bg-gray-200 rounded w-1/2 mb-4"></div>
  </div>
) : (
  <EvidencePanel studentId={student.id} />
)}
```

## 🚨 Troubleshooting

### Component Not Rendering
1. Check feature flag is enabled:
   ```tsx
   console.log(useFeatureFlag('evidencePanel')); // true?
   ```

2. Check `.env` file:
   ```bash
   cat .env | grep VITE_FEATURE_EVIDENCE_PANEL
   # Should show: VITE_FEATURE_EVIDENCE_PANEL=true
   ```

3. Restart dev server:
   ```bash
   npm run dev
   ```

### 412 Card Not Showing
1. Verify feature flag:
   ```tsx
   console.log(useFeatureFlag('missingEvidence')); // true?
   ```

2. Check error status:
   ```tsx
   console.log(err.status); // 412?
   ```

3. Verify API response format:
   ```tsx
   console.log(err.data); // { missing: [...], suggested_actions: [...] }
   ```

### API Not Responding
1. Check backend is running:
   ```bash
   curl http://localhost:4101/api/health/liveness
   ```

2. Start backend if needed:
   ```bash
   cd services/agent-framework
   npm run start
   ```
