## Phase 3: Evidence Panel (Week 0 - Read-Only)

### Step 3.1: Create Evidence Panel Component

```typescript
// src/components/v3.2/EvidencePanel.tsx
import { useEffect, useState } from 'react';
import { v32ApiService, Chip } from '@/services/v3.2ApiService';

interface EvidencePanelProps {
  studentId: string;
}

export function EvidencePanel({ studentId }: EvidencePanelProps) {
  const [chips, setChips] = useState<Chip[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    // Set RLS context
    sessionStorage.setItem('current_student_id', studentId);

    // Fetch chips
    v32ApiService.getChips(studentId, { limit: 50 })
      .then(setChips)
      .catch(err => setError(err.message))
      .finally(() => setLoading(false));
  }, [studentId]);

  if (loading) return <div>Loading evidence...</div>;
  if (error) return <div className="text-red-500">Error: {error}</div>;

  return (
    <div className="space-y-4">
      <h2 className="text-xl font-bold">Evidence Chips</h2>
      <p className="text-sm text-gray-600">
        Shows where claims about this student come from
      </p>

      <div className="space-y-2">
        {chips.map(chip => (
          <ChipCard key={chip.id} chip={chip} />
        ))}
      </div>

      {chips.length === 0 && (
        <div className="text-gray-500 italic">
          No evidence chips yet. Claims will appear here as they're verified.
        </div>
      )}
    </div>
  );
}

function ChipCard({ chip }: { chip: Chip }) {
  const kindColors = {
    SQL: 'bg-blue-100 text-blue-800',
    RAG: 'bg-green-100 text-green-800',
    LLM: 'bg-purple-100 text-purple-800',
    EQ: 'bg-pink-100 text-pink-800',
    NARRATIVE: 'bg-yellow-100 text-yellow-800',
  };

  return (
    <div className="p-4 border rounded hover:shadow-md transition-shadow cursor-pointer">
      <div className="flex justify-between items-start">
        <div>
          <span className={`px-2 py-1 rounded text-xs font-medium ${kindColors[chip.kind]}`}>
            {chip.kind}
          </span>
          <p className="mt-2 text-sm text-gray-600">
            {new Date(chip.created_at).toLocaleDateString()}
          </p>
        </div>
        <button 
          className="text-blue-600 text-sm hover:underline"
          onClick={() => window.open(`/trace/${chip.trace_id}`, '_blank')}
        >
          View Trace →
        </button>
      </div>
      <pre className="mt-2 text-xs bg-gray-50 p-2 rounded overflow-auto max-h-32">
        {JSON.stringify(chip.source, null, 2)}
      </pre>
    </div>
  );
}
```

### Step 4.2: Use in Data Fetching

```typescript
// Example: Fetching deadline with 412 handling
import { v32ApiService, MissingEvidence } from '@/services/v3.2ApiService';

function DeadlineWidget({ studentId }: { studentId: string }) {
  const [data, setData] = useState<any>(null);
  const [missing, setMissing] = useState<MissingEvidence | null>(null);

  useEffect(() => {
    v32ApiService.getDeadlineLatest(studentId).then(result => {
      if ('status' in result && result.status === 412) {
        setMissing(result as MissingEvidence);
      } else {
        setData(result);
      }
    });
  }, [studentId]);

  if (missing) {
    return <MissingEvidenceCard missing={missing} />;
  }

  if (!data) {
    return <div>Loading...</div>;
  }

  return (
    <div>
      <h3>Next Deadline</h3>
      <p>{data.college_name}: {data.deadline_date}</p>
    </div>
  );
}
```

## Phase 5: HGTI Graph (Week 0 - Read-Only)

### Step 5.1: Create HGTI Score Display

```typescript
// src/components/v3.2/HGTIScoreCard.tsx
import { useEffect, useState } from 'react';
import { v32ApiService, HGTIScore } from '@/services/v3.2ApiService';

export function HGTIScoreCard({ studentId }: { studentId: string }) {
  const [score, setScore] = useState<HGTIScore | null>(null);
  const [mode, setMode] = useState<'cached' | 'realtime'>('cached');

  useEffect(() => {
    v32ApiService.getHGTIScore(studentId, mode).then(setScore);
  }, [studentId, mode]);

  if (!score) return <div>Loading HGTI...</div>;

  return (
    <div className="p-6 border rounded-lg">
      <div className="flex justify-between items-start">
        <div>
          <h3 className="text-lg font-bold">Growth Score (HGTI)</h3>
          <p className="text-3xl font-bold mt-2">{score.score.toFixed(2)}</p>
          <p className="text-xs text-gray-500 mt-1">
            Calculated {new Date(score.calculated_at).toLocaleString()}
          </p>
        </div>

        <button
          onClick={() => setMode(mode === 'cached' ? 'realtime' : 'cached')}
          className="text-xs px-2 py-1 border rounded hover:bg-gray-50"
        >
          {mode === 'cached' ? '↻ Refresh' : '⏱ Cached'}
        </button>
      </div>

      <div className="mt-4 space-y-2">
        {score.breakdown.map(item => (
          <div key={item.barrier_type} className="flex justify-between text-sm">
            <span className="text-gray-600">{item.barrier_type}</span>
            <span className="font-medium">{item.contribution.toFixed(2)}</span>
          </div>
        ))}
      </div>

      <p className="mt-4 text-xs text-gray-500 italic">
        HGTI updates every ~5 minutes. Don't panic if it doesn't change instantly.
      </p>
    </div>
  );
}
```

## Phase 6: EQ Layer (Week 2 - Shadow Mode)

### Step 6.1: Create EQ Preview Toggle

```typescript
// src/components/v3.2/EQPreviewToggle.tsx
import { useState } from 'react';
import { v32ApiService, EQPreview } from '@/services/v3.2ApiService';

export function EQPreviewToggle({ coachId, audience }: {
  coachId: string;
  audience: 'student' | 'parent';
}) {
  const [text, setText] = useState('');
  const [preview, setPreview] = useState<EQPreview | null>(null);
  const [showOriginal, setShowOriginal] = useState(false);

  const handlePreview = async () => {
    const result = await v32ApiService.previewEQ({ text, coach_id: coachId, audience });
    setPreview(result);
  };

  return (
    <div className="space-y-4">
      <textarea
        value={text}
        onChange={e => setText(e.target.value)}
        className="w-full p-3 border rounded"
        rows={4}
        placeholder="Type your message here..."
      />

      <button
        onClick={handlePreview}
        className="px-4 py-2 bg-blue-600 text-white rounded hover:bg-blue-700"
      >
        Preview with EQ Styling
      </button>

      {preview && (
        <div className="space-y-3">
          <div className="flex items-center justify-between">
            <label className="flex items-center gap-2">
              <input
                type="checkbox"
                checked={showOriginal}
                onChange={e => setShowOriginal(e.target.checked)}
                className="rounded"
              />
              <span className="text-sm">Show original</span>
            </label>

            {preview.fallback_used && (
              <span className="text-xs px-2 py-1 bg-yellow-100 text-yellow-800 rounded">
                ⚠ Style fallback used
              </span>
            )}
          </div>

          <div className="p-4 bg-gray-50 rounded border">
            <p className="text-sm">
              {showOriginal ? preview.original_text : preview.eq_styled_text}
            </p>
          </div>

          <div className="text-xs text-gray-600">
            Similarity: {(preview.similarity_score * 100).toFixed(1)}%
            {preview.similarity_score < 0.85 && ' (below threshold - fallback applied)'}
          </div>
        </div>
      )}
    </div>
  );
}
```
