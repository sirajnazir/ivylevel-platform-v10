# v12.0 - Assessment & Scoring Enhancement - Specification

**Version:** v12.0
**Date:** 2025-10-28
**Status:** 📋 SPECIFICATION - Pending Review
**Tab:** Assessment (First Tab)
**Integration:** Existing ring simulation UI + new components

---

## 🎯 Executive Summary

This specification enhances the existing **Assessment tab** with comprehensive profile assessment and readiness scoring that integrates seamlessly with the current ring simulation design.

**Current Assessment Tab Shows:**
- Ring simulation: "How Ivy+ Ready are you today?"
- T20 target: "85% to target"
- Profile Target Level: "IVY+" (87%)
- Overall Score with trend: "+2.3% vs last 180 days"
- Achievement level: "BRONZE" with "6.6% score boost" potential
- "Boost Score" CTA

**This Spec Adds:**
- Dimensional scoring breakdown (Academic, EC, Awards, Service, Essays, Recommendations)
- Profile assessment with standout strengths
- Prioritized weak spots with ROI rankings
- Target profile visualization (Three-pillar model: Aptitude + Passion + Service)
- Historical score trending over time
- Integration with game plan for improvement roadmap

## Part 2: Enhanced Assessment Data Model

### AssessmentScore

```typescript
interface AssessmentScore {
  assessment_id: string;
  student_id: string;
  assessment_date: string; // ISO 8601

  // Overall metrics (maps to existing ring UI)
  overall_score: number; // 0-100 (currently showing 87%)
  target_level: 'IVY+' | 'T20' | 'T50' | 'T100';
  target_percentage: number; // e.g., 85% to target

  // Trend data (maps to existing "+2.3% vs last 180 days")
  trend_vs_180_days: number; // Percentage change
  trend_direction: 'up' | 'down' | 'stable';

  // Achievement system (maps to existing "BRONZE")
  achievement_level: 'BRONZE' | 'SILVER' | 'GOLD' | 'PLATINUM';
  score_boost_potential: number; // e.g., 6.6%

  // NEW: Dimensional breakdown
  dimensional_scores: DimensionalScore[];

  // NEW: Profile assessment
  profile_assessment: ProfileAssessment;

  // NEW: Recommendations for boost
  boost_recommendations: BoostRecommendation[];
}

interface DimensionalScore {
  dimension: 'academic' | 'extracurricular' | 'awards' | 'service' | 'essays' | 'recommendations' | 'testing';
  score: number; // 0-100
  weight: number; // How much this dimension contributes to overall score
  current_level: string; // e.g., "Strong", "Competitive", "Needs Improvement"
  target_level: string; // e.g., "Exceptional" for Ivy+
  gap: number; // Points needed to reach target
  improvement_potential: number; // 0-100 (how much realistically improvable)
}

interface ProfileAssessment {
  standout_strengths: Strength[];
  weak_spots: WeakSpot[];
  unique_story: string;
  competitive_positioning: string; // vs typical Ivy+ applicant
}

interface Strength {
  strength_id: string;
  category: 'academic' | 'extracurricular' | 'creative' | 'leadership' | 'entrepreneurial';
  title: string;
  description: string;
  evidence: string[];
  competitive_advantage: string; // Why this matters for admissions
}

interface WeakSpot {
  weak_spot_id: string;
  category: 'academic' | 'extracurricular' | 'awards' | 'service' | 'testing' | 'relationships';
  title: string;
  description: string;
  priority: 'P0' | 'P1' | 'P2'; // P0 = highest ROI
  roi_rationale: string;
  score_impact: number; // How many % points this costs
  estimated_time_to_address: string; // e.g., "3 months"
  actionable_next_steps: string[];
}

interface BoostRecommendation {
  recommendation_id: string;
  title: string;
  category: 'quick_win' | 'medium_effort' | 'long_term';
  estimated_score_boost: number; // Percentage points
  time_required: string;
  difficulty: 'easy' | 'medium' | 'hard';
  linked_weak_spot_id?: string;
  linked_game_plan_tactical_plan_id?: string;
  action_items: string[];
}
```

### TargetProfile

```typescript
interface TargetProfile {
  profile_id: string;
  profile_name: string; // e.g., "AI Educator & Creative Technologist"
  profile_archetype: 'technical' | 'creative' | 'leadership' | 'service' | 'hybrid';

  three_pillar_model: {
    aptitude: Pillar;
    passion: Pillar;
    service: Pillar;
  };

  narrative: string;
  differentiation: string;
  fit_with_target_schools: string; // Why this profile works for Ivy+
}

interface Pillar {
  title: string;
  description: string;
  current_strength: number; // 0-100
  target_strength: number; // 0-100
  key_activities: string[];
  evidence_needed: string[];
}
```

## Part 5: Database Schema

### New Table: assessment_scores

```sql
CREATE TABLE assessment_scores (
  assessment_id TEXT PRIMARY KEY,
  student_id TEXT NOT NULL REFERENCES students(student_id),
  assessment_date TIMESTAMPTZ NOT NULL,

  -- Overall metrics (maps to ring UI)
  overall_score NUMERIC(5,2) NOT NULL CHECK (overall_score >= 0 AND overall_score <= 100),
  target_level TEXT NOT NULL CHECK (target_level IN ('IVY+', 'T20', 'T50', 'T100')),
  target_percentage NUMERIC(5,2) NOT NULL,

  -- Trend data
  trend_vs_180_days NUMERIC(5,2),
  trend_direction TEXT CHECK (trend_direction IN ('up', 'down', 'stable')),

  -- Achievement system
  achievement_level TEXT CHECK (achievement_level IN ('BRONZE', 'SILVER', 'GOLD', 'PLATINUM')),
  score_boost_potential NUMERIC(5,2),

  -- JSONB for flexible data
  dimensional_scores JSONB NOT NULL,
  profile_assessment JSONB NOT NULL,
  boost_recommendations JSONB,

  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_assessment_scores_student ON assessment_scores(student_id);
CREATE INDEX idx_assessment_scores_date ON assessment_scores(student_id, assessment_date DESC);
CREATE INDEX idx_assessment_scores_dimensional ON assessment_scores USING GIN (dimensional_scores);
```

### New Table: target_profiles

```sql
CREATE TABLE target_profiles (
  profile_id TEXT PRIMARY KEY,
  student_id TEXT NOT NULL REFERENCES students(student_id),
  profile_name TEXT NOT NULL,
  profile_archetype TEXT NOT NULL CHECK (profile_archetype IN ('technical', 'creative', 'leadership', 'service', 'hybrid')),

  three_pillar_model JSONB NOT NULL, -- {aptitude, passion, service}

  narrative TEXT NOT NULL,
  differentiation TEXT NOT NULL,
  fit_with_target_schools TEXT,

  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_target_profiles_student ON target_profiles(student_id);
```
