# V3.3: Enhanced Academics Detection with Keyword Floor

## Problem Statement

Queries like "academic stats?", "grades", and "I meant grades" were returning `unknown` intent because:
1. LLM didn't have enough examples for generic academics queries
2. No fallback detection for clear academic keywords
3. Confidence threshold (0.75) was too high for ambiguous phrasing

## Solution: Keyword Floor Detection

Add deterministic keyword matching as a safety net before LLM classification.

### Architecture

```
User Query
   ↓
Keyword Floor Check (deterministic)
   ├─ Match? → Return intent with confidence=0.66, detector="keyword-floor"
   └─ No match? → Pass to LLM classifier
           ↓
       LLM Classification (gpt-4o-mini, JSON mode)
           ↓
       Safety Override: If LLM missed academics but keywords present → Force academics
           ↓
       Low Confidence Check (< 0.45) → Ask clarification
           ↓
       Route to SQL Resolver
```

## Implementation Details

### 1. Keyword Floor Function

```typescript
const ACADEMIC_KEYWORDS = [
  'academic', 'academics', 'academic stats', 'report card', 'transcript',
  'grade', 'grades', 'gpa', 'sat', 'act', 'test scores', 'scores'
];

const ACADEMIC_METRIC_MAP: Record<string, AcademicMetric> = {
  'grade': 'grades', 'grades': 'grades',
  'gpa': 'gpa', 'transcript': 'transcript',
  'sat': 'sat', 'act': 'act'
};

function keywordFloor(query: string): Partial<IntentData> | null {
  const t = query.toLowerCase();

  // Check if any academic keyword present
  const hasAcademics = ACADEMIC_KEYWORDS.some(k => t.includes(k));
  if (!hasAcademics) return null;

  // Extract specific metrics mentioned
  const metrics: AcademicMetric[] = [];
  for (const [keyword, metric] of Object.entries(ACADEMIC_METRIC_MAP)) {
    if (t.includes(keyword)) metrics.push(metric);
  }

  // Default to comprehensive if vague "academic stats"
  if (metrics.length === 0 && (t.includes('academic') || t.includes('stats'))) {
    metrics.push('grades', 'gpa', 'sat');
  }

  // Detect phase
  let phase: Phase | undefined;
  if (t.includes('initial') || t.includes('first') || t.includes('gameplan')) {
    phase = 'initial';
  } else if (t.includes('final') || t.includes('submitted') || t.includes('application')) {
    phase = 'final';
  }

  // Detect nth
  let nth: number | undefined;
  const nthMatch = t.match(/\b(first|second|third|fourth|fifth|\d+(?:st|nd|rd|th))\b/i);
  if (nthMatch) {
    const ordinalMap: Record<string, number> = {
      'first': 1, 'second': 2, 'third': 3, 'fourth': 4, 'fifth': 5
    };
    const word = nthMatch[1].toLowerCase();
    nth = ordinalMap[word] || parseInt(word);
    phase = 'nth';
  }

  // Determine intent
  let intent: Intent;
  if (metrics.length === 1 && metrics[0] === 'sat') {
    intent = 'academics.sat';
  } else if (metrics.length === 1 && metrics[0] === 'gpa') {
    intent = 'academics.gpa';
  } else if (metrics.length === 1 && metrics[0] === 'transcript') {
    intent = 'academics.transcript';
  } else if (metrics.length === 1 && (metrics[0] === 'grades')) {
    intent = 'academics.grades';
  } else {
    intent = 'academics.summary'; // Multiple metrics or vague
  }

  return {
    intent,
    entity: 'academics',
    phase: phase || 'none',
    slots: { metrics, nth },
    confidence: 0.66,
    detector: 'keyword-floor'
  };
}
```

### 2. Updated classifyIntent()

```typescript
async function classifyIntent(user: string) {
  // 1. Try keyword floor first
  const kwFloor = keywordFloor(user);
  if (kwFloor) {
    log.event('intent.keyword_floor', { query: user.slice(0, 50), result: kwFloor });
    return IntentSchema.parse(kwFloor);
  }

  // 2. LLM classification (existing code)
  const examples = FEW_SHOT.map(e =>
    `USER: ${e.u}\\nLABEL: ${JSON.stringify({...e.a, slots:e.a.slots ?? {}, confidence:0.95})}`
  ).join("\\n\\n");
  const prompt = `${examples}\\n\\nUSER: ${user}\\nLABEL:`;

  const r = await openai.chat.completions.create({
    model: "gpt-4o-mini",
    temperature: 0,
    response_format: { type: "json_object" },
    messages: [
      { role: "system", content: SYS },
      { role: "user", content: prompt }
    ]
  });

  const raw = r.choices[0]?.message?.content ?? "{}";
  let parsed;
  try {
    parsed = JSON.parse(raw);
    parsed.detector = 'llm';
  } catch {
    parsed = { intent:"unknown", entity:"kb", phase:"none", slots:{}, confidence:0.0, detector:"llm" };
  }

  // 3. Safety override: If LLM didn't detect academics but keywords present
  const kwCheck = keywordFloor(user);
  if (kwCheck && parsed.entity !== 'academics') {
    log.event('intent.safety_override', { llm_entity: parsed.entity, keyword_entity: 'academics' });
    return IntentSchema.parse(kwCheck);
  }

  return IntentSchema.parse(parsed);
}
```

```typescript
- NEVER hallucinate; keep confidence low if unclear.`;
```

### 5. New SQL Views

```sql
-- V_ACADEMICS_INITIAL: Pull from kb_items
DROP VIEW IF EXISTS v_academics_initial CASCADE;
CREATE VIEW v_academics_initial AS
SELECT
  student_id,
  subtype AS metric,        -- 'grades' | 'gpa' | 'sat'
  title_name AS value_text, -- "UW 3.92, W 4.28" or "1480"
  source_ref,
  item_id AS chip_id
FROM kb_items
WHERE item_type = 'academics'
  AND tier1_state = 'Planned'
ORDER BY student_id, subtype;

-- V_ACADEMICS_FINAL: Pull from kb_items
DROP VIEW IF EXISTS v_academics_final CASCADE;
CREATE VIEW v_academics_final AS
SELECT
  student_id,
  subtype AS metric,
  title_name AS value_text,
  source_ref,
  item_id AS chip_id
FROM kb_items
WHERE item_type = 'academics'
  AND tier1_state = 'Submitted'
ORDER BY student_id, subtype;
```

### 6. New Resolvers

```typescript
export async function academicsGrades(pg: Pool, studentId: string, phase: string) {
  const start = Date.now();
  log.event('resolver.sql_start', { resolver: 'academicsGrades', student_id: studentId, phase });

  const view = phase === 'final' ? 'v_academics_final' : 'v_academics_initial';
  const { rows } = await pg.query(
    `SELECT * FROM ${view} WHERE student_id=$1 AND metric IN ('grades','gpa') ORDER BY metric`,
    [studentId]
  );

  log.event('resolver.sql_complete', { resolver: 'academicsGrades', view, row_count: rows.length, took_ms: Date.now() - start });

  if (!rows.length) {
    return { answer: `No ${phase} grades/GPA found.`, chips:[], hits:[] };
  }

  const answer = rows.map((r: any) => `• ${r.metric}: ${r.value_text}`).join('\\n');
  const chips = rows.map((r: any) => ({kind:"evidence", text:r.metric, id:r.chip_id}));

  return { answer, chips, hits: rows };
}

export async function academicsSummary(pg: Pool, studentId: string, phase: string, metrics?: AcademicMetric[]) {
  const start = Date.now();
  log.event('resolver.sql_start', { resolver: 'academicsSummary', student_id: studentId, phase, metrics });

  const parts: string[] = [];
  const allChips: any[] = [];
  const allHits: any[] = [];

  // Fetch grades/GPA if requested
  if (!metrics || metrics.includes('grades') || metrics.includes('gpa')) {
    const view = phase === 'final' ? 'v_academics_final' : 'v_academics_initial';
    const { rows } = await pg.query(
      `SELECT * FROM ${view} WHERE student_id=$1 AND metric IN ('grades','gpa') ORDER BY metric`,
      [studentId]
    );
    if (rows.length) {
      parts.push(...rows.map((r: any) => `• ${r.metric}: ${r.value_text}`));
      allChips.push(...rows.map((r: any) => ({kind:"evidence", text:r.metric, id:r.chip_id})));
      allHits.push(...rows);
    }
  }

  // Fetch SAT if requested
  if (!metrics || metrics.includes('sat')) {
    const satData = await academicsSAT(pg, studentId, phase === 'initial' ? 'first' : 'latest', {});
    if (satData.answer && !satData.answer.includes('No SAT')) {
      parts.push(`• SAT: ${satData.answer}`);
      allChips.push(...satData.chips);
      allHits.push(...satData.hits);
    }
  }

  log.event('resolver.sql_complete', { resolver: 'academicsSummary', parts_count: parts.length, took_ms: Date.now() - start });

  if (parts.length === 0) {
    return { answer: `No ${phase} academic data found.`, chips:[], hits:[] };
  }

  return {
    answer: parts.join('\\n'),
    chips: allChips,
    hits: allHits
  };
}
```
